/* ---------- keyframes: [t, {props}] cumulative; numbers & arrays ease (smoothstep, or linear with _e:'lin'); other values snap at their key */
function KF(list) {
  const out = []; let cur = {};
  for (const [t, p] of list) { const { _e, ...q } = p; cur = { ...cur, ...q }; out.push([t, { ...cur }, _e || null]); }
  return t => {
    if (t <= out[0][0]) return { ...out[0][1] };
    for (let i = 0; i < out.length - 1; i++) {
      const [t0, a] = out[i], [t1, b, e] = out[i + 1];
      if (t < t1) {
        let u = (t - t0) / (t1 - t0); u = e === 'lin' ? u : u * u * (3 - 2 * u);
        const r = {};
        for (const k in b) {
          const va = a[k], vb = b[k];
          if (typeof va === 'number' && typeof vb === 'number') r[k] = va + (vb - va) * u;
          else if (Array.isArray(va) && Array.isArray(vb)) r[k] = va.map((v, j) => v + (vb[j] - v) * u);
          else r[k] = va === undefined ? vb : va;
        }
        return r;
      }
    }
    return { ...out[out.length - 1][1] };
  };
}

function girlQ(t) {
  const g = gTrack(t), h = 1 / 48, g0 = gTrack(t - h), g1 = gTrack(t + h);
  let vx = (g1.x - g0.x) / (2 * h); if (Math.abs(vx) > 900) vx = 0;
  const run = Math.abs(vx) > 220, moving = Math.abs(vx) > 14 && (g.mode === 'stand');
  const stride = run ? 62 : 34;
  const Q = { x: g.x, y: g.x > 1290 ? Gy(g.x) : 0, K: KG, face: g.face, mode: moving ? 'gait' : g.mode, rise: g.rise, crouch: g.crouch,
    speed: Math.abs(vx) / KG / stride, stride, duty: run ? 0.36 : 0.62, lift: run ? 10 : 6, lean: g.lean + (run ? 0.16 : moving ? 0.04 : 0), flare: run ? 0.9 : 0.5,
    nod: g.nod + g.laugh * 0.07 * Math.sin(t * 26), bob: g.laugh * 1.2 * Math.abs(Math.sin(t * 13)), mouth: g.laugh * (0.6 + 0.4 * Math.abs(Math.sin(t * 13))), smile: g.smile, browUp: g.browUp * 0.8,
    handN: g.hN, handF: g.hF, scarf: 1, wind: t > 31 && t < 85 ? 0.5 + 0.5 * Math.sin(t * 0.7) : 0,
    blink: g.eyes < 0.5 ? 1 : ((t * 0.31 + 0.2) % 1 < 0.035 ? 1 : 0) };
  if (g.shiver > 0) Q.x += Math.sin(t * 47) * 0.8 * g.shiver;
  Q.onStep = () => ev('step', { who: 'girl', surf: g.x < 1300 ? 'wood' : 'grass', run: run ? 1 : 0 });
  return { Q, g, vx };
}
