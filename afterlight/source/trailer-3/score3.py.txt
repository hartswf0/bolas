def load(fid, t0=0, dur=None, sr=SR):
    import glob
    p = glob.glob(A + fid + '*.mp4')[0]
    cmd = ['ffmpeg', '-v', 'error', '-ss', str(t0), '-i', p] + (['-t', str(dur)] if dur else []) + ['-ac', '1', '-ar', str(sr), '-f', 'f32le', '-']
    x = np.frombuffer(subprocess.run(cmd, capture_output=True).stdout, np.float32).astype(np.float64)
    return x / (np.abs(x).max() + 1e-9)
L = np.zeros(N); R = np.zeros(N)           # stereo bus (dry)
RV = np.zeros(N)                           # reverb send (mono)
def put(sig, t, g=1.0, pan=0.0, rev=0.0):
    i = int(t * SR); n = min(len(sig), N - i)
    if n <= 0 or i < 0: return
    s = sig[:n] * g; L[i:i + n] += s * np.sqrt(0.5 * (1 - pan)); R[i:i + n] += s * np.sqrt(0.5 * (1 + pan)); RV[i:i + n] += s * rev
def env(n, a, r):
    e = np.ones(n); A_ = max(1, int(a * SR)); R_ = max(1, int(r * SR)); e[:A_] = np.linspace(0, 1, A_); e[-R_:] *= np.linspace(1, 0, R_); return e
def bp(x, lo, hi, o=2): b, a = signal.butter(o, [lo / (SR / 2), hi / (SR / 2)], 'band'); return signal.lfilter(b, a, x)
def lp(x, f, o=2): b, a = signal.butter(o, min(f, SR / 2.2) / (SR / 2), 'low'); return signal.lfilter(b, a, x)
def hp(x, f, o=2): b, a = signal.butter(o, f / (SR / 2), 'high'); return signal.lfilter(b, a, x)
def mtof(m): return 440 * 2 ** ((m - 69) / 12)
def resamp(x, ratio):  # ratio > 1 raises pitch (and shortens)
    n = int(len(x) / ratio); return np.interp(np.arange(n) * ratio, np.arange(len(x)), x)
def wow(x, depth=0.0025, rate=0.6):
    n = len(x); tt = np.arange(n) / SR; d = depth * SR * (np.sin(2 * np.pi * rate * tt) + 0.3 * np.sin(2 * np.pi * 2.3 * rate * tt))
    return np.interp(np.clip(np.arange(n) - d - depth * SR * 1.5, 0, n - 1), np.arange(n), x)
def tape(x, drive=1.6): return np.tanh(x * drive) / np.tanh(drive)

BPM = 92; BEAT = 60 / BPM; BAR = 4 * BEAT; DROP = 19.6
def beat(k): return DROP + k * BEAT

def frames_rms(x, w=1024):
    n = len(x) // w; return np.sqrt((x[:n * w].reshape(n, w) ** 2).mean(1))
def speech_bits(x, minlen=0.5, maxlen=1.8):
    r = frames_rms(bp(x, 300, 3400)); thr = 0.22 * np.percentile(r, 99); on = r > thr; out = []; i = 0; w = 1024 / SR
    while i < len(on):
        if on[i]:
            j = i
            while j < len(on) and (on[j] or any(on[j + 1:j + 7])): j += 1
            a, b = i * w, j * w
            if b - a >= minlen: out.append((a, min(b, a + maxlen)))
            i = j
        else: i += 1
    return out
def cut(x, a, b, fade=0.03):
    s = x[int(a * SR):int(b * SR)].copy(); return s * env(len(s), fade, fade * 2)
def radio(x): return tape(bp(x, 350, 3200, 3) * 1.8, 1.8)
def tonal_window(x, win=2.5, hop=0.25, upto=None):
    best, bt = 1e9, 0; upto = upto or len(x) / SR - win
    for t0 in np.arange(0, upto, hop):
        s = x[int(t0 * SR):int((t0 + win) * SR)]
        if len(s) < win * SR * 0.9: break
        f, P = signal.welch(s, SR, nperseg=4096); P = P[(f > 80) & (f < 4000)] + 1e-14
        fl = np.exp(np.log(P).mean()) / P.mean()
        if fl < best and np.sqrt((s ** 2).mean()) > 0.05: best, bt = fl, t0
    return bt
def pitch_class(x):
    f, P = signal.welch(x, SR, nperseg=16384); ch = np.zeros(12)
    for fi, p in zip(f, P):
        if 60 < fi < 2000: ch[int(round(12 * np.log2(fi / 440) + 69)) % 12] += p
    return int(ch.argmax())
def freeze(x, t0, dur, out_len, grain=0.18, dens=40, jitter=0.6):
    """granular freeze: hold one moment of a recording as a sustained chord"""
    out = np.zeros(int(out_len * SR) + int(grain * SR) + 10); g = int(grain * SR); w = np.hanning(g)
    src = x[int(t0 * SR):int((t0 + dur) * SR)]
    for k in range(int(out_len * dens)):
        at = int(k / dens * SR); o = int(rng.random() * jitter * (len(src) - g))
        if o + g < len(src): out[at:at + g] += src[o:o + g] * w
    return out[:int(out_len * SR)] / (np.abs(out).max() + 1e-9)

def rhodes(m, t, d, g=0.12, pan=0.0, rev=0.25):
    f = mtof(m) * 2 ** (rng.normal(0, 0.06) / 12); n = int(d * SR); tt = np.arange(n) / SR
    mod = np.sin(2 * np.pi * f * 1.0 * tt) * 1.3 * np.exp(-tt * 3)
    s = np.sin(2 * np.pi * f * tt + mod) + 0.25 * np.sin(2 * np.pi * 2 * f * tt) * np.exp(-tt * 2) + 0.12 * np.sin(2 * np.pi * 7.03 * f * tt) * np.exp(-tt * 9)
    s *= np.exp(-tt * 1.1) * (1 + 0.18 * np.sin(2 * np.pi * 4.6 * tt)) * np.minimum(1, tt / 0.004)
    put(lp(s, 3200) * env(n, 0.004, 0.3), t, g, pan, rev)
def strings(ms, t, d, g=0.05, bright=1800, pan=0.0):
    n = int(d * SR); tt = np.arange(n) / SR; s = np.zeros(n)
    for m in ms:
        for det in (-0.13, 0.0, 0.12):
            f = mtof(m) * 2 ** (det / 12); s += signal.sawtooth(2 * np.pi * f * tt * (1 + 0.003 * np.sin(2 * np.pi * 5.5 * tt)) + rng.random() * 6.28)
    put(lp(s, bright) * env(n, min(2.5, d * 0.4), min(2.5, d * 0.4)) / len(ms) / 3, t, g, pan, 0.5)
def bass(m, t, d, g=0.28):
    f = mtof(m); n = int(d * SR); tt = np.arange(n) / SR
    s = np.sin(2 * np.pi * f * tt) + 0.3 * lp(signal.sawtooth(2 * np.pi * f * tt), 600)
    put(tape(s * env(n, 0.01, 0.12), 2.0), t, g, 0, 0.0)
def kick(t, g=0.55):
    n = int(0.45 * SR); tt = np.arange(n) / SR; f = 50 + 90 * np.exp(-tt * 30)
    put(np.sin(2 * np.pi * np.cumsum(f) / SR) * np.exp(-tt * 7) + lp(rng.standard_normal(n), 900) * np.exp(-tt * 60) * 0.3, t, g)
def snare(t, g=0.28):
    n = int(0.35 * SR); tt = np.arange(n) / SR
    s = bp(rng.standard_normal(n), 900, 6000) * np.exp(-tt * 14) + np.sin(2 * np.pi * 185 * tt) * np.exp(-tt * 22) * 0.6
    put(s, t, g, 0.05, 0.35)
def heart(t, g=0.35):

        n = int(0.3 * SR); tt = np.arange(n) / SR; put(np.sin(2 * np.pi * 46 * tt) * np.exp(-tt * 14), t + dt, g * gg)

# ---------------- SSTV: each scanline that deposits a picture is heard as the picture ----------------
def sstv(rows, t, d, g=0.06, pan=0.6):
    """rows: (n_lines, n_px) luminance 0..1. tone 1500 Hz black .. 2300 Hz white, 1200 Hz sync between lines"""
    nl = rows.shape[0]; per = d / nl; out = []
    for r_ in rows:
        sync = np.full(int(0.004 * SR), 1200.0)
        px = np.repeat(1500 + 800 * r_, max(1, int((per - 0.004) * SR / len(r_))))
        out += [sync, px]
    f = np.concatenate(out); s = np.sin(2 * np.pi * np.cumsum(f) / SR)
    put(bp(s, 900, 3000) * env(len(s), 0.01, 0.05), t, g, pan, 0.15)
