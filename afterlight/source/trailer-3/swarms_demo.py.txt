_cache = {}
def plate(name):
    if name not in _cache: _cache[name] = np.asarray(Image.open(f'{ROOT}/plates/{name}.png').convert('RGB').resize((w, h), Image.BILINEAR), np.float32) / 255

def sprite(name, scale=0.5):
    k = (name, scale)
    if k not in _cache:
        im = Image.open(f'{ROOT}/sprites/{name}.png').convert('RGBA'); im = im.resize((int(im.width * scale), int(im.height * scale)), Image.BILINEAR)
        _cache[k] = np.asarray(im, np.float32) / 255
    return _cache[k]
def blank(c=(0.02, 0.02, 0.03)): return np.zeros((h, w, 3), np.float32) + np.array(c, np.float32)
def comp(dst, rgba, x, y):
    """alpha-composite rgba (hh,ww,4) with top-left at x,y"""
    hh, ww = rgba.shape[:2]; x0, y0 = int(x), int(y); x1, y1 = max(x0, 0), max(y0, 0); x2, y2 = min(x0 + ww, w), min(y0 + hh, h)
    if x2 <= x1 or y2 <= y1: return dst
    s = rgba[y1 - y0:y2 - y0, x1 - x0:x2 - x0]; a = s[..., 3:4]
    dst[y1:y2, x1:x2] = dst[y1:y2, x1:x2] * (1 - a) + s[..., :3] * a; return dst
def rotate_rgba(rgba, deg, center=None):
    im = Image.fromarray((rgba * 255).astype(np.uint8), 'RGBA')
    return np.asarray(im.rotate(deg, resample=Image.BILINEAR, center=center), np.float32) / 255
def warp(img, bx, by):  # backward map: output pixel samples img at (bx, by)
    return np.stack([map_coordinates(img[..., c], [by, bx], order=1, mode='constant', cval=0.02) for c in range(img.shape[2])], -1)
YY, XX = np.mgrid[0:h, 0:w].astype(np.float32)
def mls(p, q, v_x, v_y, kind='affine', alpha=1.0):
    """Schaefer et al. 2006 moving least squares. maps points v near handles p to follow q. returns mapped x, y"""
    p = np.asarray(p, np.float32); q = np.asarray(q, np.float32)
    V = np.stack([v_x.ravel(), v_y.ravel()], 1)
    d2 = ((V[:, None, :] - p[None]) ** 2).sum(-1) + 1e-3; wgt = 1 / d2 ** alpha
    sw = wgt.sum(1, keepdims=True); ps = (wgt @ p) / sw; qs = (wgt @ q) / sw
    ph = p[None] - ps[:, None]; qh = q[None] - qs[:, None]
    if kind == 'affine':
        A = np.einsum('ni,nij,nik->njk', wgt, ph, ph)
        B = np.einsum('ni,nij,nik->njk', wgt, ph, qh)
        M = np.linalg.solve(A + np.eye(2) * 1e-3, B)
        out = np.einsum('nj,njk->nk', V - ps, M) + qs
    else:  # similarity / rigid
        vp = V - ps
        ph_perp = np.stack([-ph[..., 1], ph[..., 0]], -1)
        mu = (wgt * (ph ** 2).sum(-1)).sum(1)[:, None]
        a = (wgt * (ph * qh).sum(-1)).sum(1); b = (wgt * (ph_perp * qh).sum(-1)).sum(1)
        if kind == 'rigid': nrm = np.sqrt(a * a + b * b)[:, None] + 1e-6
        else: nrm = mu
        c, s = a[:, None] / nrm, b[:, None] / nrm
        out = np.stack([vp[:, 0] * c[:, 0] - vp[:, 1] * s[:, 0], vp[:, 0] * s[:, 0] + vp[:, 1] * c[:, 0]], 1) + qs
    return out[:, 0].reshape(v_x.shape), out[:, 1].reshape(v_y.shape)
def thermal(t):
    t = np.clip(t, 0, 1)[..., None]
    stops = np.array([[0.03, 0.01, 0.08], [0.32, 0.03, 0.22], [0.70, 0.07, 0.07], [0.97, 0.42, 0.06], [1.0, 0.86, 0.52]], np.float32)
    x = t * 4; i = np.clip(np.floor(x).astype(int), 0, 3); f = x - i
    return stops[i[..., 0]] * (1 - f) + stops[i[..., 0] + 1] * f
def lum(img): return img @ np.array([0.3, 0.59, 0.11], np.float32)
def grade(img, heat=0.35):
    return img * (1 - heat) + thermal(lum(img) * 0.95) * heat
def overlay(img):
    pil = Image.fromarray((np.clip(img, 0, 1) * 255).astype(np.uint8)); return pil, ImageDraw.Draw(pil, 'RGBA')
def back(pil): return np.asarray(pil, np.float32) / 255
def grid_lines(d, fx, fy, step=24, col=PHOS, a=110):
    for gy in range(0, h + 1, step):
        xs = np.arange(0, w + 1, 8, dtype=np.float32); pts = list(zip(*fx(xs, np.full_like(xs, gy)))); d.line(pts, fill=col + (a,), width=1)
    for gx in range(0, w + 1, step):
        ys = np.arange(0, h + 1, 8, dtype=np.float32); pts = list(zip(*fx(np.full_like(ys, gx), ys))); d.line(pts, fill=col + (a,), width=1)
def ease(u): u = min(1, max(0, u)); return u * u * (3 - 2 * u)

    return back(pil)

def d36(u, t, e):  # bounded biharmonic weights: four handles, each owning a smooth, bounded share of the coins
    img = plate('crystal'); m = gaussian_filter((lum(img) > 0.35).astype(np.float32), 1.5)
    H_ = np.array([[150, 110], [280, 220], [430, 120], [520, 260]], np.float32); cols = np.array([[1, .3, .2], [1, .75, .2], [.4, 1, .6], [.45, .65, 1]], np.float32)
    D = np.stack([1 / (1e-3 + ((XX - x) ** 2 + (YY - y) ** 2) / 6000) ** 2 for x, y in H_], -1); Wt = D / D.sum(-1, keepdims=True)
    off = np.array([[math.sin(u * 6 + k * 1.7) * 26, math.cos(u * 5 + k) * 18] for k in range(4)], np.float32)
    bx = XX - (Wt * off[:, 0]).sum(-1); by = YY - (Wt * off[:, 1]).sum(-1)
    base = warp(img, bx, by); mm = warp(m[..., None], bx, by); col = (Wt[..., None] * cols).sum(-2)
    out = base * 0.35 + col * mm * (0.35 + 0.65 * lum(base)[..., None]) + base * 0.3 * (1 - mm); pil, d = overlay(out)
    for (x, y), o, c in zip(H_, off, cols): d.ellipse([x + o[0] - 8, y + o[1] - 8, x + o[0] + 8, y + o[1] + 8], outline=tuple(int(v * 255) for v in c) + (255,), width=3)
    return back(pil)
def d37(u, t, e):  # weighted skinning: two bones bend the boy with the reel; the seam blends
    img = plate('reel'); a = math.sin(u * math.pi) * 0.5; pivot = (320, 200)
    wgt = 1 / (1 + np.exp(-(XX - 320) / 28))       # skin weight: 0 on the left bone, 1 on the right
    ca, sa = math.cos(-a), math.sin(-a); rx = (XX - pivot[0]) * ca - (YY - pivot[1]) * sa + pivot[0]; ry = (XX - pivot[0]) * sa + (YY - pivot[1]) * ca + pivot[1]
    bx = XX * (1 - wgt) + rx * wgt; by = YY * (1 - wgt) + ry * wgt
    out = grade(warp(img, bx, by), 0.35); band = np.exp(-((XX - 320) / 30) ** 2)[..., None]
    out = out * (1 - band * 0.35) + thermal(np.full((h, w), 0.8)) * band * 0.35; pil, d = overlay(out)
    end = (pivot[0] + 200 * math.cos(a), pivot[1] - 200 * math.sin(a)); d.line([(120, 200), pivot], fill=PHOS + (255,), width=4); d.line([pivot, end], fill=AMB + (255,), width=4)
