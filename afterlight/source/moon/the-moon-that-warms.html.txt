function runTests(){
  const base=filmTrace, out=[];
  const maxAbs=(a,b)=>a.reduce((m,v,i)=>Math.max(m,Math.abs(v-b[i])),0);
  const z=simulate(FILM.map(e=>e.k==='heater'?{...e,v:0}:e));const zmax=Math.max(...z.dT);
  out.push([zmax<1e-3,`Heater at zero all film: maximum skin rise ${zmax.toExponential(1)} °C (reflected plus emitted moonlight only). Bar: below 0.001 °C.`]);
  const sw=simulate([...FILM,{t:3,k:'image',v:'sun',by:'scene'}]);const d1=maxAbs(sw.dT,base.dT);
  out.push([d1===0,`Moon image swapped for Sun at 3 s: skin trace difference ${d1} °C. Bar: exactly 0.`]);
  const dim=simulate(FILM.map(e=>e.k==='bright'?{...e,v:0.1}:e));const d2=maxAbs(dim.dT,base.dT);
  out.push([d2<1e-4,`Brightness 1.0 → 0.1: skin trace difference ${d2.toExponential(1)} °C. Bar: below 0.0001 °C.`]);
  const part=simulate(FILM,7.3);const j=Math.round(7.3*SHZ);const d3=Math.abs(part.dT[part.dT.length-1]-base.dT[j]);
  out.push([d3===0,`Scrub determinism: state at 7.30 s computed alone vs. from the full run differs by ${d3}. Bar: exactly 0.`]);
  const after=base.peff.slice(Math.ceil(9.6*SHZ)).every(v=>v===0), cool=base.dT[Math.round(11.9*SHZ)]<base.dT[Math.round(9.6*SHZ)];
  out.push([after&&cool,`Opt-out at 9.6 s: IR at skin is 0 afterward (${after}); skin at 11.9 s is cooler than at 9.6 s (${cool}).`]);
  const keep=base.vis[Math.round(11*SHZ)]===1;
  out.push([keep,`Image stays lit after opt-out: brightness at 11 s = ${base.vis[Math.round(11*SHZ)]}.`]);
  const a6=base.aM[Math.round(5.9*SHZ)];out.push([a6===0,`Association before Act III: ${a6} at 5.9 s. Bar: exactly 0.`]);
  const sk=base.dT[Math.round(30*SHZ)], rc=base.aM[Math.round(30*SHZ)];out.push([sk<0.01&&rc>0,`Next-night re-exposure at 30 s: skin ΔT ${sk.toFixed(4)} °C, stored association ${rc.toFixed(2)}. The image returns without heat.`]);
  $('tests').innerHTML=out.map(([ok,s])=>`<div class="test"><span class="pill ${ok?'pass':'fail'}">${ok?'PASS':'FAIL'}</span><span>${s}</span></div>`).join('');
}
