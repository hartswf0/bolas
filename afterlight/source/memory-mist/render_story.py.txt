def mk(kind,a):
    n=min(len(MH[kind])-1,max(0,a-MI0)); m=MH[kind][n].astype(np.float32)/255
    if kind=='human' and 156.0<=T0+a/SRC<=168.3: m=m.copy(); m[:,:int(0.42*sw)]=0      # burial: only the crouched child, not blobs in the black

def noise(kind,a,S_):
    k=(kind,a,S_)
    if k not in NC:
        if len(NC)>700: NC.pop(next(iter(NC)))
        ny,nx=GR[S_][0].shape; r=np.random.default_rng(a*4+S_*2+(kind=='j'))
        NC[k]=r.random((ny,nx)).astype(np.float32) if kind=='m' else r.normal(0,1,(ny,nx,2)).astype(np.float32)
    return NC[k]
def entry(T):
    for i,e in enumerate(EDL):
        if e['t0']<=T<e['t1']: return i,e
    return (len(EDL)-1,EDL[-1]) if T>=EDL[-1]['t1'] else (0,EDL[0])
def src_time(T):
    _,e=entry(T); u=min(1,max(0,(T-e['t0'])/(e['t1']-e['t0']))); return e['src0']+u*(e['src1']-e['src0'])
def pulse(T):
    d=T-ON[:,0]; m=(d>=0)&(d<2.0)
    return float((ON[m,1]*(1-np.exp(-d[m]/0.04))*np.exp(-d[m]/0.45)).sum()) if m.any() else 0.0
def bass(T): return float(np.interp(T,BT[:,0],BT[:,1]))
def smooth(a,b,u): u=u*u*(3-2*u); return a+(b-a)*u

def stage_of(ut): return 0 if ut<0.45 else (1 if ut<1.35 else (2 if ut<2.3 else 3))
def bgweight(g,ut):
    end=BGEND.get(g.get('role','place'),0.4); s=np.clip((ut-1.2)/1.1,0,1); s=s*s*(3-2*s); return 0.62+(end-0.62)*s

def blendmask(kind,fi): a=int(np.floor(fi)); w=fi-a; return (1-w)*mk(kind,a)+w*mk(kind,a+1)

def core(T,probe=False):
    ei,e=entry(T); g=SEG[e['seg']]; u=(T-e['t0'])/(e['t1']-e['t0']); ut=T-e['t0']; hero=g.get('hero',False)
    pu=0.0 if probe else min(1.3,pulse(T)); bs=bass(T); bgw=bgweight(g,ut)
    mist=g['mist']; N=30 if hero else 22; dz=0.040 if hero else 0.050
    yaw=np.radians(smooth(-3,4,u) if not hero else smooth(-11,9,u)); pitch=np.radians(smooth(1.5,-1,u))
    zoom=g.get('zoom',1.0); zcx,zcy=g.get('zc',(0.5,0.5)); cxU=(zcx*2-1)*(sw/sh); cyU=-(zcy*2-1)
    D=2.4-0.22*u; foc=1.18*H*zoom
    scan=np.clip(ut/0.45,0,1)
    acc=[np.zeros((H*W,3),np.float32) for _ in range(3)]
    cy,sy=np.cos(yaw),np.sin(yaw); cp,sp=np.cos(pitch),np.sin(pitch)
    for j in range(N):
        age=float(j); tj=src_time(max(0.0,T-j/SRC)); fi=(tj-T0)*SRC; a=int(np.floor(fi)); wg=fi-a
        S_=1 if j<3 else 2; PX,PY=GR[S_]; PX=PX-cxU; PY=PY-cyU; ny,nx=PX.shape
        A,B=load(a),load(a+1); cur=((1-wg)*A+wg*B)
        hum=np.clip(blendmask('human',fi)+blendmask('hand',fi)+blendmask('toy',fi)+blendmask('face',fi),0,1)
        lum_full=cur@np.array([.3,.59,.11],np.float32)
        fire=np.maximum(np.clip((cur[...,0]-cur[...,2])*3.2,0,1)*np.clip((lum_full-0.30)*3,0,1),np.clip((lum_full-0.55)*4,0,1))
        form=np.clip(hum+g.get('firew',0.55)*fire,0,1)[::S_,::S_]; toy=blendmask('toy',fi)[::S_,::S_]; cur=cur[::S_,::S_]
        mv=cv2.GaussianBlur(np.abs(A-load(a-1)).mean(2),(0,0),2.0); mot=np.clip((mv-0.02)*8,0,1)[::S_,::S_]
        nm=(1-wg)*noise('m',a,S_)+wg*noise('m',a+1,S_); jr=(1-wg)*noise('j',a,S_)+wg*noise('j',a+1,S_)
        lum=cur@np.array([.3,.59,.11],np.float32)
        w=(0.05+lum**0.6)*(1+1.2*mot)*(np.exp(-age/1.3)+0.10*(1-age/(N+1))**2)*(1.0 if S_==1 else 4.0)*(1+0.15*pu)
        w*=(0.80+0.20*nm)*(bgw+0.8*form)*(1+2.5*toy)
        # construction: the scan line reveals the frame top to bottom
        yn=(1-(PY+cyU))/2; w*=1/(1+np.exp(-(scan*1.25-yn)*14)) if scan<1 else 1.0
        r=np.hypot((PX+cxU)/(sw/sh),(PY+cyU)); w*=np.clip(1.15-r*(0.35+mist*(0.5+age/N)*(1-0.7*form)),0.02,1)
        z=-age*dz+0.16*form-0.10*(1-form); zz=z-0.08*mot*age/N
        X=PX*cy+(zz+0.5)*sy; Zr=-PX*sy+(zz+0.5)*cy-0.5; Y=PY*cp-Zr*sp; Z2=PY*sp+Zr*cp
        amp=(0.0022+0.0045*(age**0.8)*(0.4+mist*2))*(0.35+1.7*(1-form)); X=X+jr[...,0]*amp; Y=Y+jr[...,1]*amp
        dep=D-Z2; ok=dep>0.3; xs=(X/dep*foc+W/2).astype(np.int32); ys=(-Y/dep*foc+H/2).astype(np.int32)
        ok&=(xs>=0)&(xs<W)&(ys>=0)&(ys<H); k=min(2,int(age/N*3.2)); idx=(ys*W+xs)[ok]
        col=cur.copy(); gm=col.mean(2,keepdims=True); col=np.clip(gm+(col-gm)*(0.30+1.1*form[...,None]),0,1)
        far=min(1,age/N); col=col*(1-far*mist*0.6)+np.array([1,.93,.82],np.float32)*far*mist*0.35
        for c in range(3): acc[k][:,c]+=np.bincount(idx,weights=(w*col[...,c])[ok],minlength=H*W).astype(np.float32)
    sig=[1.5,2.4,4.2]; gain=[3.2,3.6,3.2]; out=np.zeros((H,W,3),np.float32)
    for k in range(3): out+=cv2.GaussianBlur(acc[k].reshape(H,W,3),(0,0),sig[k])*gain[k]
    if probe: return float(out.mean())
    gx_=EXP[ei]; out=1-np.exp(-out*gx_)
    hz=g.get('haze',1.0)*(0.5+0.5*bs)
    bloom=cv2.GaussianBlur(out,(0,0),18); haze=cv2.GaussianBlur(out.mean(2),(0,0),40)[...,None]
    out=out+bloom*(0.30+mist*0.30)*hz+haze*np.array([1,.94,.84],np.float32)*(0.20+0.30*mist)*hz
    # a clean cut-out of the figures sits under the mist so people read at a glance
    ts=src_time(T); fi=(ts-T0)*SRC; a=int(np.floor(fi)); wg=fi-a; src=(1-wg)*load6(a)+wg*load6(a+1)
    fm=np.clip(blendmask('human',fi)+blendmask('hand',fi)+blendmask('toy',fi),0,1)
    k_=foc/D; Mx=np.float32([[k_*2*(sw/sh)/sw,0,W/2-k_*((sw/sh)+cxU)],[0,k_*2/sh,H/2-k_*(1-cyU)]]); Mx6=Mx.copy(); Mx6[0,0]*=0.5; Mx6[1,1]*=0.5   # source px -> screen px (same projection as the points at z=0)
    _cl=cv2.createCLAHE(clipLimit=4.0,tileGridSize=(6,3)); lab=cv2.cvtColor((src*255).astype(np.uint8),cv2.COLOR_RGB2LAB); lab[...,0]=_cl.apply(lab[...,0]); srcc=cv2.cvtColor(lab,cv2.COLOR_LAB2RGB).astype(np.float32)/255
    fm_s=cv2.warpAffine(fm,Mx,(W,H)); cut=cv2.warpAffine(srcc,Mx6,(W,H)); cut=cv2.GaussianBlur(cut,(0,0),0.8)
    cg=g.get('cutgain',1.0); cl=np.minimum(np.clip(cut,0,1)**(0.85-0.12*(cg-1))*1.15,0.8); cutl=cl*np.clip(cv2.GaussianBlur(fm_s,(0,0),1.2),0,1)[...,None]
    reveal=1/(1+np.exp(-(np.clip(ut/0.45,0,1)*1.25-np.linspace(0,1,H)[:,None])*14)) if ut<0.45 else 1.0
    cutw=0.32*cg*(0.5+0.5*np.clip((ut-0.3)/0.8,0,1))*(0.9+0.2*min(1,pu))
    out=out+cutl*cutw*(reveal[...,None] if np.ndim(reveal)>0 else reveal)
    return np.clip(out,0,1)**0.9,dict(e=e,g=g,u=u,ut=ut,pu=pu,bs=bs,fm_s=fm_s,face_s=cv2.warpAffine(blendmask('face',fi),Mx,(W,H)),toy_s=cv2.warpAffine(blendmask('toy',fi),Mx,(W,H)),hand_s=cv2.warpAffine(blendmask('hand',fi),Mx,(W,H)))

def rrect(img,x0,y0,x1,y1,r,col,th):
    L=cv2.LINE_AA
    for p,q in (((x0+r,y0),(x1-r,y0)),((x0+r,y1),(x1-r,y1)),((x0,y0+r),(x0,y1-r)),((x1,y0+r),(x1,y1-r))): cv2.line(img,p,q,col,th,L)
    for c,a in (((x0+r,y0+r),180),((x1-r,y0+r),270),((x1-r,y1-r),0),((x0+r,y1-r),90)): cv2.ellipse(img,c,(r,r),0,a,a+90,col,th,L)

def contours_of(mask_s):
    m=cv2.resize((mask_s>0.5).astype(np.uint8),(OW,OH),interpolation=cv2.INTER_NEAREST)
    m=cv2.morphologyEx(m,cv2.MORPH_CLOSE,np.ones((9,9),np.uint8)); m=cv2.morphologyEx(m,cv2.MORPH_OPEN,np.ones((7,7),np.uint8))
    cs,_=cv2.findContours(m,cv2.RETR_EXTERNAL,cv2.CHAIN_APPROX_SIMPLE)
    return [c for c in cs if cv2.contourArea(c)>OW*OH*0.004]
def draw_progress(img,c,p,col,th):
    pts=cv2.approxPolyDP(c,2.2,True)[:,0,:]; n=max(2,int(len(pts)*p))
    if n>=2: cv2.polylines(img,[pts[:n].reshape(-1,1,2)],p>=0.999,col,th,cv2.LINE_AA)

    dyn=np.zeros((OH,OW,3),np.uint8); labels=[]

def frame_mist(T):
    out,c=core(T); ST=statics(); pu=c['pu']
    base=cv2.resize(out,(OW,OH),interpolation=cv2.INTER_CUBIC); lum=base.mean(2,keepdims=True)
    t=np.clip(lum*2.0,0,1); tone=(1-t)*np.array([0.45,1.0,0.95],np.float32)+t*np.array([1.0,0.80,0.50],np.float32)
    warm=np.clip((base[...,0:1]-base[...,2:3])*3.2,0,1); tone=(1-warm)*tone+warm*np.array([1.0,0.70,0.28],np.float32)
    base=0.40*base+0.60*lum*tone*(1.05+0.35*warm)+np.array([0.0,0.030,0.036],np.float32)*(1-lum)
    base=base*ST['vig'][...,None]
    base[...,0]=np.roll(base[...,0],1,1); base[...,2]=np.roll(base[...,2],-1,1)
    base=base+np.random.default_rng(int(T*24)).normal(0,0.008,(OH,OW,1)).astype(np.float32)
    ui=console(T,c); layer=ST['ui'].astype(np.float32)/255*(0.7+0.4*min(1,pu))+ui; layer=layer+cv2.GaussianBlur(layer,(0,0),4)*0.8
    fade=min(1,T/2.5)*min(1,(DUR-T)/2.2); uf=min(1,T/1.2)*min(1,(DUR-T)/1.2)
    fin=np.clip(base*(1-0.62*ST['plate'][...,None])*fade*ST['inside'][...,None]+layer*uf+ST['dirt'][...,None]*np.array([0.35,0.8,0.78],np.float32)*uf,0,1)
    return (fin*255).astype(np.uint8)
