"""The hand-and-horse passage, drawn two ways over the unaltered plate.
S: Reiniger silhouette (backlit cut paper, the horse a black hinged silhouette).
C: Gilliam/Ernst substitution card (engraved plate, the horse an engraved, pinned wooden card).
Observed: the plate (featurette frames), the fist and toy masks. Generated: everything drawn: bands, hatching,
paper, shadows, and the horse itself (drawn by hand in code from polygons; it is not the film's toy)."""
import os,glob,math
import numpy as np,cv2
HERE=os.path.dirname(os.path.abspath(__file__))
OW,OH=1280,720; T0=109.0; SRC=12
WX,WW,WH=0,469,264; KS=OW/WW                      # the window of the 640x264 plate that fills 16:9
F6=sorted(glob.glob(HERE+'/f640/*.jpg')); _c={}
def plate_frame(i):
    i=max(0,min(len(F6)-1,i))
    if i not in _c:
        if len(_c)>40: _c.pop(next(iter(_c)))
        _c[i]=cv2.cvtColor(cv2.imread(F6[i]),cv2.COLOR_BGR2RGB).astype(np.float32)/255
    return _c[i]
M=np.load(HERE+'/m07.npz'); MI0=int(M['i0']); NM=M['fist'].shape[0]
def clean_toy(t):
    n,lab,st,_=cv2.connectedComponentsWithStats((t>0).astype(np.uint8)); o=np.zeros_like(t)
    for k in range(1,n):
        x,y,w,h,a=st[k]
        if a>40 and x>16 and h<2.2*w: o[lab==k]=255          # drop the coat strip on the left edge
    return o
TOY=[clean_toy(M['toy'][k]) for k in range(NM)]; FIST=[M['fist'][k] for k in range(NM)]; HAND=[M['hand'][k] for k in range(NM)]
def centroid(m):
    ys,xs=np.nonzero(m); return (xs.mean(),ys.mean()) if len(xs)>50 else None
# fit of a standing horse to the toy parts that stick out of the fist (head notch, hooves), in 640x264 source px
def _fit():
    off=[];hs=[]
    for k in range(6,18):
        c=centroid(FIST[k]); ys,xs=np.nonzero(TOY[k])
        if c is None or len(xs)<60: continue
        off.append(((xs.min()+xs.max())/2-c[0],(ys.min()+ys.max())/2-c[1])); hs.append(ys.max()-ys.min())
    return (np.mean(off,0) if off else np.array([40.,50.]),np.mean(hs) if hs else 110.)
OFF,TH=_fit(); S0=max(70.,TH/1.05)               # px per horse unit in source px

# ---------- the horse, drawn by hand from polygons (facing right; unit = body length) ----------
def horse_alpha(cx,cy,scale,rot=0.0,nod=0.0,legs=(0.0,0.0,0.0,0.0),ss=2,pad=0):
    """Return float alpha (OH,OW) of a wooden toy horse centred at (cx,cy) output px, `scale` px per unit, rot radians,
    head nod and 4 leg swing angles in radians (front-far, front-near, rear-far, rear-near)."""
    P=[]
    body=[(-.52,0),(-.46,-.15),(-.30,-.22),(.30,-.22),(.46,-.16),(.52,-.02),(.47,.14),(.30,.19),(-.30,.19),(-.47,.14)]
    P.append(body)
    tail=[(-.50,-.14),(-.66,-.12),(-.76,.04),(-.80,.22),(-.70,.14),(-.62,.02),(-.52,-.02)]; P.append(tail)
    # neck and head hinge at the withers; nod rotates head about it
    def rot_pt(p,o,a):
        c,s=math.cos(a),math.sin(a); x,y=p[0]-o[0],p[1]-o[1]; return (o[0]+c*x-s*y,o[1]+s*x+c*y)
    neck=[(.26,-.20),(.46,-.20),(.60,-.52),(.50,-.60),(.34,-.30)]
    head=[(.50,-.60),(.52,-.74),(.58,-.70),(.62,-.76),(.66,-.66),(.86,-.52),(.88,-.44),(.76,-.40),(.62,-.50)]
    mane=[(.30,-.28),(.34,-.34),(.38,-.30),(.40,-.40),(.44,-.36),(.46,-.46),(.50,-.42),(.52,-.54),(.46,-.58),(.38,-.46)]
    o=(.44,-.34)
    P.append(neck); P.append([rot_pt(p,o,nod) for p in head]); P.append([rot_pt(p,o,nod) for p in mane])
    def leg(hx,hy,a,w=.075):
        L=.52; tip=(hx+math.sin(a)*L,hy+math.cos(a)*L)
        n=(math.cos(a),-math.sin(a)); wb=w*.72
        quad=[(hx-n[0]*w,hy-n[1]*w),(hx+n[0]*w,hy+n[1]*w),(tip[0]+n[0]*wb,tip[1]+n[1]*wb),(tip[0]-n[0]*wb,tip[1]-n[1]*wb)]
        hoof=[(tip[0]-n[0]*wb,tip[1]-n[1]*wb),(tip[0]+n[0]*wb*1.5,tip[1]+n[1]*wb*1.5),(tip[0]+n[0]*wb*1.5+.01,tip[1]+n[1]*wb*1.5+.05),(tip[0]-n[0]*wb-.01,tip[1]-n[1]*wb+.05)]
        return [quad,hoof]
    for (hx,a) in ((.36,legs[0]),(.40,legs[1]),(-.36,legs[2]),(-.32,legs[3])): P+=leg(hx,.14,a)
    c,s=math.cos(rot),math.sin(rot)
    img=np.zeros((OH*ss,OW*ss),np.uint8)
    for poly in P:
        pts=np.array([[ (cx+scale*(c*x-s*y))*ss,(cy+scale*(s*x+c*y))*ss] for x,y in poly],np.int32)
        cv2.fillPoly(img,[pts],255,cv2.LINE_AA)
    return cv2.resize(img,(OW,OH),interpolation=cv2.INTER_AREA).astype(np.float32)/255

def ss(a,b,x): t=np.clip((x-a)/(b-a),0,1); return t*t*(3-2*t)
def up(m,blur=1.2):
    """640x264 mask -> window -> 1280x720 soft alpha."""
    w=cv2.resize(m[:,WX:WX+WW].astype(np.float32)/255,(OW,OH),interpolation=cv2.INTER_CUBIC)
    return np.clip(ss(.35,.65,cv2.GaussianBlur(w,(0,0),blur*2)),0,1)
def plate_at(ts):
    fi=(ts-T0)*SRC; a=int(math.floor(fi)); w=fi-a; return (1-w)*plate_frame(a)+w*plate_frame(a+1),a
def to_out(img): return cv2.resize(img[:,WX:WX+WW],(OW,OH),interpolation=cv2.INTER_CUBIC)

# ---------- pose of the drawn horse over each passage ----------
def pose(passage,u,fistc):
    """u seconds into the passage. fistc: fist centroid (source px). Returns (cx,cy,scale_px,rot,nod,legs)."""
    hx=(fistc[0]+OFF[0]-WX)*KS; hy=(fistc[1]+OFF[1])*KS; sc=S0*KS
    w=math.sin(u*2*math.pi*1.1)
    if passage=='reveal':
        a=ss(1.3,2.6,u)                                   # the toy wakes after the first beat, then hops out of the fist
        legs=(a*.35*w,-a*.35*w,-a*.30*w,a*.30*w); return hx+a*.55*sc,hy-a*(.30*sc+6*abs(w)),sc,0.0,a*.20*math.sin(u*2*math.pi*.9),legs
    # hold: held, then lifted out into a clear profile, then it stays
    lift=ss(1.0,3.0,u); tx,ty,ts_=860.0,400.0,sc*1.5
    cx=hx+(tx-hx)*lift; cy=hy+(ty-hy)*lift; scl=sc+(ts_-sc)*lift
    calm=.18+.12*lift
    legs=(calm*.5*w,-calm*.5*w,-calm*.45*w,calm*.45*w)
    return cx,cy,scl,0.045*math.sin(u*2*math.pi*.35)*lift,.10*math.sin(u*2*math.pi*.5),legs

# ---------- S: Reiniger silhouette ----------
PAL=np.array([(8,16,17),(86,42,24),(222,118,36),(255,214,122)],np.float32)/255
def style_S(ts,u,passage,T):
    P,a=plate_at(ts); k=int(np.clip(round((ts-147.0)*12),0,NM-1)); fistc=centroid(FIST[k]) or (190,70)
    toy=up(TOY[k]); hand=up(HAND[k])
    P=P.copy(); P=cv2.inpaint((P*255).astype(np.uint8),(cv2.dilate(TOY[k],np.ones((9,9),np.uint8))>0).astype(np.uint8)*255,5,cv2.INPAINT_TELEA).astype(np.float32)/255 if passage=='hold' else P
    L=to_out(P)@np.array([.3,.59,.11],np.float32); L=cv2.GaussianBlur(L,(0,0),2.2); L=np.clip((L-.02)/.85,0,1)**.85
    w1,w2,w3=ss(.10-.03,.10+.03,L),ss(.34-.03,.34+.03,L),ss(.62-.03,.62+.03,L)
    col=PAL[0]+(PAL[1]-PAL[0])*w1[...,None]; col=col+(PAL[2]-PAL[1])*w2[...,None]*1+(PAL[3]-PAL[2])*w3[...,None]
    col=PAL[0]+(PAL[1]-PAL[0])*w1[...,None]+(PAL[2]-PAL[1])*w2[...,None]+(PAL[3]-PAL[2])*w3[...,None]
    for lv,wk in ((.10,w1),(.34,w2),(.62,w3)):              # paper layers cast a little shadow on the layer below
        sh=cv2.GaussianBlur(np.roll(np.roll(wk,5,0),4,1),(0,0),3.0)*(1-wk); col=col*(1-.42*sh[...,None])
    cx,cy,sc,rot,nod,legs=pose(passage,u,fistc)
    hz=horse_alpha(cx,cy,sc,rot,nod,legs)
    rim=np.clip(cv2.GaussianBlur(np.clip(cv2.dilate(hand,np.ones((7,7),np.uint8))-hand,0,1),(0,0),2.0)*2.2,0,1)
    ink=np.array([6,4,4],np.float32)/255; hfill=np.array([66,30,14],np.float32)/255
    hrim=np.clip(cv2.GaussianBlur(np.clip(cv2.dilate(hz,np.ones((5,5),np.uint8))-hz,0,1),(0,0),1.4)*3.0,0,1)
    front=(passage=='hold') or (passage=='reveal' and u>=1.3)
    sil=np.maximum(hand,toy*(0.0 if passage=='hold' else 1.0))
    def put_horse(c):
        c=c+hrim[...,None]*np.array([1,.66,.22],np.float32)*.9*(1-hz[...,None]); return c*(1-hz[...,None])+hfill*hz[...,None]
    if not front: col=put_horse(col)
    col=col*(1-sil[...,None])+ink*sil[...,None]
    col=col+rim[...,None]*np.array([1,.62,.2],np.float32)*.55*L[...,None]*(1-sil[...,None])
    if front: col=put_horse(col)
    g=np.random.default_rng(int(T*24)).normal(0,.012,(OH,OW,1)).astype(np.float32)             # paper grain
    bl=cv2.GaussianBlur(col*np.clip(L[...,None]*1.4-.5,0,1),(0,0),14)*.35
    return np.clip(col+bl+g,0,1)

# ---------- C: Gilliam substitution card ----------
PAPER=np.array([236,226,204],np.float32)/255; INK=np.array([28,20,24],np.float32)/255
def hatch(L,spacing=5.0,angle=35.0,warp=1.8,cross=.55,seed=0):
    yy,xx=np.mgrid[0:OH,0:OW].astype(np.float32); Ls=cv2.GaussianBlur(L,(0,0),5.0); out=np.zeros((OH,OW),np.float32)
    for th,dk in ((angle,np.clip((1-L)*1.05,0,1)),(angle+62,np.clip(((1-L)-cross)/(1-cross),0,1)*.9)):
        t=math.radians(th); u=xx*math.cos(t)+yy*math.sin(t)+warp*spacing*(Ls-.5)*2; d=np.abs(((u/spacing)+.5)%1-.5)*spacing
        out=np.maximum(out,np.clip(dk*spacing*.5-d+.7,0,1))
    return out
def card(alpha,body_rgb,shadow=(7,9),edge=3):
    """paper card with a white edge and a soft cast shadow; returns (rgb, a, shadow_a)"""
    a=np.clip(alpha,0,1); ring=np.clip(cv2.dilate(a,np.ones((edge*2+1,edge*2+1),np.uint8)),0,1)
    sh=cv2.GaussianBlur(np.roll(np.roll(ring,shadow[0],0),shadow[1],1),(0,0),4)
    rgb=body_rgb*a[...,None]+PAPER*(ring-a)[...,None]
    return rgb,ring,sh
def style_C(ts,u,passage,T):
    Tq=math.floor(T*8)/8; uq=u-(T-Tq)                       # stop-motion on threes
    ts_=ts if abs(T-Tq)<1e-6 else ts-((T-Tq)/1.0)*0.0
    P,a=plate_at(ts); k=int(np.clip(round((ts-147.0)*12),0,NM-1)); fistc=centroid(FIST[k]) or (190,70)
    if passage=='hold': P=cv2.inpaint((P*255).astype(np.uint8),(cv2.dilate(TOY[k],np.ones((9,9),np.uint8))>0).astype(np.uint8)*255,5,cv2.INPAINT_TELEA).astype(np.float32)/255
    Po=to_out(P); L=np.clip((Po@np.array([.3,.59,.11],np.float32)-.02)/.85,0,1)**.8
    base_ink=hatch(L,5.2,35.0); wash=cv2.GaussianBlur(Po,(0,0),9); wash=wash/max(.35,float(wash.max()))
    plate=PAPER*(1-base_ink[...,None])+INK*base_ink[...,None]; plate=plate*(.62+.38*np.clip(wash*1.1,0,1))      # engraved plate, hand-coloured wash
    jit=np.random.default_rng(int(Tq*8)).integers(-1,2,2)                                                    # cards shiver on threes
    hand=up(np.maximum(HAND[k],0),blur=2.2); toy=up(TOY[k])
    Lh=L; ih=hatch(np.clip(Lh*1.25+.08,0,1),4.2,-28.0,cross=.45)
    hrgb=PAPER*(1-ih[...,None])+INK*ih[...,None]; hrgb=hrgb*(.55+.45*np.clip(wash*1.1,0,1))
    hrgb,hring,hsh=card(np.roll(np.roll(hand,jit[0],0),jit[1],1),np.roll(np.roll(hrgb,jit[0],0),jit[1],1))
    cx,cy,sc,rot,nod,legs=pose(passage,uq,fistc)
    hz=horse_alpha(cx,cy,sc,rot,nod,legs)
    # wooden horse card: grain, shading from the card's own form, engraved
    yy,xx=np.mgrid[0:OH,0:OW].astype(np.float32)
    grain=.55+.14*np.sin((xx*.011+np.sin(yy*.012)*2.0))+.05*np.sin(yy*.05+xx*.02)
    Lw=np.clip(grain*(0.55+0.45*np.clip(1-(yy-cy)/(sc*1.2)+.2,0,1)),0,1)
    iw=hatch(Lw,9.0,52.0,cross=.97); wood=np.array([204,150,92],np.float32)/255
    wrgb=wood*(1-.30*iw[...,None])*(.88+.2*grain[...,None])
    wrgb,wring,wsh=card(hz,wrgb,(6,8),3)
    out=plate*1.0
    if passage=='reveal':                                     # held: the horse card first, the fist card over it
        out=out*(1-.5*wsh[...,None]); out=out*(1-wring[...,None])+wrgb*wring[...,None]
        out=out*(1-.45*hsh[...,None]); out=out*(1-hring[...,None])+hrgb*hring[...,None]
    else:
        out=out*(1-hring[...,None])+hrgb*hring[...,None]
        out=out*(1-.5*wsh[...,None]); out=out*(1-wring[...,None])+wrgb*wring[...,None]
    # the pin through the horse card
    pin=np.zeros((OH,OW),np.float32); cv2.circle(pin,(int(cx+.05*sc),int(cy-.02*sc)),9,1,-1,cv2.LINE_AA); pin=cv2.GaussianBlur(pin,(0,0),.8)
    out=out*(1-pin[...,None])+np.array([40,32,34],np.float32)/255*pin[...,None]
    hl=np.zeros((OH,OW),np.float32); cv2.circle(hl,(int(cx+.05*sc)-3,int(cy-.02*sc)-3),3,1,-1,cv2.LINE_AA); out=out*(1-hl[...,None]*.7)+hl[...,None]*.7*PAPER
    g=np.random.default_rng(int(Tq*8)+7).normal(0,.008,(OH,OW,1)).astype(np.float32)
    return np.clip(out+g,0,1)


# ---------- whole-frame plates for the scenes and events (the faces stay in mist) ----------
def place_plate(P):
    """640x264 plate -> 1280x528 letterboxed into the 1280x720 canvas."""
    big=cv2.resize(P,(OW,528),interpolation=cv2.INTER_CUBIC); out=np.zeros((OH,OW,3),np.float32); out[96:624]=big; return out
def _levels(L,m):
    lo,hi=np.percentile(L[96:624],2),np.percentile(L[96:624],99.6); return np.clip((L-lo)/max(.08,hi-lo),0,1)**.85
def plate_S(ts,T):
    P,a=plate_at(ts); Po=place_plate(P); L=cv2.GaussianBlur(Po@np.array([.3,.59,.11],np.float32),(0,0),2.6); L=_levels(L,None)
    w1,w2,w3=ss(.14-.03,.14+.03,L),ss(.38-.03,.38+.03,L),ss(.66-.03,.66+.03,L)
    col=PAL[0]+(PAL[1]-PAL[0])*w1[...,None]+(PAL[2]-PAL[1])*w2[...,None]+(PAL[3]-PAL[2])*w3[...,None]
    for wk in (w1,w2,w3):
        sh=cv2.GaussianBlur(np.roll(np.roll(wk,5,0),4,1),(0,0),3.0)*(1-wk); col=col*(1-.42*sh[...,None])
    col[:96]=PAL[0]*.6; col[624:]=PAL[0]*.6
    g=np.random.default_rng(int(T*24)).normal(0,.012,(OH,OW,1)).astype(np.float32); bl=cv2.GaussianBlur(col*np.clip(L[...,None]*1.4-.5,0,1),(0,0),14)*.35
    return np.clip(col+bl+g,0,1)
def plate_C(ts,T):
    Tq=math.floor(T*8)/8; P,a=plate_at(ts); Po=place_plate(P); L=_levels(Po@np.array([.3,.59,.11],np.float32),None)
    ink=hatch(L,5.2,35.0); wash=cv2.GaussianBlur(Po,(0,0),9); wash=wash/max(.35,float(wash.max()))
    out=PAPER*(1-ink[...,None])+INK*ink[...,None]; out=out*(.62+.38*np.clip(wash*1.1+.1,0,1))
    out[:96]=INK*.7; out[624:]=INK*.7
    return np.clip(out+np.random.default_rng(int(Tq*8)+3).normal(0,.008,(OH,OW,1)).astype(np.float32),0,1)


# ---------- the burial: the drawn horse is laid in the ash and covered (drawn; the footage does not show the toy) ----------
def _pile():
    P=plate_frame(int(round((162.0-T0)*SRC)))
    L=P@np.array([.3,.59,.11],np.float32); sub=L[150:264,60:360]; thr=np.percentile(sub,97); ys,xs=np.nonzero(sub>=thr)
    return (xs.mean()+60)*2.0,96+(ys.mean()+150)*2.0
PX_,PY_=_pile()
def burial(img,variant,ub,T):
    """overlay on a finished plate frame: ub seconds into the burial beat."""
    ha=float(ss(5.6,6.4,ub)*(1-ss(11.0,14.6,ub)))
    cover=float(ss(10.2,15.2,ub))
    cx,cy=PX_-30,PY_-8
    if ha>0.01:
        hz=horse_alpha(cx,cy,118,3.0,0.0,(.10,-.10,.12,-.12))*ha
        if variant=='S':
            img=img+( -img+np.array([66,30,14],np.float32)/255)*hz[...,None]
            rim=np.clip(cv2.GaussianBlur(np.clip(cv2.dilate(hz,np.ones((5,5),np.uint8))-hz,0,1),(0,0),1.4)*3.0,0,1)*(1-hz); img=img+rim[...,None]*np.array([1,.66,.22],np.float32)*.8
        else:
            yy,xx=np.mgrid[0:OH,0:OW].astype(np.float32); grain=.55+.14*np.sin(xx*.011+np.sin(yy*.012)*2)
            wrgb=np.array([204,150,92],np.float32)/255*(.88+.2*grain[...,None]); wrgb,wring,wsh=card(hz,wrgb,(6,8),3)
            img=img*(1-.5*wsh[...,None]); img=img*(1-wring[...,None])+wrgb*wring[...,None]
    if cover>0.01:                                            # ash: a mound drawn over it, and flecks falling
        th=np.linspace(0,2*np.pi,90); rr=1+.10*np.sin(3*th+.7)+.07*np.sin(7*th+2.1)+.04*np.sin(13*th)
        ax,ay=40+200*cover,10+62*cover; pts=np.array([[cx+ax*r*np.cos(t),cy+16+ay*r*np.sin(t)*(0.75 if np.sin(t)<0 else 1.0)] for r,t in zip(rr,th)],np.int32)
        m=np.zeros((OH,OW),np.float32); cv2.fillPoly(m,[pts],1,cv2.LINE_AA); m=cv2.GaussianBlur(m,(0,0),2.5)*min(1,cover*1.8)
        yy2,xx2=np.mgrid[0:OH,0:OW].astype(np.float32); dome=np.clip(1.0-.32*((yy2-(cy+16-ay*.6))/max(1,ay*1.4)),.55,1.05)
        speck=1+.09*np.random.default_rng(int(ub*8)).normal(0,1,(OH,OW)).astype(np.float32)
        if variant=='S': mound=np.array([232,150,52],np.float32)/255; edge=np.array([80,36,16],np.float32)/255
        else: mound=PAPER*.94; edge=INK
        ring=np.clip(cv2.dilate(m,np.ones((5,5),np.uint8))-m,0,1)
        img=img*(1-m[...,None])+np.clip(mound*(dome*speck)[...,None],0,1)*m[...,None]; img=img*(1-ring[...,None]*.7)+edge*ring[...,None]*.7
    rn=np.random.default_rng(5)
    for i in range(70):
        x0=rn.uniform(cx-260,cx+260); sp=rn.uniform(60,160); ph=rn.uniform(0,6); t=(ub*sp*0.5+ph*60)%260
        y=cy-190+t*1.5; a=ss(5.8,6.8,ub)*(1-ss(15.4,16.6,ub))
        if a>0.01 and 100<y<620: cv2.circle(img,(int(x0+6*math.sin(ub*2+ph)),int(y)),2,(1,1,1) if variant=='S' else (.11,.08,.09),-1,cv2.LINE_AA)
    return np.clip(img,0,1)
