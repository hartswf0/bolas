// AFTERLIGHT — Film 5 · Past the Red
// Copperplate engraving idiom: sepia ink hatching on plate paper. The spectrum is the only colour.
(function () {
  const INK = '#231b14', PAPER = '#efe5cf', GLASS = '#f6f0e2';

  // ---------- deterministic helpers ----------
  function rng(seed) { let a = seed >>> 0; return () => { a = (a + 0x6D2B79F5) >>> 0; let t = a; t = Math.imul(t ^ (t >>> 15), t | 1); t ^= t + Math.imul(t ^ (t >>> 7), t | 61); return ((t ^ (t >>> 14)) >>> 0) / 4294967296; }; }
  function P(pts) { const p = new Path2D(); p.moveTo(pts[0][0], pts[0][1]); for (let i = 1; i < pts.length; i++) p.lineTo(pts[i][0], pts[i][1]); p.closePath(); return p; }
  function bbOf(pts) { let a = 1e9, b = 1e9, c = -1e9, d = -1e9; pts.forEach(q => { a = Math.min(a, q[0]); b = Math.min(b, q[1]); c = Math.max(c, q[0]); d = Math.max(d, q[1]); }); return [a, b, c, d]; }

  function begin(ctx, w, h, seed) {
    ctx.save(); ctx.scale(w / 1280, h / 720);
    ctx.fillStyle = PAPER; ctx.fillRect(0, 0, 1280, 720);
    const R = rng(seed);
    ctx.fillStyle = 'rgba(120,90,50,0.06)';
    for (let i = 0; i < 900; i++) ctx.fillRect(R() * 1280, R() * 720, 1 + R() * 2, 1);
    return R;
  }
  function end(ctx) {
    // plate mark
    ctx.strokeStyle = 'rgba(35,27,20,0.6)'; ctx.lineWidth = 2; ctx.strokeRect(12, 12, 1256, 696);
    ctx.restore();
  }

  // parallel engraved lines clipped to a region
  function hatch(ctx, R, path, ang, sp, lw, alpha, bb) {
    bb = bb || [0, 0, 1280, 720];
    ctx.save(); ctx.clip(path);
    ctx.strokeStyle = INK; ctx.globalAlpha = alpha == null ? 0.8 : alpha; ctx.lineWidth = lw;
    const cx = (bb[0] + bb[2]) / 2, cy = (bb[1] + bb[3]) / 2, D = Math.hypot(bb[2] - bb[0], bb[3] - bb[1]) / 2 + 6;
    const dx = Math.cos(ang), dy = Math.sin(ang), nx = -dy, ny = dx;
    ctx.beginPath();
    for (let d = -D; d <= D; d += sp) {
      const j = (R() - 0.5) * sp * 0.25, px = cx + nx * (d + j), py = cy + ny * (d + j);
      ctx.moveTo(px - dx * D, py - dy * D);
      for (let s = 1; s <= 3; s++) { const t = -D + 2 * D * s / 3, wb = (R() - 0.5) * 0.9; ctx.lineTo(px + dx * t + nx * wb, py + dy * t + ny * wb); }
    }
    ctx.stroke(); ctx.restore();
  }
  // tonal value 0..4 by layered hatching
  function tone(ctx, R, path, lvl, bb, a0, sp) {
    a0 = a0 == null ? -0.6 : a0; sp = sp || 6;
    const angs = [a0, a0 + 1.35, a0 + 0.7, a0 + 2.05];
    for (let i = 0; i < Math.min(lvl, 4); i++) hatch(ctx, R, path, angs[i], sp * (i ? 1.15 : 1), i ? 0.7 : 0.9, 0.78, bb);
  }
  function solid(ctx, R, pts, lvl, a0, sp, outline) {
    const p = P(pts); ctx.fillStyle = PAPER; ctx.fill(p);
    if (lvl) tone(ctx, R, p, lvl, bbOf(pts), a0, sp);
    if (outline !== false) { ctx.strokeStyle = INK; ctx.lineWidth = outline || 1.4; ctx.stroke(p); }
    return p;
  }
  function rays(ctx, R, path, from, to, n, alpha) {
    ctx.save(); ctx.clip(path); ctx.strokeStyle = INK; ctx.globalAlpha = alpha; ctx.lineWidth = 0.6; ctx.beginPath();
    for (let i = 0; i < n; i++) { const t = R(); const a = [from[0][0] + (from[1][0] - from[0][0]) * t, from[0][1] + (from[1][1] - from[0][1]) * t]; const b = [to[0][0] + (to[1][0] - to[0][0]) * t, to[0][1] + (to[1][1] - to[0][1]) * t]; ctx.moveTo(a[0], a[1]); ctx.lineTo(b[0], b[1]); }
    ctx.stroke(); ctx.restore();
  }

  // the only colour: a prismatic band, violet at x0, red ending at x1 (soft red falloff)
  function spectrum(ctx, R, path, x0, x1, alpha) {
    const g = ctx.createLinearGradient(x0, 0, x1, 0);
    g.addColorStop(0.00, 'rgba(92,58,140,0)');
    g.addColorStop(0.05, 'rgba(92,58,140,1)');
    g.addColorStop(0.18, 'rgba(52,86,168,1)');
    g.addColorStop(0.38, 'rgba(63,150,80,1)');
    g.addColorStop(0.56, 'rgba(232,206,58,1)');
    g.addColorStop(0.73, 'rgba(229,138,43,1)');
    g.addColorStop(0.90, 'rgba(196,38,42,1)');
    g.addColorStop(0.97, 'rgba(160,28,34,0.85)');
    g.addColorStop(1.00, 'rgba(160,28,34,0)');
    ctx.save(); ctx.clip(path); ctx.globalAlpha = alpha == null ? 0.88 : alpha; ctx.fillStyle = g; ctx.fill(path); ctx.restore();
    // engraved horizontal lines laid over the colour, so it stays inside the plate
    hatch(ctx, R, path, 0.0, 3.2, 0.5, 0.16, null);
  }

  // thermometer: ball centre (x,y), tube direction ang, length len, mercury fraction lvl
  function thermo(ctx, R, x, y, ang, len, lvl, o) {
    o = o || {}; const r = o.r || 9, tw = o.tw || r * 0.55, bw = o.bw || r * 1.9, tk = o.tick || r * 0.8;
    ctx.save(); ctx.translate(x, y); ctx.rotate(ang);
    const bp = new Path2D(); bp.rect(r + 5, -bw / 2, len - r - 5, bw);
    ctx.fillStyle = PAPER; ctx.fill(bp); tone(ctx, R, bp, o.boardTone == null ? 1 : o.boardTone, [-10, -bw, len + 10, bw], 0.25, Math.max(2.2, r * 0.32));
    ctx.strokeStyle = INK; ctx.lineWidth = Math.max(0.8, r * 0.1); ctx.stroke(bp);
    ctx.beginPath(); let k = 0;
    for (let t = r + 14; t < len - 6; t += tk, k++) { const big = k % 5 === 0; ctx.moveTo(t, -bw / 2); ctx.lineTo(t, -bw / 2 + (big ? bw * 0.38 : bw * 0.2)); }
    ctx.lineWidth = Math.max(0.6, r * 0.06); ctx.stroke();
    ctx.fillStyle = GLASS; ctx.fillRect(0, -tw / 2, len + 8, tw);
    ctx.strokeStyle = INK; ctx.lineWidth = Math.max(0.7, r * 0.08); ctx.strokeRect(0, -tw / 2, len + 8, tw);
    ctx.beginPath(); ctx.moveTo(0, 0); ctx.lineTo(r + (len - r) * lvl, 0); ctx.lineWidth = tw * 0.5; ctx.strokeStyle = INK; ctx.stroke();
    ctx.beginPath(); ctx.arc(0, 0, r, 0, 7); ctx.fillStyle = '#15100b'; ctx.fill();
    ctx.beginPath(); ctx.arc(-r * 0.3, -r * 0.38, r * 0.2, 0, 7); ctx.fillStyle = 'rgba(239,229,207,0.55)'; ctx.fill();
    ctx.restore();
  }
  // small inclined plane under a thermometer (screen coords)
  function wedge(ctx, R, x, y, ang, r, L) {
    const yb = y + r + 1, xb = x + Math.cos(ang) * L, yc = y + Math.sin(ang) * L + r * 1.1;
    solid(ctx, R, [[x + r * 0.6, yb], [xb, yb], [xb, yc]], 2, 1.2, Math.max(2.5, r * 0.35), Math.max(0.8, r * 0.1));
  }

  // head in profile facing left, local unit ~100 tall
  function head(ctx, R, x, y, s, tilt) {
    ctx.save(); ctx.translate(x, y); ctx.rotate(tilt || 0); ctx.scale(s, s);
    const lw = 1.5 / Math.max(s, 0.6);
    const H = new Path2D('M18,-58 C-8,-64 -28,-52 -31,-30 L-33,-17 L-45,1 L-35,7 L-37,14 L-34,19 L-36,24 Q-36,37 -23,41 Q-6,45 8,33 L12,58 L38,58 L38,26 Q52,6 48,-22 Q44,-52 18,-58 Z');
    ctx.fillStyle = PAPER; ctx.fill(H);
    const shade = P([[0, -70], [60, -70], [60, 60], [-8, 60], [-14, 20]]);
    ctx.save(); ctx.clip(H); hatch(ctx, R, shade, -0.9, 3.2, 0.6, 0.6, [-10, -70, 60, 60]); ctx.restore();
    ctx.strokeStyle = INK; ctx.lineWidth = lw; ctx.stroke(H);
    // powdered hair, tied back
    const hair = new Path2D('M-28,-38 Q-22,-64 16,-62 Q48,-56 51,-20 Q53,6 42,26 Q36,10 31,-4 Q22,-28 -4,-38 Q-17,-43 -28,-38 Z');
    ctx.fillStyle = PAPER; ctx.fill(hair);
    ctx.save(); ctx.clip(hair); ctx.strokeStyle = INK; ctx.globalAlpha = 0.55; ctx.lineWidth = 0.7 / Math.max(s, 0.6); ctx.beginPath();
    for (let i = 0; i < 16; i++) { const o = i * 4.2 - 30; ctx.moveTo(-26 + o * 0.2, -44 + o * 0.1); ctx.quadraticCurveTo(20 + o * 0.3, -64 + o, 50, -16 + o * 0.8); }
    ctx.stroke(); ctx.restore();
    ctx.strokeStyle = INK; ctx.lineWidth = lw; ctx.stroke(hair);
    ctx.beginPath(); ctx.ellipse(30, -8, 8, 12, 0.2, 0, 7); ctx.fillStyle = PAPER; ctx.fill(); ctx.stroke();
    ctx.beginPath(); ctx.ellipse(30, -8, 4, 7, 0.2, 0, 7); ctx.stroke();
    ctx.fillStyle = '#15100b'; ctx.fill(new Path2D('M40,26 L52,18 L50,34 Z M40,26 L30,40 L44,38 Z'));
    // features
    ctx.beginPath();
    ctx.moveTo(-31, -24); ctx.quadraticCurveTo(-22, -28, -12, -23);   // brow
    ctx.moveTo(-28, -14); ctx.quadraticCurveTo(-22, -17, -16, -13);   // upper lid
    ctx.moveTo(-36, 19); ctx.lineTo(-29, 19);                         // mouth
    ctx.moveTo(-36, 5); ctx.quadraticCurveTo(-32, 3, -30, 6);          // nostril
    ctx.moveTo(-24, 2); ctx.quadraticCurveTo(-16, 14, -22, 28);        // cheek
    ctx.moveTo(-6, 30); ctx.quadraticCurveTo(4, 26, 8, 33);            // jaw
    ctx.stroke();
    ctx.beginPath(); ctx.arc(-24, -13.5, 1.8, 0, 7); ctx.fillStyle = INK; ctx.fill();
    ctx.restore();
  }

  // hands, fingers along +x. kind: 'fist' (gripping across a tube) or 'open'
  function hand(ctx, R, x, y, ang, s, kind, flip) {
    ctx.save(); ctx.translate(x, y); ctx.rotate(ang); ctx.scale(s, flip ? -s : s);
    const lw = 1.5 / s;
    // lace cuff
    const cuff = new Path2D('M-16,-22 L-2,-18 L-6,-12 L0,-6 L-5,0 L0,6 L-6,12 L-2,18 L-16,22 Z');
    ctx.fillStyle = PAPER; ctx.fill(cuff); ctx.strokeStyle = INK; ctx.lineWidth = lw * 0.8; ctx.stroke(cuff);
    if (kind === 'fist') {
      const f = new Path2D('M0,-15 Q14,-19 28,-17 Q42,-16 46,-6 Q49,6 44,15 Q36,22 22,21 Q8,20 0,15 Z');
      ctx.fillStyle = PAPER; ctx.fill(f);
      hatch(ctx, R, f, 0.3, 2.6 / s * 1.4, 0.5 / s, 0.55, [0, 0, 50, 22]);
      ctx.lineWidth = lw; ctx.strokeStyle = INK; ctx.stroke(f);
      ctx.beginPath();
      for (let i = 0; i < 3; i++) { const yy = -8 + i * 8; ctx.moveTo(30, yy); ctx.quadraticCurveTo(40, yy - 1, 47, yy + 1); }
      ctx.lineWidth = lw * 0.8; ctx.stroke();
      const th = new Path2D('M6,-15 Q20,-31 40,-25 Q47,-22 44,-17 Q30,-15 16,-11 Z');
      ctx.fillStyle = PAPER; ctx.fill(th); ctx.lineWidth = lw; ctx.stroke(th);
      ctx.beginPath(); ctx.ellipse(40, -21, 3.2, 2.2, -0.2, 0, 7); ctx.lineWidth = lw * 0.6; ctx.stroke();
    } else {
      const palm = new Path2D('M0,-15 Q16,-19 32,-16 L35,15 Q16,19 0,15 Z');
      ctx.fillStyle = PAPER; ctx.fill(palm); hatch(ctx, R, palm, 0.4, 3.4, 0.5 / s, 0.5, [0, -20, 36, 20]);
      ctx.strokeStyle = INK; ctx.lineWidth = lw; ctx.stroke(palm);
      const fingers = [[-12, 33, -0.10], [-3, 39, -0.02], [6, 36, 0.06], [13, 27, 0.16]];
      fingers.forEach(([yy, L, a]) => {
        ctx.save(); ctx.translate(31, yy); ctx.rotate(a);
        const fp = new Path2D(); fp.moveTo(0, -4); fp.lineTo(L * 0.55, -3.8); fp.quadraticCurveTo(L * 0.8, -2, L, 0); fp.quadraticCurveTo(L * 0.85, 4, L * 0.55, 3.8); fp.lineTo(0, 4); fp.closePath();
        ctx.fillStyle = PAPER; ctx.fill(fp); ctx.lineWidth = lw; ctx.stroke(fp);
        ctx.beginPath(); ctx.moveTo(L * 0.5, -3); ctx.lineTo(L * 0.5, 3); ctx.lineWidth = lw * 0.5; ctx.stroke();
        ctx.restore();
      });
      ctx.save(); ctx.translate(10, -15); ctx.rotate(-0.75);
      const tp = new Path2D('M0,-5 L20,-5 Q30,-3 30,0 Q28,5 20,5 L0,5 Z');
      ctx.fillStyle = PAPER; ctx.fill(tp); ctx.lineWidth = lw; ctx.stroke(tp); ctx.restore();
    }
    ctx.restore();
  }
  function sleeve(ctx, R, pts, lvl) { return solid(ctx, R, pts, lvl == null ? 3 : lvl, -0.9, 4.5, 1.6); }
