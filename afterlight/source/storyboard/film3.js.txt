// AFTERLIGHT film 3 — The Last Beacon
// Keyframe module. Plain browser JS, Canvas 2D only, deterministic.
(function () {
  // ---------- shared drawing kit ----------
  const W0 = 1280, H0 = 720;
  function rng(seed) { let s = seed >>> 0; return () => { s = (s * 1664525 + 1013904223) >>> 0; return s / 4294967296; }; }
  function setup(ctx, w, h) { ctx.save(); ctx.scale(w / W0, h / H0); }

  // palette: tinted night layers
  const P = {
    skyTop: '#121827', skyMid: '#2b3352', dusk: '#b8683f', duskHi: '#e0a060',
    far: '#323a58', mid: '#1d2238', near: '#0c0e18', black: '#06070c',
    storm: '#161a28', fire: '#ffb347', fireCore: '#fff1c4', ember: '#c4471c',
    warm: '#f0b060', rim: '#ff9a3c'
  };

  function sky(ctx, top, mid, low, lowY) {
    const g = ctx.createLinearGradient(0, 0, 0, lowY);
    g.addColorStop(0, top); g.addColorStop(0.6, mid); g.addColorStop(1, low);
    ctx.fillStyle = g; ctx.fillRect(0, 0, W0, H0);
  }
  // ridge silhouette: y(x) = base + sum of sines (seeded phases) + optional peaks
  function ridgeY(x, o) {
    let y = o.base;
    for (let i = 0; i < o.f.length; i++) y += Math.sin(x * o.f[i] + o.ph[i]) * o.a[i];
    if (o.peaks) for (const p of o.peaks) { const d = (x - p.x) / p.wd; y -= p.ht * Math.exp(-d * d); }
    return y;
  }
  function ridge(ctx, o, color) {
    ctx.fillStyle = color; ctx.beginPath(); ctx.moveTo(0, H0);
    for (let x = 0; x <= W0; x += 8) ctx.lineTo(x, ridgeY(x, o));
    ctx.lineTo(W0, H0); ctx.closePath(); ctx.fill();
  }
  function R(base, seed, amp, peaks) {
    const r = rng(seed);
    return { base, f: [0.004 + r() * 0.002, 0.011 + r() * 0.004, 0.031 + r() * 0.01], ph: [r() * 6, r() * 6, r() * 6], a: [amp, amp * 0.4, amp * 0.12], peaks };
  }
  function glow(ctx, x, y, r, col, a) {
    const g = ctx.createRadialGradient(x, y, 0, x, y, r);
    g.addColorStop(0, col); g.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.globalAlpha = a; ctx.fillStyle = g; ctx.fillRect(x - r, y - r, r * 2, r * 2); ctx.globalAlpha = 1;
  }
  // a flame stand: tongues of fire, seeded so each fire has its own shape
  function flame(ctx, x, y, size, seed, strength) {
    const r = rng(seed);
    glow(ctx, x, y - size * 0.6, size * 4, 'rgba(255,150,60,0.9)', 0.45 * strength);
    const tongues = 7;
    for (let i = 0; i < tongues; i++) {
      const dx = (r() - 0.5) * size * 1.1, hgt = size * (1.2 + r() * 1.6) * strength, wd = size * (0.25 + r() * 0.25);
      ctx.fillStyle = i % 3 === 0 ? P.ember : P.fire;
      ctx.beginPath(); ctx.moveTo(x + dx - wd, y);
      ctx.quadraticCurveTo(x + dx - wd * 0.8, y - hgt * 0.6, x + dx + (r() - 0.5) * wd, y - hgt);
      ctx.quadraticCurveTo(x + dx + wd * 0.8, y - hgt * 0.5, x + dx + wd, y);
      ctx.closePath(); ctx.fill();
    }
    ctx.fillStyle = P.fireCore; ctx.beginPath();
    ctx.ellipse(x, y - size * 0.35 * strength, size * 0.35, size * 0.6 * strength, 0, 0, Math.PI * 2); ctx.fill();
  }
  // beacon crib: stone footing + log-cabin stack of split wood
  function crib(ctx, x, y, s, col, seed, rimCol) {
    const r = rng(seed);
    ctx.fillStyle = col;
    // stone footing
    ctx.beginPath(); ctx.moveTo(x - 70 * s, y); ctx.lineTo(x - 62 * s, y - 16 * s); ctx.lineTo(x + 62 * s, y - 16 * s); ctx.lineTo(x + 70 * s, y); ctx.closePath(); ctx.fill();
    for (let i = 0; i < 9; i++) {
      const yy = y - 16 * s - i * 11 * s, inset = i * 3.2 * s, cross = i % 2;
      if (cross) { // log ends facing camera
        for (const sx of [-1, 1]) { ctx.beginPath(); ctx.ellipse(x + sx * (46 * s - inset), yy - 5 * s, 7 * s, 5.5 * s, 0, 0, Math.PI * 2); ctx.fill(); }
        ctx.fillRect(x - 46 * s + inset, yy - 9 * s, 92 * s - inset * 2, 4 * s);
      } else {
        ctx.fillRect(x - 58 * s + inset + r() * 3 * s, yy - 10 * s, 116 * s - inset * 2, 9 * s);
      }
    }
    // brush crown
    for (let i = 0; i < 26; i++) {
      const a = -Math.PI / 2 + (r() - 0.5) * 1.8, L = (14 + r() * 22) * s, bx = x + (r() - 0.5) * 50 * s, by = y - 112 * s;
      ctx.strokeStyle = col; ctx.lineWidth = 2 * s; ctx.beginPath(); ctx.moveTo(bx, by); ctx.lineTo(bx + Math.cos(a) * L, by + Math.sin(a) * L); ctx.stroke();
    }
    if (rimCol) { // rim light on the right edge of the stack
      ctx.strokeStyle = rimCol; ctx.lineWidth = 1.5 * s; ctx.beginPath();
      ctx.moveTo(x + 70 * s, y); ctx.lineTo(x + 30 * s, y - 112 * s); ctx.stroke();
    }
  }
  // human silhouette. feet at (x,y), height hgt. pose controls arms.
  // pose: {face:1|-1, lean, lArm:[angle,len,elbowBend], rArm:..., hood, cloak, stride}
  function person(ctx, x, y, hgt, col, pose) {
    const u = hgt / 100, f = pose.face || 1, lean = (pose.lean || 0) * u;
    ctx.fillStyle = col; ctx.strokeStyle = col; ctx.lineCap = 'round'; ctx.lineJoin = 'round';
    const hipY = y - 48 * u, shY = y - 80 * u, hx = x + lean * 0.4, sx = x + lean;
    // legs
    const st = (pose.stride || 0.12) * 40 * u;
    ctx.lineWidth = 8 * u;
    ctx.beginPath(); ctx.moveTo(hx - 3 * u, hipY); ctx.lineTo(x - st, y - 2 * u); ctx.stroke();
    ctx.beginPath(); ctx.moveTo(hx + 3 * u, hipY); ctx.lineTo(x + st, y - 2 * u); ctx.stroke();
    // feet
    ctx.beginPath(); ctx.ellipse(x - st + f * 3 * u, y - 2 * u, 6 * u, 2.5 * u, 0, 0, Math.PI * 2); ctx.fill();

    ctx.beginPath();
    if (pose.cloak) {
      ctx.moveTo(sx - 11 * u, shY); ctx.lineTo(sx + 11 * u, shY);
      ctx.quadraticCurveTo(hx + 20 * u, y - 40 * u, hx + 17 * u, y - 20 * u);
      ctx.lineTo(hx - 17 * u, y - 18 * u); ctx.quadraticCurveTo(hx - 20 * u, y - 40 * u, sx - 11 * u, shY);
    } else {
      ctx.moveTo(sx - 11 * u, shY); ctx.lineTo(sx + 11 * u, shY); ctx.lineTo(hx + 9 * u, hipY + 4 * u); ctx.lineTo(hx - 9 * u, hipY + 4 * u);
    }
    ctx.closePath(); ctx.fill();
    // head + neck
    const headX = sx + f * 1.5 * u + lean * 0.15, headY = shY - 12 * u;
    ctx.lineWidth = 6 * u; ctx.beginPath(); ctx.moveTo(sx, shY); ctx.lineTo(headX, headY); ctx.stroke();
    ctx.beginPath(); ctx.ellipse(headX, headY - 2 * u, 7.5 * u, 9 * u, 0, 0, Math.PI * 2); ctx.fill();
    // nose hint (profile)
    ctx.beginPath(); ctx.moveTo(headX + f * 6.5 * u, headY - 3 * u); ctx.lineTo(headX + f * 9.5 * u, headY + 1 * u); ctx.lineTo(headX + f * 6 * u, headY + 2 * u); ctx.fill();
    if (pose.hood) { ctx.beginPath(); ctx.moveTo(headX - f * 10 * u, headY + 8 * u); ctx.quadraticCurveTo(headX - f * 12 * u, headY - 14 * u, headX + f * 2 * u, headY - 12 * u); ctx.quadraticCurveTo(headX + f * 7 * u, headY - 10 * u, headX + f * 5 * u, headY - 4 * u); ctx.lineTo(headX - f * 4 * u, headY + 9 * u); ctx.fill(); }
    // arms: shoulder -> elbow -> hand, angles in radians (0 = right, PI/2 = down)
    function arm(side, spec) {
      if (!spec) return;
      const [ang, len, bend] = spec, L = len * u;
      const ax = sx + side * 9 * u, ay = shY + 3 * u;
      const ex = ax + Math.cos(ang) * L * 0.5, ey = ay + Math.sin(ang) * L * 0.5;
      const a2 = ang + (bend || 0);
      const hx2 = ex + Math.cos(a2) * L * 0.5, hy2 = ey + Math.sin(a2) * L * 0.5;
      ctx.lineWidth = 6.5 * u; ctx.beginPath(); ctx.moveTo(ax, ay); ctx.lineTo(ex, ey); ctx.lineTo(hx2, hy2); ctx.stroke();
      // hand: palm + thumb so it reads as a hand
      ctx.beginPath(); ctx.ellipse(hx2 + Math.cos(a2) * 3 * u, hy2 + Math.sin(a2) * 3 * u, 4.6 * u, 3.4 * u, a2, 0, Math.PI * 2); ctx.fill();
      ctx.lineWidth = 2.2 * u; ctx.beginPath(); ctx.moveTo(hx2, hy2); ctx.lineTo(hx2 + Math.cos(a2 - 0.9) * 6 * u, hy2 + Math.sin(a2 - 0.9) * 6 * u); ctx.stroke();
    }
    arm(-1, pose.lArm); arm(1, pose.rArm);
    if (pose.staff) { ctx.lineWidth = 3 * u; ctx.beginPath(); ctx.moveTo(pose.staff[0], pose.staff[1]); ctx.lineTo(pose.staff[2], pose.staff[3]); ctx.stroke(); }
  }
  function stormFront(ctx, x0, x1, yTop, yBase, seed, col, rainCol, rainN) {
    const r = rng(seed);
    ctx.fillStyle = col; ctx.beginPath(); ctx.moveTo(x0, yTop);
    for (let x = x0; x <= x1; x += 22) ctx.lineTo(x, yBase - r() * 30 - Math.sin(x * 0.01) * 18);
    ctx.lineTo(x1, yTop); ctx.closePath(); ctx.fill();
    // billows
    for (let i = 0; i < 18; i++) { ctx.beginPath(); ctx.arc(x0 + r() * (x1 - x0), yBase - 20 - r() * 60, 30 + r() * 50, 0, Math.PI * 2); ctx.fill(); }
    ctx.strokeStyle = rainCol; ctx.lineWidth = 1;
    for (let i = 0; i < rainN; i++) { const rx = x0 + r() * (x1 - x0), ry = yBase - 10 + r() * 140, L = 40 + r() * 60; ctx.globalAlpha = 0.25 + r() * 0.35; ctx.beginPath(); ctx.moveTo(rx, ry); ctx.lineTo(rx - L * 0.35, ry + L); ctx.stroke(); }
    ctx.globalAlpha = 1;
  }
  function stars(ctx, seed, n, yMax, a) {
    const r = rng(seed); ctx.fillStyle = '#dfe6ff';
    for (let i = 0; i < n; i++) { ctx.globalAlpha = a * (0.3 + r() * 0.7); ctx.fillRect(r() * W0, r() * yMax, 1.4, 1.4); }
    ctx.globalAlpha = 1;
  }
  function house(ctx, x, y, s, col, lit) { // small stone house: walls + pitched roof + window
    ctx.fillStyle = col; ctx.fillRect(x, y - 30 * s, 46 * s, 30 * s);
    ctx.beginPath(); ctx.moveTo(x - 5 * s, y - 30 * s); ctx.lineTo(x + 23 * s, y - 50 * s); ctx.lineTo(x + 51 * s, y - 30 * s); ctx.fill();
    if (lit) { ctx.fillStyle = P.warm; ctx.fillRect(x + 28 * s, y - 22 * s, 7 * s, 7 * s); glow(ctx, x + 31 * s, y - 18 * s, 18 * s, 'rgba(240,170,90,0.8)', 0.35); }
  }
  function grain(ctx, seed, n, a) { const r = rng(seed); for (let i = 0; i < n; i++) { ctx.fillStyle = r() > 0.5 ? '#fff' : '#000'; ctx.globalAlpha = a * r(); ctx.fillRect(r() * W0, r() * H0, 1.5, 1.5); } ctx.globalAlpha = 1; }
  function vignette(ctx, a) { const g = ctx.createRadialGradient(W0 / 2, H0 / 2, H0 * 0.35, W0 / 2, H0 / 2, H0 * 0.95); g.addColorStop(0, 'rgba(0,0,0,0)'); g.addColorStop(1, 'rgba(0,0,0,' + a + ')'); ctx.fillStyle = g; ctx.fillRect(0, 0, W0, H0); }
