/* AFTERLIGHT · Film 1 · The Stone That Remembered
   Keyframe module. Multiplane cut paper; one room, one cast, one stone.
   All drawing is Canvas 2D, deterministic (seeded PRNG), no text in picture. */
(function () {
  'use strict';

  /* ---------- seeded random ---------- */
  function rng(seed) {
    let a = seed >>> 0;
    return function () {
      a = (a + 0x6D2B79F5) >>> 0;
      let t = a;
      t = Math.imul(t ^ (t >>> 15), t | 1);
      t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  /* ---------- geometry ---------- */
  function ell(cx, cy, rx, ry, rot, n) {
    n = n || 30; rot = rot || 0;
    const p = [], c = Math.cos(rot), s = Math.sin(rot);
    for (let i = 0; i < n; i++) {
      const a = i / n * Math.PI * 2, x = Math.cos(a) * rx, y = Math.sin(a) * ry;
      p.push([cx + x * c - y * s, cy + x * s + y * c]);
    }
    return p;
  }
  function capsule(x1, y1, x2, y2, r1, r2) {
    if (r2 == null) r2 = r1;
    const a = Math.atan2(y2 - y1, x2 - x1), p = [];
    for (let i = 0; i <= 8; i++) { const t = a + Math.PI / 2 + i / 8 * Math.PI; p.push([x1 + Math.cos(t) * r1, y1 + Math.sin(t) * r1]); }
    for (let i = 0; i <= 8; i++) { const t = a - Math.PI / 2 + i / 8 * Math.PI; p.push([x2 + Math.cos(t) * r2, y2 + Math.sin(t) * r2]); }
    return p;
  }
  /* torn / scissor-cut edge: subdivide, jitter perpendicular */
  function jag(pts, R, amp, step) {
    step = step || 10;
    const out = [];
    for (let i = 0; i < pts.length; i++) {
      const a = pts[i], b = pts[(i + 1) % pts.length];
      const dx = b[0] - a[0], dy = b[1] - a[1], L = Math.hypot(dx, dy) || 1;
      const n = Math.max(1, Math.round(L / step)), nx = -dy / L, ny = dx / L;
      for (let k = 0; k < n; k++) {
        const t = k / n, j = (R() - 0.5) * amp;
        out.push([a[0] + dx * t + nx * j, a[1] + dy * t + ny * j]);
      }
    }
    return out;
  }
  function path(ctx, pts) {
    ctx.beginPath(); ctx.moveTo(pts[0][0], pts[0][1]);
    for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i][0], pts[i][1]);
    ctx.closePath();
  }
  /* one cut-paper piece: cast shadow on the plane below, fill, pale cut edge */
  function piece(ctx, pts, fill, R, o) {
    o = o || {};
    const amp = o.amp == null ? 1.6 : o.amp, sh = o.sh == null ? 3 : o.sh;
    const p = amp > 0 ? jag(pts, R, amp, o.step) : pts;
    if (sh > 0) {
      ctx.save(); ctx.translate(sh * 0.8, sh);
      ctx.fillStyle = o.shc || 'rgba(28,14,6,0.32)'; path(ctx, p); ctx.fill(); ctx.restore();
    }
    if (o.grad) {
      const g = ctx.createLinearGradient(o.grad[0], o.grad[1], o.grad[2], o.grad[3]);
      g.addColorStop(0, fill); g.addColorStop(1, o.grad[4]); ctx.fillStyle = g;
    } else ctx.fillStyle = fill;
    path(ctx, p); ctx.fill();
    if (o.edge !== false) { ctx.strokeStyle = o.edgec || 'rgba(255,238,214,0.22)'; ctx.lineWidth = o.ew || 0.8; ctx.stroke(); }
    return p;
  }
  function line(ctx, pts, col, wdt) {
    ctx.beginPath(); ctx.moveTo(pts[0][0], pts[0][1]);
    for (let i = 1; i < pts.length; i++) ctx.lineTo(pts[i][0], pts[i][1]);
    ctx.strokeStyle = col; ctx.lineWidth = wdt; ctx.lineCap = 'round'; ctx.lineJoin = 'round'; ctx.stroke();
  }

  /* ---------- palette ---------- */
  const C = {
    wall: '#b48a58', wallDark: '#8c6a43', floor: '#6a4529', plank: '#5a3a22',
    wood: '#3d2617', woodL: '#5a3a24', sill: '#a8957a',
    stone: '#7f7a70', stoneD: '#5f5a52', stoneL: '#a39d92',
    eSkin: '#c4977a', eSkinD: '#9c7058', eShawl: '#6f4f41', eShawlD: '#563a30', eDress: '#3f3436', eHair: '#cfc6bb',
    cSkin: '#dba882', cSkinD: '#b5815f', cSmock: '#5f7184', cSmockD: '#465566', cHair: '#4a2f1e', cSkirt: '#7b5d3b',
    cloth: '#d6cab0', clothD: '#b3a68a', chair: '#2e1d12', mirror: '#c9d2d4', mirrorRim: '#6b5232'
  };

  /* ---------- the set (world space 1280 x 720) ----------
     window: x 140..420, y 110..420 ; sill y 420..446 ; stone on sill at (330,410)
     elder seated at x ~900, hands in lap at (900,478)
     sun beam = window swept by (380,260)                                   */
  const WIN = { x0: 140, x1: 420, y0: 110, y1: 420 };
  const SUN = [380, 260];
  const STONE = [330, 408];
  const HANDS = [900, 478];
  const PATCH = [968, 486];

  function inBeam(x, y) {
    const tx0 = (x - WIN.x1) / SUN[0], tx1 = (x - WIN.x0) / SUN[0];
    const ty0 = (y - WIN.y1) / SUN[1], ty1 = (y - WIN.y0) / SUN[1];
    const lo = Math.max(0, tx0, ty0), hi = Math.min(1, tx1, ty1);
    return lo <= hi;
  }

  function setCam(ctx, w, h, cam, depth) {
    const s = w / 1280, z = cam.z * s;
    const px = (cam.cx - 640) * (depth - 1) * 0.35;
    ctx.setTransform(z, 0, 0, z, w / 2 - (cam.cx + px) * z, h / 2 - cam.cy * z);
  }

  /* back plane: wall, floor, window, sill, stone */
  function backPlane(ctx, st, R) {
    // wall
    piece(ctx, [[-400, -200], [1700, -200], [1700, 540], [-400, 540]], C.wall, R, { amp: 0, sh: 0, edge: false, grad: [0, 0, 0, 540, C.wallDark] });
    // wainscot strip
    piece(ctx, [[-400, 470], [1700, 470], [1700, 545], [-400, 545]], '#7d5a37', R, { amp: 1.2, sh: 4 });
    // floor
    piece(ctx, [[-400, 535], [1700, 535], [1700, 1100], [-400, 1100]], C.floor, R, { amp: 1, sh: 0, grad: [0, 535, 0, 800, '#4a2f1b'] });
    ctx.save();
    for (let i = 0; i < 9; i++) { const y = 560 + i * i * 4.2 + i * 8; line(ctx, [[-400, y], [1700, y + (R() - 0.5) * 3]], 'rgba(40,22,10,0.35)', 1.4); }
    ctx.restore();
    // window recess
    piece(ctx, [[WIN.x0 - 26, WIN.y0 - 26], [WIN.x1 + 26, WIN.y0 - 26], [WIN.x1 + 26, WIN.y1 + 4], [WIN.x0 - 26, WIN.y1 + 4]], '#6e4f31', R, { amp: 1.4, sh: 5 });
    // sky through the glass (photographed plate in production; painted stand-in here)
    const g = ctx.createLinearGradient(0, WIN.y0, 0, WIN.y1);
    if (st.sun > 0.5) { g.addColorStop(0, '#d9c8a4'); g.addColorStop(0.55, '#f2d8a2'); g.addColorStop(0.8, '#f6c98a'); g.addColorStop(1, '#c99a68'); }
    else { g.addColorStop(0, '#2c3550'); g.addColorStop(0.6, '#4a5470'); g.addColorStop(0.88, '#8a6d6a'); g.addColorStop(1, '#5e4a50'); }
    ctx.fillStyle = g; ctx.fillRect(WIN.x0, WIN.y0, WIN.x1 - WIN.x0, WIN.y1 - WIN.y0);
    // distant roofline cut-out across the bottom of the panes
    const roof = [[WIN.x0, WIN.y1], [WIN.x0, 352], [190, 352], [190, 330], [232, 312], [274, 330], [274, 356], [330, 356], [330, 366], [372, 366], [372, 344], [WIN.x1, 344], [WIN.x1, WIN.y1]];
    piece(ctx, roof, st.sun > 0.5 ? '#6d5a4c' : '#232838', R, { amp: 0.8, sh: 0, edge: false });
    if (st.sun > 0.5) { // low disc of sun, half behind roof, pale (not a glow)
      ctx.save(); ctx.beginPath(); ctx.rect(WIN.x0, WIN.y0, WIN.x1 - WIN.x0, 350 - WIN.y0); ctx.clip();
      ctx.fillStyle = '#fff1d0'; ctx.beginPath(); ctx.arc(232, 336, 22, 0, Math.PI * 2); ctx.fill(); ctx.restore();
    }
    // frame + muntins
    const fw = C.wood;
    piece(ctx, [[WIN.x0 - 8, WIN.y0 - 8], [WIN.x1 + 8, WIN.y0 - 8], [WIN.x1 + 8, WIN.y0 + 4], [WIN.x0 - 8, WIN.y0 + 4]], fw, R, { amp: 0.8 });
    piece(ctx, [[WIN.x0 - 8, WIN.y0 - 8], [WIN.x0 + 4, WIN.y0 - 8], [WIN.x0 + 4, WIN.y1], [WIN.x0 - 8, WIN.y1]], fw, R, { amp: 0.8 });
    piece(ctx, [[WIN.x1 - 4, WIN.y0 - 8], [WIN.x1 + 8, WIN.y0 - 8], [WIN.x1 + 8, WIN.y1], [WIN.x1 - 4, WIN.y1]], fw, R, { amp: 0.8 });
    piece(ctx, [[279, WIN.y0], [285, WIN.y0], [285, WIN.y1], [279, WIN.y1]], fw, R, { amp: 0.6, sh: 2 });
    for (const y of [210, 315]) piece(ctx, [[WIN.x0, y], [WIN.x1, y], [WIN.x1, y + 6], [WIN.x0, y + 6]], fw, R, { amp: 0.6, sh: 2 });
    // curtain, left
    piece(ctx, [[60, 70], [150, 70], [140, 200], [132, 330], [150, 480], [70, 486], [84, 300], [74, 180]], '#7a5d4a', R, { amp: 2.2, sh: 6, grad: [60, 0, 150, 0, '#634b3c'] });
    for (let i = 0; i < 4; i++) line(ctx, [[86 + i * 15, 80], [92 + i * 14, 300], [86 + i * 16, 480]], 'rgba(40,24,14,0.35)', 2);
    // sill
    piece(ctx, [[WIN.x0 - 34, WIN.y1], [WIN.x1 + 34, WIN.y1], [WIN.x1 + 44, WIN.y1 + 16], [WIN.x0 - 44, WIN.y1 + 16]], C.sill, R, { amp: 1, sh: 3 });
    piece(ctx, [[WIN.x0 - 44, WIN.y1 + 16], [WIN.x1 + 44, WIN.y1 + 16], [WIN.x1 + 44, WIN.y1 + 28], [WIN.x0 - 44, WIN.y1 + 28]], '#7f6c55', R, { amp: 1, sh: 6 });
    // a cup on the sill for scale
    piece(ctx, [[196, 386], [222, 386], [219, 420], [199, 420]], '#d8cfbd', R, { amp: 0.7, sh: 3 });
    if (st.stoneOnSill) stone(ctx, STONE[0], STONE[1], 1, st, R);
  }

  function stone(ctx, x, y, k, st, R) {
    const p = ell(x, y, 34 * k, 16 * k, -0.06, 34);
    piece(ctx, p, C.stone, R, { amp: 1.4 * k, sh: 4 * k, grad: [x, y - 16 * k, x, y + 16 * k, C.stoneD] });
    // flatter top face, lit only as grey light on grey stone
    piece(ctx, ell(x - 4 * k, y - 6 * k, 24 * k, 7 * k, -0.06, 24), st.sun > 0.5 ? '#b2ab9f' : '#8a857c', R, { amp: 0.8 * k, sh: 0, edge: false });
    const r = rng(77);
    ctx.fillStyle = 'rgba(40,36,30,0.45)';
    for (let i = 0; i < 26; i++) { const a = r() * 6.28, d = Math.sqrt(r()); ctx.fillRect(x + Math.cos(a) * 30 * k * d, y + Math.sin(a) * 13 * k * d, 1.4 * k, 1.4 * k); }
    ctx.fillStyle = 'rgba(230,225,214,0.35)';
    for (let i = 0; i < 14; i++) { const a = r() * 6.28, d = Math.sqrt(r()); ctx.fillRect(x + Math.cos(a) * 28 * k * d, y + Math.sin(a) * 12 * k * d - 2 * k, 1.1 * k, 1.1 * k); }
  }

  /* sunbeam: pale translucent plane + floor patch + dust */
  function beam(ctx, st, R) {
    if (st.sun < 0.5) return;
    ctx.save();
    ctx.globalCompositeOperation = 'screen';
    const q = [[WIN.x1, WIN.y0], [WIN.x1 + SUN[0], WIN.y0 + SUN[1]], [WIN.x1 + SUN[0], WIN.y1 + SUN[1]], [WIN.x0 + SUN[0], WIN.y1 + SUN[1]], [WIN.x0, WIN.y1], [WIN.x0, WIN.y0]];
    const g = ctx.createLinearGradient(WIN.x0, WIN.y0, WIN.x1 + SUN[0], WIN.y1 + SUN[1]);
    g.addColorStop(0, 'rgba(255,226,170,0.20)'); g.addColorStop(1, 'rgba(255,214,150,0.05)');
    ctx.fillStyle = g; path(ctx, q); ctx.fill();
    // patch on the floor (window shape, muntin shadows)
    const fy = 610, sk = 0.5;
    const pt = (u, v) => [WIN.x0 + SUN[0] - 40 + u * 300 + v * 120, fy - 36 + v * 100 * sk];
    ctx.fillStyle = 'rgba(255,220,160,0.26)';
    for (const [u0, u1] of [[0, 0.48], [0.52, 1]]) for (const [v0, v1] of [[0, 0.31], [0.35, 0.65], [0.69, 1]]) {
      path(ctx, [pt(u0, v0), pt(u1, v0), pt(u1, v1), pt(u0, v1)]); ctx.fill();
    }
    // light on the sill and stone
    ctx.fillStyle = 'rgba(255,230,180,0.22)';
    path(ctx, [[WIN.x0 - 30, WIN.y1 + 1], [WIN.x1 + 30, WIN.y1 + 1], [WIN.x1 + 40, WIN.y1 + 15], [WIN.x0 - 40, WIN.y1 + 15]]); ctx.fill();
    // dust motes
    const r = rng(311);
    for (let i = 0; i < 520; i++) {
      const x = WIN.x0 + r() * (WIN.x1 - WIN.x0 + SUN[0]), y = WIN.y0 + r() * (WIN.y1 - WIN.y0 + SUN[1]);
      if (!inBeam(x, y)) continue;
      const a = 0.25 + r() * 0.5, s = 0.6 + r() * 1.6;
      ctx.fillStyle = 'rgba(255,240,210,' + a.toFixed(2) + ')';
      ctx.beginPath(); ctx.arc(x, y, s, 0, 6.283); ctx.fill();
    }
    ctx.restore();
  }

  /* ---------- hands ----------
     local frame: origin at wrist, +y toward fingertips, unit ~ world px */
  function hand(ctx, x, y, ang, k, mode, skin, skinD, side, R, sh) {
    ctx.save(); ctx.translate(x, y); ctx.rotate(ang); ctx.scale(k * side, k);
    const o = { amp: 0.7, sh: sh == null ? 2 : sh, step: 5, edgec: 'rgba(255,236,214,0.3)', ew: 0.5 };
    const fx = [-7.5, -2.5, 2.5, 7.5], fl = [17, 21, 20, 15];
    if (mode === 'cup') {
      piece(ctx, ell(0, 14, 12, 13, 0, 26), skin, R, o);
      piece(ctx, ell(0, 15, 8, 8, 0, 20), skinD, R, { amp: 0.4, sh: 0, edge: false });
      for (let i = 0; i < 4; i++) piece(ctx, capsule(fx[i], 24, fx[i] * 1.05, 24 + fl[i] * 0.55, 3.3, 3.0), skin, R, o);
      for (let i = 0; i < 4; i++) piece(ctx, ell(fx[i] * 1.05, 24 + fl[i] * 0.55, 2.6, 1.6, 0, 10), skinD, R, { amp: 0, sh: 0, edge: false });
      piece(ctx, capsule(11, 6, 15, 22, 3.8, 3.2), skin, R, o);
    } else if (mode === 'fist') { // cold: fingers folded in, knuckles up
      piece(ctx, ell(0, 14, 12, 12, 0, 24), skin, R, o);
      for (let i = 0; i < 4; i++) piece(ctx, ell(fx[i] * 1.02, 26, 3.6, 3.2, 0, 14), skin, R, o);
      line(ctx, [[-11, 23], [11, 23]], skinD, 0.9);
      piece(ctx, capsule(10, 10, -1, 21, 3.6, 3.2), skin, R, o);
      for (let i = 0; i < 4; i++) line(ctx, [[fx[i] - 1.2, 25], [fx[i] + 1.2, 25]], 'rgba(120,70,50,0.6)', 0.7);
    } else if (mode === 'flat') { // palm down, fingers out
      piece(ctx, ell(0, 13, 11, 13, 0, 26), skin, R, o);
      for (let i = 0; i < 4; i++) piece(ctx, capsule(fx[i], 22, fx[i] * 1.25, 22 + fl[i], 3.3, 2.8), skin, R, o);
      piece(ctx, capsule(10, 8, 18, 22, 3.8, 3.1), skin, R, o);
      for (let i = 0; i < 4; i++) line(ctx, [[fx[i] * 1.12 - 1.4, 22 + fl[i] * 0.55], [fx[i] * 1.12 + 1.4, 22 + fl[i] * 0.55]], 'rgba(120,70,50,0.45)', 0.6);
    } else if (mode === 'lift') { // withdrawn: wrist bent back, fingertips raised
      piece(ctx, ell(0, 12, 11, 12, 0, 26), skin, R, o);
      for (let i = 0; i < 4; i++) piece(ctx, capsule(fx[i], 20, fx[i] * 1.35 + 2, 20 + fl[i] * 0.8, 3.2, 2.7), skin, R, o);
      piece(ctx, capsule(10, 6, 19, 17, 3.6, 3.0), skin, R, o);
    } else if (mode === 'wrap') { // holding: fingers curve over an object
      piece(ctx, ell(0, 12, 12, 12, 0, 24), skin, R, o);
      for (let i = 0; i < 4; i++) {
        const p0 = [fx[i], 20], p1 = [fx[i] * 0.9 + 3, 32], p2 = [fx[i] * 0.7 + 8, 40];
        piece(ctx, capsule(p0[0], p0[1], p1[0], p1[1], 3.4, 3.2), skin, R, o);
        piece(ctx, capsule(p1[0], p1[1], p2[0], p2[1], 3.2, 2.8), skin, R, o);
        line(ctx, [[p1[0] - 2.4, p1[1] - 0.6], [p1[0] + 2.4, p1[1] + 0.6]], 'rgba(120,70,50,0.55)', 0.6);
      }
      piece(ctx, capsule(11, 6, 6, 24, 3.8, 3.2), skin, R, o);
    }
    ctx.restore();
  }

  /* ---------- the elder (seated, right of room) ---------- */
  function chair(ctx, R) {
    piece(ctx, [[808, 170], [992, 170], [1004, 560], [796, 560]], C.chair, R, { amp: 1.6, sh: 8 });
    piece(ctx, [[818, 186], [982, 186], [990, 540], [810, 540]], '#3c2818', R, { amp: 1.2, sh: 0 });
    for (const x of [802, 990]) piece(ctx, [[x, 540], [x + 12, 540], [x + 14, 700], [x + 2, 700]], C.chair, R, { amp: 1, sh: 5 });
  }
  function elder(ctx, st, R) {

    piece(ctx, [[812, 470], [990, 470], [1040, 560], [1046, 700], [800, 700], [790, 560]], C.eDress, R, { amp: 2, sh: 6, grad: [0, 470, 0, 700, '#2c2426'] });
    line(ctx, [[860, 560], [850, 700]], 'rgba(20,14,14,0.5)', 2); line(ctx, [[950, 560], [962, 700]], 'rgba(20,14,14,0.5)', 2);
    // torso
    piece(ctx, [[846, 300 + lift], [954, 300 + lift], [978, 480], [822, 480]], C.eDress, R, { amp: 1.5, sh: 4 });
    // shawl
    const shawl = cold
      ? [[826, 286 + lift], [974, 286 + lift], [1000, 360], [980, 450], [900, 470], [820, 450], [800, 360]]

    piece(ctx, [[886, 262 + lift], [914, 262 + lift], [916, 300 + lift], [884, 300 + lift]], C.eSkinD, R, { amp: 0.6, sh: 2 });
    const hy = 232 + lift * 1.4, hx = 896;
    const head = [[hx + 26, hy - 30], [hx + 30, hy], [hx + 24, hy + 28], [hx + 4, hy + 38], [hx - 12, hy + 34], [hx - 18, hy + 22], [hx - 16, hy + 14], [hx - 24, hy + 6], [hx - 18, hy + 1], [hx - 20, hy - 10], [hx - 12, hy - 30], [hx + 6, hy - 38]];

    }
  }

  /* ---------- the child ---------- */
  function child(ctx, st, R) {
    if (st.child === 'window' || st.child === 'mirror') {

  /* dusk: cool multiply over the whole frame, keeps paper but drains warmth */
  function dusk(ctx, w, h) {
    ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0);
    ctx.globalCompositeOperation = 'multiply';
    ctx.fillStyle = '#8692ab'; ctx.fillRect(0, 0, w, h);
    ctx.restore();
  }
  function finish(ctx, w, h, seed, warm) {
    ctx.save(); ctx.setTransform(1, 0, 0, 1, 0, 0);
    // paper grain
    const r = rng(seed), n = Math.round(w * h / 300);
    for (let i = 0; i < n; i++) {
      const v = r();
      ctx.fillStyle = v < 0.5 ? 'rgba(255,245,225,0.05)' : 'rgba(20,10,4,0.06)';
      ctx.fillRect(r() * w, r() * h, 1 + r() * 2.2, 1 + r() * 1.2);
    }
    // vignette
    const g = ctx.createRadialGradient(w / 2, h / 2, h * 0.3, w / 2, h / 2, w * 0.72);
    g.addColorStop(0, 'rgba(0,0,0,0)'); g.addColorStop(1, warm ? 'rgba(30,14,4,0.55)' : 'rgba(8,10,20,0.6)');
    ctx.fillStyle = g; ctx.fillRect(0, 0, w, h);
    ctx.restore();
  }

  function room(ctx, w, h, st, cam, seed) {
    ctx.save();
    ctx.fillStyle = '#1b120b'; ctx.fillRect(0, 0, w, h);
    const R = rng(seed);
    setCam(ctx, w, h, cam, 0.9); backPlane(ctx, st, rng(11));

    if (st.patch) { setCam(ctx, w, h, cam, 1.0); reflection(ctx); }
    ctx.restore();
    if (st.sun < 0.5) dusk(ctx, w, h);
    finish(ctx, w, h, seed + 5, st.sun > 0.5);
  }
