// AFTERLIGHT · F6 · The Sun at the Door
// An adaptation of the Ama-no-Iwato episode (Kojiki, 712; Nihon Shoki, 720), read through
// Chamberlain's 1882 and Aston's 1896 English translations. Not an account of Japanese belief at large.
(function () {
  const W = 1280, H = 720;

  // ---------- deterministic helpers ----------
  function rng(seed) {
    return function () {
      seed |= 0; seed = seed + 0x6D2B79F5 | 0;
      let t = Math.imul(seed ^ seed >>> 15, 1 | seed);
      t = t + Math.imul(t ^ t >>> 7, 61 | t) ^ t;
      return ((t ^ t >>> 14) >>> 0) / 4294967296;
    };
  }
  const C = {
    night: '#121830', night2: '#1b2340', ink: '#0a0d17',
    rock: '#3a4358', rock2: '#2b3245', rockHi: '#58627a',
    inside: '#efd38a', inside2: '#d9ad55', goldPale: '#f8e7b0', gold: '#eabf5c',
    verm: '#c4412a', fire: '#e8782f', fire2: '#f6c34c',
    white: '#f4efe3', cream: '#ece2cb', mirror: '#dfe4e1', bronze: '#9a7a45',
    hair: '#16131a', straw: '#c9a75a', straw2: '#8e7034',
    leaf: '#2f5a3d', leaf2: '#447652', ground: '#262631',
    dayGround: '#cdb68a', dayGround2: '#b99f72', daySky: '#f2e2ab',
    uzume: '#a9c39a', moss: '#4c7a3a', tajik: '#b7772f', futo: '#5d6f8f', koya: '#6b4a6e',
    blue: '#5a7dab'
  };
  const SK = { night: '#6e5747', fire: '#d99a6c', day: '#e6c3a0', inside: '#efcfa8' };

  function jit(R, pts, amp) {
    const out = [];
    const n = pts.length;
    for (let i = 0; i < n; i++) {
      const a = pts[i], b = pts[(i + 1) % n];
      const L = Math.hypot(b[0] - a[0], b[1] - a[1]) || 1;
      const k = Math.max(1, Math.floor(L / 16));
      const nx = -(b[1] - a[1]) / L, ny = (b[0] - a[0]) / L;
      for (let j = 0; j < k; j++) {
        const t = j / k, d = (R() - 0.5) * amp;
        out.push([a[0] + (b[0] - a[0]) * t + nx * d, a[1] + (b[1] - a[1]) * t + ny * d]);
      }
    }
    return out;
  }
  function trace(ctx, pts) {
    ctx.beginPath();
    pts.forEach((p, i) => i ? ctx.lineTo(p[0], p[1]) : ctx.moveTo(p[0], p[1]));
    ctx.closePath();
  }
  // one sheet of cut paper with a drop shadow
  function paper(ctx, R, pts, col, amp, sh) {
    amp = amp == null ? 3 : amp; sh = sh == null ? 0.32 : sh;
    const p = amp ? jit(R, pts, amp) : pts;
    if (sh > 0) {
      ctx.save(); ctx.translate(4, 5); ctx.fillStyle = 'rgba(0,0,0,' + sh + ')'; trace(ctx, p); ctx.fill(); ctx.restore();
    }
    ctx.fillStyle = col; trace(ctx, p); ctx.fill();
  }
  function ell(cx, cy, rx, ry, n, rot) {
    n = n || 28; rot = rot || 0;
    const pts = [];
    for (let i = 0; i < n; i++) {
      const a = i / n * Math.PI * 2;
      const x = Math.cos(a) * rx, y = Math.sin(a) * ry;
      pts.push([cx + x * Math.cos(rot) - y * Math.sin(rot), cy + x * Math.sin(rot) + y * Math.cos(rot)]);
    }
    return pts;
  }
  function plane(ctx, col, pts) { ctx.fillStyle = col; trace(ctx, pts); ctx.fill(); }

  // ---------- cast ----------
  // o: x, y (feet), h, robe, under, skin, hair:'long'|'mizura', la, ra (arm angle from straight down, radians; + = outward),
  //    back (seen from behind), mouth:'open'|'line', tilt (head offset x), seated, sash, headdress, lean
  function figure(ctx, R, o) {
    const h = o.h, x = o.x, y = o.y;
    const hr = h * 0.075;
    const lean = o.lean || 0;
    const top = y - h;
    const hx = x + (o.tilt || 0) + lean, hy = top + hr;
    const sy = top + hr * 2.35;           // shoulder line
    const sw = h * 0.22, hw = o.seated ? h * 0.5 : h * 0.34;
    const skin = o.skin || SK.day;
    // long hair behind
    if (o.hair === 'long') paper(ctx, R, [[hx - hr * 1.05, hy - hr * 0.6], [hx + hr * 1.05, hy - hr * 0.6], [x + lean * 0.5 + hr * 1.3, sy + h * (o.back ? 0.38 : 0.3)], [x + lean * 0.5 - hr * 1.3, sy + h * (o.back ? 0.38 : 0.3)]], C.hair, 2, 0.2);
    // robe body
    const robe = o.seated
      ? [[x + lean - sw / 2, sy], [x + lean + sw / 2, sy], [x + hw * 0.55, y - h * 0.12], [x + hw / 2, y], [x - hw / 2, y], [x - hw * 0.55, y - h * 0.12]]
      : [[x + lean - sw / 2, sy], [x + lean + sw / 2, sy], [x + hw / 2, y], [x - hw / 2, y]];
    paper(ctx, R, robe, o.robe, 3);
    // under-robe hem + collar
    if (o.under) {
      if (!o.seated) paper(ctx, R, [[x - hw / 2 + 4, y - h * 0.06], [x + hw / 2 - 4, y - h * 0.06], [x + hw / 2, y], [x - hw / 2, y]], o.under, 1, 0);
      if (!o.back) paper(ctx, R, [[hx - sw * 0.28, sy - 2], [hx + sw * 0.28, sy - 2], [x + lean * 0.7, sy + h * 0.16]], o.under, 1, 0);
    }
    if (o.sash) paper(ctx, R, [[x + lean * 0.6 - sw * 0.62, sy + h * 0.27], [x + lean * 0.6 + sw * 0.62, sy + h * 0.27], [x + lean * 0.6 + sw * 0.64, sy + h * 0.32], [x + lean * 0.6 - sw * 0.64, sy + h * 0.32]], o.sash, 2, 0.15);
    // arms: wide sleeve + hand
    const hands = [];
    [[-1, o.la == null ? 0.25 : o.la], [1, o.ra == null ? 0.25 : o.ra]].forEach(([side, a]) => {
      const S = [x + lean + side * sw * 0.44, sy + h * 0.02];
      const L = h * 0.33;
      const dx = Math.sin(a) * side, dy = Math.cos(a);
      const E = [S[0] + dx * L, S[1] + dy * L];
      const px = -dy, py = dx;
      const w0 = h * 0.045, w1 = h * 0.075;
      paper(ctx, R, [[S[0] + px * w0, S[1] + py * w0], [E[0] - dx * h * 0.03 + px * w1, E[1] - dy * h * 0.03 + py * w1], [E[0] - dx * h * 0.03 - px * w1, E[1] - dy * h * 0.03 - py * w1], [S[0] - px * w0, S[1] - py * w0]], o.robe, 2, 0.25);
      const hp = [E[0] + dx * h * 0.012, E[1] + dy * h * 0.012];
      if (!o.hideHands) paper(ctx, R, ell(hp[0], hp[1], h * 0.026, h * 0.03, 12, a * side), skin, 1, 0.15);
      hands.push(hp);
    });
    // neck + head
    paper(ctx, R, [[hx - hr * 0.35, hy + hr * 0.6], [hx + hr * 0.35, hy + hr * 0.6], [hx + hr * 0.4, sy + 2], [hx - hr * 0.4, sy + 2]], o.back ? C.hair : skin, 0, 0);
    paper(ctx, R, ell(hx, hy, hr * 0.86, hr, 20), o.back ? C.hair : skin, 1.5, 0.25);
    // hair cap / styles
    if (!o.back) {
      ctx.fillStyle = C.hair;
      ctx.beginPath(); ctx.ellipse(hx, hy - hr * 0.35, hr * 0.92, hr * 0.7, 0, Math.PI, 0); ctx.fill();
      if (o.hair === 'mizura') {
        [-1, 1].forEach(s => { paper(ctx, R, ell(hx + s * hr * 1.05, hy + hr * 0.55, hr * 0.3, hr * 0.55, 12), C.hair, 1, 0); });
        paper(ctx, R, ell(hx, hy - hr * 1.0, hr * 0.32, hr * 0.26, 10), C.hair, 0, 0);
      }
      // face
      ctx.fillStyle = 'rgba(20,14,12,0.85)';
      const ey = hy + hr * 0.05;
      if (o.mouth === 'open') {
        ctx.lineWidth = Math.max(1.5, hr * 0.12); ctx.strokeStyle = 'rgba(20,14,12,0.85)';
        [-1, 1].forEach(s => { ctx.beginPath(); ctx.arc(hx + s * hr * 0.33, ey + hr * 0.04, hr * 0.14, Math.PI * 1.1, Math.PI * 1.9); ctx.stroke(); });
        ctx.beginPath(); ctx.ellipse(hx, hy + hr * 0.5, hr * 0.24, hr * 0.17, 0, 0, Math.PI * 2); ctx.fill();
      } else {
        [-1, 1].forEach(s => { ctx.fillRect(hx + s * hr * 0.33 - hr * 0.13, ey, hr * 0.26, Math.max(1.5, hr * 0.08)); });
        if (o.mouth === 'line') ctx.fillRect(hx - hr * 0.12, hy + hr * 0.5, hr * 0.24, Math.max(1, hr * 0.06));
      }
    } else if (o.hair === 'mizura') {
      [-1, 1].forEach(s => { paper(ctx, R, ell(hx + s * hr * 1.0, hy + hr * 0.5, hr * 0.3, hr * 0.55, 12), C.hair, 1, 0); });
    }
    if (o.headdress) {   // spindle-tree sprig: leaves + berries
      for (let i = 0; i < 7; i++) {
        const a = Math.PI * (1.05 + i * 0.15);
        paper(ctx, R, ell(hx + Math.cos(a) * hr * 1.05, hy + Math.sin(a) * hr * 1.0, hr * 0.32, hr * 0.14, 8, a), C.leaf2, 0, 0);
        if (i % 2) paper(ctx, R, ell(hx + Math.cos(a) * hr * 1.25, hy + Math.sin(a) * hr * 1.2, hr * 0.1, hr * 0.1, 8), C.verm, 0, 0);
      }
    }
    return { hands, head: [hx, hy], hr };
  }

  function fire(ctx, R, x, y, s) {
    paper(ctx, R, [[x - 30 * s, y], [x - 18 * s, y - 60 * s], [x - 4 * s, y - 34 * s], [x + 6 * s, y - 92 * s], [x + 18 * s, y - 40 * s], [x + 28 * s, y - 64 * s], [x + 34 * s, y]], C.verm, 4 * s, 0);
    paper(ctx, R, [[x - 18 * s, y], [x - 8 * s, y - 40 * s], [x + 4 * s, y - 62 * s], [x + 14 * s, y - 34 * s], [x + 22 * s, y]], C.fire, 3 * s, 0);
    paper(ctx, R, [[x - 8 * s, y], [x + 2 * s, y - 32 * s], [x + 12 * s, y]], C.fire2, 2 * s, 0);
    paper(ctx, R, [[x - 40 * s, y - 4 * s], [x + 44 * s, y - 10 * s], [x + 46 * s, y + 4 * s], [x - 42 * s, y + 8 * s]], '#3a2a20', 2, 0.2);
  }
  function glow(ctx, x, y, r, col, a) {
    const g = ctx.createRadialGradient(x, y, 0, x, y, r);
    g.addColorStop(0, col.replace('A', a)); g.addColorStop(1, col.replace('A', 0));
    ctx.fillStyle = g; ctx.fillRect(x - r, y - r, r * 2, r * 2);
  }
  function rooster(ctx, R, x, y, s, flip) {
    ctx.save(); ctx.translate(x, y); ctx.scale(flip ? -s : s, s);
    paper(ctx, R, [[-10, 0], [-30, -40], [-48, -30], [-36, -6]], '#2c3a2e', 2, 0.2);   // tail
    paper(ctx, R, [[-14, -2], [-34, -52], [-20, -30]], '#1f2a24', 1, 0);
    paper(ctx, R, ell(0, -12, 20, 14, 14), '#6e3a22', 2, 0.25);                       // body
    paper(ctx, R, ell(16, -30, 8, 9, 10), '#7c4426', 1, 0);                             // head
    paper(ctx, R, [[12, -38], [16, -46], [19, -39], [22, -45], [24, -36]], C.verm, 1, 0); // comb
    paper(ctx, R, [[23, -31], [32, -28], [23, -26]], C.gold, 0, 0);                       // beak
    ctx.fillStyle = '#c9a75a'; ctx.fillRect(-4, 0, 2.5, 14); ctx.fillRect(4, 0, 2.5, 14);
    ctx.restore();
  }
  function perch(ctx, R, x1, x2, y, col) {
    paper(ctx, R, [[x1, y], [x2, y], [x2, y + 7], [x1, y + 7]], col, 1, 0.2);
    paper(ctx, R, [[x1 + 10, y], [x1 + 18, y], [x1 + 18, y + 120], [x1 + 10, y + 120]], col, 1, 0.2);
    paper(ctx, R, [[x2 - 18, y], [x2 - 10, y], [x2 - 10, y + 120], [x2 - 18, y + 120]], col, 1, 0.2);
  }
  // overturned tub (Aston: "a tub bottom upwards"; Kojiki: a sounding-board / ukefune)
  function tub(ctx, R, x, y, w, h, lit) {
    const body = lit || '#7a5532';
    paper(ctx, R, [[x - w / 2, y - h], [x + w / 2, y - h], [x + w / 2 + 8, y], [x - w / 2 - 8, y]], body, 2);
    ctx.fillStyle = 'rgba(0,0,0,0.18)';
    for (let i = 1; i < 7; i++) { const xx = x - w / 2 + i * w / 7; ctx.fillRect(xx, y - h + 4, 2, h - 6); }
    ['#3b2a1a', '#3b2a1a'].forEach((c, i) => paper(ctx, R, [[x - w / 2 - 2 - i * 4, y - h * (0.25 + i * 0.5)], [x + w / 2 + 2 + i * 4, y - h * (0.25 + i * 0.5)], [x + w / 2 + 3 + i * 4, y - h * (0.25 + i * 0.5) + 8], [x - w / 2 - 3 - i * 4, y - h * (0.25 + i * 0.5) + 8]], c, 1, 0));
    paper(ctx, R, ell(x, y - h, w / 2, 9, 20), '#93693f', 1, 0);
  }
  // twisted straw rope with root-ends (Kojiki: "bottom-tied rope")
  function rope(ctx, R, x1, y1, x2, y2, sag, tufts) {
    const n = 26;
    const P = t => [x1 + (x2 - x1) * t, y1 + (y2 - y1) * t + Math.sin(Math.PI * t) * sag];
    ctx.lineCap = 'round';
    ctx.strokeStyle = 'rgba(0,0,0,0.35)'; ctx.lineWidth = 16; ctx.beginPath();
    for (let i = 0; i <= n; i++) { const p = P(i / n); i ? ctx.lineTo(p[0] + 4, p[1] + 5) : ctx.moveTo(p[0] + 4, p[1] + 5); } ctx.stroke();
    ctx.strokeStyle = C.straw; ctx.lineWidth = 14; ctx.beginPath();
    for (let i = 0; i <= n; i++) { const p = P(i / n); i ? ctx.lineTo(p[0], p[1]) : ctx.moveTo(p[0], p[1]); } ctx.stroke();
    ctx.strokeStyle = C.straw2; ctx.lineWidth = 3;
    for (let i = 0; i < n * 2; i++) { const a = P(i / (n * 2)), b = P((i + 0.6) / (n * 2)); ctx.beginPath(); ctx.moveTo(a[0] - 4, a[1] - 6); ctx.lineTo(b[0] + 4, b[1] + 6); ctx.stroke(); }
    for (let k = 0; k < (tufts || 0); k++) {
      const p = P((k + 0.5) / tufts);
      paper(ctx, R, [[p[0] - 5, p[1] + 4], [p[0] + 5, p[1] + 4], [p[0] + 7 + (R() - .5) * 8, p[1] + 34 + R() * 10], [p[0] - 7, p[1] + 30 + R() * 10]], C.straw, 2, 0.15);
    }
    // the untrimmed root-end ("bottom-tied")
    const e = P(1);
    for (let i = 0; i < 6; i++) paper(ctx, R, [[e[0], e[1] - 4], [e[0] + 2, e[1] + 4], [e[0] + 18 + i * 3, e[1] - 14 + i * 7]], C.straw2, 1, 0);
  }
  // uprooted sakaki with jewels (upper), mirror (middle), offerings (lower)
  function sakaki(ctx, R, x, yb, ht, mirrorFace) {
    const yt = yb - ht;
    // roots
    for (let i = 0; i < 6; i++) paper(ctx, R, [[x - 4, yb - 6], [x + 4, yb - 6], [x + (i - 2.5) * 16, yb + 28 + (i % 2) * 10]], '#4a3826', 1, 0.15);
    paper(ctx, R, [[x - 7, yb], [x + 7, yb], [x + 4, yt + 20], [x - 4, yt + 20]], '#4a3826', 2);
    const tiers = [[yt + ht * 0.12, ht * 0.30], [yt + ht * 0.40, ht * 0.38], [yt + ht * 0.66, ht * 0.32]];
    tiers.forEach(([ty, span], ti) => {
      [-1, 1].forEach(s => {
        paper(ctx, R, [[x, ty + 6], [x + s * span, ty - 14], [x + s * span, ty - 8], [x, ty + 12]], '#4a3826', 1, 0.1);
        for (let i = 0; i < 9; i++) {
          const t = 0.25 + i / 11;
          paper(ctx, R, ell(x + s * span * t, ty - 4 - 14 * t + (i % 2 ? -9 : 9), 15, 6, 10, s * (0.4 + (i % 3) * 0.3)), i % 3 ? C.leaf : C.leaf2, 1, 0.12);
        }
      });
    });
    // upper: string of curved jewels (magatama)
    const jy = tiers[0][0];
    for (let i = 0; i < 9; i++) {
      const t = i / 8, jx = x - ht * 0.24 + t * ht * 0.48, jyy = jy + 6 + Math.sin(Math.PI * t) * 34;
      paper(ctx, R, ell(jx, jyy, 7, 9, 10), i % 2 ? '#3f8a77' : '#5aa58c', 0, 0.12);
      ctx.fillStyle = '#2b5b4f'; ctx.beginPath(); ctx.arc(jx - 2, jyy - 3, 2, 0, 7); ctx.fill();
    }
    // middle: the mirror
    const my = tiers[1][0] + 26, mr = ht * 0.085;
    paper(ctx, R, ell(x, my, mr + 7, mr + 7, 30), C.bronze, 0, 0.35);
    paper(ctx, R, ell(x, my, mr, mr, 30), C.mirror, 0, 0);
    if (mirrorFace) mirrorFace(x, my, mr);
    else { ctx.fillStyle = 'rgba(255,255,255,0.45)'; ctx.beginPath(); ctx.ellipse(x - mr * 0.35, my - mr * 0.35, mr * 0.35, mr * 0.18, -0.7, 0, 7); ctx.fill(); }
    ctx.strokeStyle = '#6d4a2a'; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(x, my - mr - 7); ctx.lineTo(x, my - mr - 26); ctx.stroke();
    // lower: white and blue cloth offerings
    const oy = tiers[2][0];
    for (let i = 0; i < 6; i++) {
      const ox = x - ht * 0.2 + i * ht * 0.08;
      paper(ctx, R, [[ox - 7, oy], [ox + 7, oy], [ox + 9, oy + 60 + (i % 3) * 14], [ox - 5, oy + 64 + (i % 3) * 14]], i % 2 ? C.blue : C.white, 2, 0.2);
    }
    return { mirror: [x, my, mr] };
  }
  function rockMass(ctx, R, pts, col, amp) { paper(ctx, R, pts, col, amp || 10, 0.4); }
  function shadow(ctx, x, y, len, ang, w) {
    ctx.fillStyle = 'rgba(70,45,25,0.28)';
    ctx.beginPath(); ctx.moveTo(x - w / 2, y); ctx.lineTo(x + w / 2, y);
    ctx.lineTo(x + w / 2 + Math.cos(ang) * len, y + Math.sin(ang) * len * 0.25); ctx.lineTo(x - w / 2 + Math.cos(ang) * len, y + Math.sin(ang) * len * 0.25);
    ctx.closePath(); ctx.fill();
  }
  function frameOn(ctx, w, h, fn) { ctx.save(); ctx.scale(w / W, h / H); fn(ctx); ctx.restore(); }
  function grain(ctx, R, n, col) {
    ctx.fillStyle = col;
    for (let i = 0; i < n; i++) ctx.fillRect(R() * W, R() * H, 1 + R() * 2, 1 + R() * 2);
  }
