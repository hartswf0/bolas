// AFTERLIGHT · Film 2 · The Picture You Could Feel
// Keyframe module. Plain browser JS, Canvas 2D only, deterministic.
// Medium: embossed cut-paper relief on a tabletop under one lamp.
(function () {
  const TAU = Math.PI * 2;
  const VW = 1280, VH = 720;

  // ---------- utilities ----------
  function rng(seed) {
    let a = seed >>> 0;
    return function () {
      a = (a + 0x6D2B79F5) >>> 0;
      let t = a;
      t = Math.imul(t ^ (t >>> 15), t | 1);
      t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }
  function cv(w, h) { const c = document.createElement('canvas'); c.width = w; c.height = h; return c; }
  function applyM(ctx, m) { ctx.transform(m.a, m.b, m.c, m.d, m.e, m.f); }
  const hyp = Math.hypot;

  // consistent cast: A (ochre sleeve, darker skin) and B (slate sleeve, lighter skin)
  const CAST = {
    A: { skin: '#8d5b3e', hi: '#b98762', sleeve: '#b07a30', cuff: '#8a5c20', hair: '#1c1512' },
    B: { skin: '#dcae8f', hi: '#f1d0b8', sleeve: '#4c6176', cuff: '#3a4b5c', hair: '#5b3f2c' }
  };

  // ---------- the picture (sheet units, u across, v down) ----------
  const HILL = [[0.06, 0.80], [0.18, 0.70], [0.30, 0.64], [0.42, 0.66], [0.55, 0.68], [0.68, 0.585], [0.80, 0.60], [0.94, 0.72]];
  const TRUNK = [[0.30, 0.645], [0.305, 0.52], [0.31, 0.38]];
  const BR = [
    [[0.307, 0.50], [0.25, 0.44], [0.19, 0.40]],
    [[0.309, 0.45], [0.36, 0.39], [0.43, 0.34]],
    [[0.31, 0.40], [0.28, 0.32], [0.25, 0.26]],
    [[0.36, 0.39], [0.40, 0.42], [0.44, 0.43]]
  ];
  const SUN = [0.72, 0.28, 0.075];
  function ringPts(cx, cy, r, n) { const p = []; for (let i = 0; i <= n; i++) { const a = i / n * TAU; p.push([cx + Math.cos(a) * r, cy + Math.sin(a) * r * 1.3]); } return p; }
  function PIC(ver) {
    if (ver === 'v1') {
      return {
        lines: [{ w: 0.016, p: HILL }, { w: 0.013, p: TRUNK }].concat(BR.map(p => ({ w: 0.0055, p }))),
        discs: [SUN]
      };
    }
    // v2 (corrected) and v2p (corrected, sun stipple half done)
    const lines = [{ w: 0.018, p: HILL }, { w: 0.021, p: TRUNK }].concat(BR.map(p => ({ w: 0.017, p })));
    lines.push({ w: 0.016, p: ringPts(SUN[0], SUN[1], SUN[2], 48) });
    const discs = [];
    for (let du = -0.06; du <= 0.0601; du += 0.024) for (let dv = -0.06; dv <= 0.0601; dv += 0.024) {
      if (hyp(du, dv) > 0.052) continue;
      if (ver === 'v2p' && dv > 0.001) continue;
      discs.push([SUN[0] + du, SUN[1] + dv * 1.3, 0.0068]);
    }
    return { lines, discs };
  }
  function tracePts(x, pts, W, H) {
    x.moveTo(pts[0][0] * W, pts[0][1] * H);
    for (let i = 1; i < pts.length - 1; i++) {
      const mx = (pts[i][0] + pts[i + 1][0]) / 2 * W, my = (pts[i][1] + pts[i + 1][1]) / 2 * H;
      x.quadraticCurveTo(pts[i][0] * W, pts[i][1] * H, mx, my);
    }
    const l = pts[pts.length - 1]; x.lineTo(l[0] * W, l[1] * H);
  }
  function drawPic(x, ver, W, H, mirror) {
    x.save();
    if (mirror) { x.translate(W, 0); x.scale(-1, 1); }
    x.lineCap = 'round'; x.lineJoin = 'round';
    const P = PIC(ver);
    for (const s of P.lines) { x.lineWidth = s.w * W; x.beginPath(); tracePts(x, s.p, W, H); x.stroke(); }
    for (const d of P.discs) { x.beginPath(); x.arc(d[0] * W, d[1] * H, d[2] * W, 0, TAU); x.fill(); }
    x.restore();
  }
  function maskOf(ver, W, H, mirror, blur) {
    const c = cv(W, H), x = c.getContext('2d');
    x.fillStyle = '#000'; x.fillRect(0, 0, W, H);
    if (blur) x.filter = 'blur(' + blur + 'px)';
    x.fillStyle = x.strokeStyle = '#fff';
    drawPic(x, ver, W, H, mirror);
    const d = x.getImageData(0, 0, W, H).data, m = new Float32Array(W * H);
    for (let i = 0; i < m.length; i++) m[i] = d[i * 4] / 255;
    return m;
  }

  // ---------- the sheet: heat-diffusion height field + raking-light shading ----------
  // Height = narrow-diffusion term (thin lines lose heat and stay shallow)
  //        + wide-diffusion term (large dark areas keep heat and over-swell).
  function makeSheet(o) {
    const W = o.W || 520, H = Math.round(W / 1.3), N = W * H;
    const mirror = o.side === 'back';
    const ink = maskOf(o.ver, W, H, mirror, 0);
    const h = new Float32Array(N);
    if (o.relief > 0) {
      const b1 = maskOf(o.ver, W, H, mirror, 4), b2 = maskOf(o.ver, W, H, mirror, 14);
      for (let i = 0; i < N; i++) {
        const I1 = b1[i], I2 = b2[i];
        h[i] = o.relief * (26 * Math.max(0, I1 - 0.28) + 30 * I2 * I2);
      }
    }
    const r = rng(o.seed || 7);
    const pasteEdge = o.side === 'back' ? (o.pasteEdge == null ? 0 : o.pasteEdge) : 2;
    // paper fibre / paste grain, plus knife streaks in fresh paste
    const streak = new Float32Array(H);
    for (let y = 0; y < H; y++) streak[y] = Math.sin(y * 0.21 + r() * 0.6) * 0.6 + Math.sin(y * 0.047) * 0.8;
    for (let y = 0; y < H; y++) for (let x = 0; x < W; x++) {
      const i = y * W + x;
      const edge = pasteEdge * W + Math.sin(y * 0.09) * 9 + Math.sin(y * 0.31) * 4;
      const onPaste = x > edge;
      h[i] += (r() - 0.5) * (onPaste ? 0.9 : 0.25);
      if (onPaste && o.wetPaste) h[i] += streak[y] + (x - edge < 10 ? (10 - (x - edge)) * 0.25 : 0);
    }
    const c = cv(W, H), x = c.getContext('2d');
    const img = x.createImageData(W, H), D = img.data;
    const L = o.light, amb = o.amb, kd = o.kd, tint = o.tint || [1, 1, 1];
    const paper = [240, 234, 221], paste = [233, 229, 219], inkC = [26, 24, 27];
    for (let y = 0; y < H; y++) for (let xx = 0; xx < W; xx++) {
      const i = y * W + xx;
      let v = 1;
      if (xx > 0 && yy(y) && xx < W - 1) {
        const hx = (h[i + 1] - h[i - 1]) * 0.5, hy = (h[i + W] - h[i - W]) * 0.5;
        const nl = hyp(hx, hy, 1);
        const lx = L.x - xx, ly = L.y - y, lz = L.z - h[i], ll = hyp(lx, ly, lz);
        const dif = Math.max(0, (-hx * lx - hy * ly + lz) / (nl * ll));
        const flat = Math.max(0.02, L.z / hyp(L.x - xx, L.y - y, L.z));
        let sh = 1;
        if (o.relief > 0) {
          const dxy = hyp(lx, ly) || 1, ux = lx / dxy, uy = ly / dxy, sl = lz / dxy;
          for (let s = 2; s <= 30; s += 2) {
            const px = Math.round(xx + ux * s), py = Math.round(y + uy * s);
            if (px < 0 || py < 0 || px >= W || py >= H) break;
            if (h[py * W + px] > h[i] + sl * s + 0.4) { sh = 0.42; break; }
          }
        }
        v = (amb + kd * dif * sh) / (amb + kd * flat);
      }
      const edge = pasteEdge * W + Math.sin(y * 0.09) * 9 + Math.sin(y * 0.31) * 4;
      const base = xx > edge ? paste : paper;
      let cr = base[0], cg = base[1], cb = base[2];
      if (!mirror) { const k = ink[i]; cr += (inkC[0] - cr) * k; cg += (inkC[1] - cg) * k; cb += (inkC[2] - cb) * k; }
      else { const k = ink[i] * (xx > edge ? 0.05 : 0.14); cr *= 1 - k; cg *= 1 - k; cb *= 1 - k * 0.8; }
      const j = i * 4;
      D[j] = Math.min(255, cr * v * tint[0]);
      D[j + 1] = Math.min(255, cg * v * tint[1]);
      D[j + 2] = Math.min(255, cb * v * tint[2]);
      D[j + 3] = 255;
    }
    function yy(y) { return y > 0 && y < H - 1; }
    x.putImageData(img, 0, 0);
    return c;
  }

  // sheet placement on the table: centre, width, rotation (deg), foreshortening ys
  function placeM(pl) { return new DOMMatrix().translate(pl.cx, pl.cy).scale(1, pl.ys).rotate(pl.rot); }
  function sp(pl, u, v) { const m = placeM(pl), hh = pl.w / 1.3; const p = m.transformPoint(new DOMPoint((u - 0.5) * pl.w, (v - 0.5) * hh)); return [p.x, p.y]; }
  function picPt(pl, u, v, back) { return sp(pl, back ? 1 - u : u, v); }
  function lightInSheet(pl, sx, sy, z, W) {
    const inv = placeM(pl).inverse(), p = inv.transformPoint(new DOMPoint(sx, sy)), hh = pl.w / 1.3, H = Math.round(W / 1.3);
    return { x: (p.x / pl.w + 0.5) * W, y: (p.y / hh + 0.5) * H, z };
  }
  function drawSheet(ctx, canvas, pl, L, shadowA) {
    const hh = pl.w / 1.3;
    const c = [pl.cx, pl.cy], dx = c[0] - L[0], dy = c[1] - L[1], d = hyp(dx, dy) || 1;
    ctx.save();
    ctx.translate(dx / d * 9, dy / d * 9);
    applyM(ctx, placeM(pl));
    ctx.filter = 'blur(6px)';
    ctx.fillStyle = 'rgba(12,8,6,' + (shadowA == null ? 0.45 : shadowA) + ')';
    ctx.fillRect(-pl.w / 2, -hh / 2, pl.w, hh);
    ctx.restore();
    ctx.save();
    applyM(ctx, placeM(pl));
    ctx.drawImage(canvas, -pl.w / 2, -hh / 2, pl.w, hh);
    ctx.strokeStyle = 'rgba(255,250,240,0.35)'; ctx.lineWidth = 1.2;
    ctx.strokeRect(-pl.w / 2, -hh / 2, pl.w, hh);
    ctx.restore();
  }

  // ---------- embossed paper shapes ----------
  function paperShape(ctx, path, color, c, L, depth, size, opt) {
    opt = opt || {};
    const dx = c[0] - L[0], dy = c[1] - L[1], d = hyp(dx, dy) || 1, ux = dx / d, uy = dy / d;
    if (!opt.noShadow) {
      ctx.save(); ctx.translate(ux * depth, uy * depth);
      ctx.filter = 'blur(' + Math.max(1, depth * 0.7) + 'px)';
      ctx.fillStyle = 'rgba(10,6,4,' + (opt.shadowA == null ? 0.5 : opt.shadowA) + ')';
      ctx.fill(path); ctx.restore();
    }
    ctx.fillStyle = color; ctx.fill(path);
    ctx.save(); ctx.clip(path);
    const R = size;
    const g = ctx.createLinearGradient(c[0] - ux * R, c[1] - uy * R, c[0] + ux * R, c[1] + uy * R);
    const hi = opt.hi || 'rgba(255,238,210,0.30)';
    g.addColorStop(0, hi); g.addColorStop(0.5, 'rgba(0,0,0,0)'); g.addColorStop(1, 'rgba(0,0,0,0.34)');
    ctx.fillStyle = g; ctx.fillRect(c[0] - R * 2, c[1] - R * 2, R * 4, R * 4);
    const k = Math.max(1, depth * 0.35);
    ctx.lineWidth = k * 1.6;
    ctx.save(); ctx.translate(ux * k, uy * k); ctx.strokeStyle = opt.rim || 'rgba(255,240,215,0.45)'; ctx.stroke(path); ctx.restore();
    ctx.save(); ctx.translate(-ux * k, -uy * k); ctx.strokeStyle = 'rgba(0,0,0,0.35)'; ctx.stroke(path); ctx.restore();
    ctx.restore();
  }
  function capsule(p, x1, y1, x2, y2, r) {
    const a = Math.atan2(y2 - y1, x2 - x1), q = Math.PI / 2;
    p.moveTo(x1 + Math.cos(a - q) * r, y1 + Math.sin(a - q) * r);
    p.lineTo(x2 + Math.cos(a - q) * r, y2 + Math.sin(a - q) * r);
    p.arc(x2, y2, r, a - q, a + q);
    p.lineTo(x1 + Math.cos(a + q) * r, y1 + Math.sin(a + q) * r);
    p.arc(x1, y1, r, a + q, a + 3 * q);
    p.closePath();
  }

  // ---------- hands ----------
  const FING = [[-0.29, 0.74, 0.105], [-0.10, 0.82, 0.11], [0.10, 0.76, 0.105], [0.27, 0.60, 0.092]];
  const CURL = { flat: [1, 1, 1, 1], rest: [0.92, 0.96, 0.9, 0.86], point: [1, 0.34, 0.3, 0.3], trace: [1, 0.93, 0.36, 0.32], pen: [0.66, 0.36, 0.32, 0.3] };
  function handLocal(pose) {
    const p = new Path2D(), parts = [];
    p.ellipse(0.42, 0.0, 0.52, 0.43, 0, 0, TAU);
    const cm = CURL[pose];
    FING.forEach((f, k) => {
      const spread = (pose === 'flat' || pose === 'rest') ? f[0] * 0.35 : f[0] * 0.12;
      const len = f[1] * cm[k], bx = 0.82, by = f[0];
      const tx = bx + Math.cos(spread) * len, ty = by + Math.sin(spread) * len;
      const q = new Path2D(); capsule(q, bx - 0.08, by, tx, ty, f[2]);
      p.addPath(q); parts.push({ q, tx, ty, r: f[2], ext: cm[k] > 0.6, a: spread });
    });
    const ta = pose === 'pen' ? -0.42 : -0.95, tl = pose === 'pen' ? 0.62 : 0.5;
    const t = new Path2D(); capsule(t, 0.28, -0.33, 0.28 + Math.cos(ta) * tl, -0.33 + Math.sin(ta) * tl, 0.12);
    p.addPath(t); parts.push({ q: t, tx: 0.28 + Math.cos(ta) * tl, ty: -0.33 + Math.sin(ta) * tl, r: 0.12, ext: true, a: ta, thumb: true });
    return { p, parts, tip: [parts[0].tx, parts[0].ty] };
  }
  // o: {tip:[x,y], from:[x,y], s, ys, pose, who, flip, L, armW, tool:'pen'|'knife', dim}
  function drawHand(ctx, o) {
    const C = CAST[o.who], hl = handLocal(o.pose);
    let anchor = hl.tip;
    if (o.tool) anchor = [hl.tip[0] + 0.3, hl.tip[1] + 0.05];
    const vx = o.tip[0] - o.from[0], vy = o.tip[1] - o.from[1];
    const ang = Math.atan2(vy / o.ys, vx) * 180 / Math.PI + (o.twist || 0);
    const M = new DOMMatrix().translate(o.tip[0], o.tip[1]).scale(1, o.ys).rotate(ang)
      .scale(o.s, o.s * (o.flip ? -1 : 1)).translate(-anchor[0], -anchor[1]);
    const wr = M.transformPoint(new DOMPoint(-0.02, 0)), wr2 = M.transformPoint(new DOMPoint(-0.02, 0.42)), wr1 = M.transformPoint(new DOMPoint(-0.02, -0.42));
    const L = o.L, depth = o.s * 0.09;
    // arm (sleeve)
    const ax = wr.x - o.from[0], ay = wr.y - o.from[1], al = hyp(ax, ay) || 1, nx = -ay / al, ny = ax / al;
    const aw = (o.armW || o.s * 1.1) / 2, ww = hyp(wr2.x - wr1.x, wr2.y - wr1.y) / 2 * 1.08;
    const arm = new Path2D();
    arm.moveTo(o.from[0] + nx * aw, o.from[1] + ny * aw);
    arm.lineTo(wr.x + nx * ww, wr.y + ny * ww);
    arm.quadraticCurveTo(wr.x + ax / al * ww * 0.5, wr.y + ay / al * ww * 0.5, wr.x - nx * ww, wr.y - ny * ww);
    arm.lineTo(o.from[0] - nx * aw, o.from[1] - ny * aw);
    arm.closePath();
    const mid = [(o.from[0] + wr.x) / 2, (o.from[1] + wr.y) / 2];
    paperShape(ctx, arm, C.sleeve, mid, L, depth * 1.3, al / 2, { shadowA: o.dim ? 0.3 : 0.5, hi: o.hiCol, rim: o.rimCol });
    // cuff
    const cuff = new Path2D();
    const cb = [wr.x - ax / al * o.s * 0.22, wr.y - ay / al * o.s * 0.22];
    cuff.moveTo(cb[0] + nx * ww * 1.04, cb[1] + ny * ww * 1.04); cuff.lineTo(wr.x + nx * ww, wr.y + ny * ww);
    cuff.lineTo(wr.x - nx * ww, wr.y - ny * ww); cuff.lineTo(cb[0] - nx * ww * 1.04, cb[1] - ny * ww * 1.04); cuff.closePath();
    ctx.fillStyle = C.cuff; ctx.fill(cuff);
    // hand
    const P = new Path2D(); P.addPath(hl.p, M);
    const hc = M.transformPoint(new DOMPoint(0.6, 0));
    paperShape(ctx, P, C.skin, [hc.x, hc.y], L, depth, o.s, { shadowA: o.dim ? 0.35 : 0.55, hi: o.hiCol, rim: o.rimCol });
    ctx.save(); ctx.clip(P);
    ctx.strokeStyle = 'rgba(50,25,15,0.32)'; ctx.lineWidth = Math.max(1, o.s * 0.018);
    hl.parts.forEach(f => { const q = new Path2D(); q.addPath(f.q, M); ctx.stroke(q); });
    // nails and knuckle creases
    hl.parts.forEach(f => {
      if (!f.ext) {
        const k = M.transformPoint(new DOMPoint(f.tx, f.ty));
        ctx.fillStyle = 'rgba(40,20,10,0.22)'; ctx.beginPath(); ctx.arc(k.x, k.y, o.s * 0.05, 0, TAU); ctx.fill();
        return;
      }
      const nxp = f.tx - Math.cos(f.a) * f.r * 0.9, nyp = f.ty - Math.sin(f.a) * f.r * 0.9;
      const nm = new DOMMatrix().multiply(M).translate(nxp, nyp).rotate(f.a * 180 / Math.PI);
      const nail = new Path2D(); nail.ellipse(0, 0, f.r * 0.85, f.r * 0.62, 0, 0, TAU);
      const q = new Path2D(); q.addPath(nail, nm);
      ctx.fillStyle = o.dim ? 'rgba(220,210,220,0.18)' : 'rgba(255,225,205,0.42)'; ctx.fill(q);
    });
    ctx.restore();
    // tools
    if (o.tool === 'pen') {
      const a = M.transformPoint(new DOMPoint(anchor[0], anchor[1]));
      const b = M.transformPoint(new DOMPoint(-0.25, -0.75));
      const pen = new Path2D(); capsuleScreen(pen, a, b, o.s * 0.07);
      paperShape(ctx, pen, '#1d1f24', [(a.x + b.x) / 2, (a.y + b.y) / 2], L, depth * 0.8, o.s, { shadowA: 0.4 });
      ctx.fillStyle = '#0b0b0c'; ctx.beginPath(); ctx.arc(a.x, a.y, o.s * 0.045, 0, TAU); ctx.fill();
      // re-draw thumb and index over the pen
      const fp = new Path2D(); fp.addPath(hl.parts[0].q, M); fp.addPath(hl.parts[4].q, M);
      ctx.fillStyle = C.skin; ctx.fill(fp);
      ctx.save(); ctx.clip(fp); ctx.strokeStyle = 'rgba(50,25,15,0.35)'; ctx.lineWidth = Math.max(1, o.s * 0.02); ctx.stroke(fp); ctx.restore();
    }
    if (o.tool === 'knife') {
      const a = M.transformPoint(new DOMPoint(anchor[0] + 0.1, anchor[1]));
      const b = M.transformPoint(new DOMPoint(hl.tip[0] - 0.1, hl.tip[1] + 0.1));
      const blade = new Path2D(); capsuleScreen(blade, a, b, o.s * 0.16);
      paperShape(ctx, blade, '#b9bcc0', [(a.x + b.x) / 2, (a.y + b.y) / 2], L, depth * 0.4, o.s, { shadowA: 0.3 });
      ctx.save(); ctx.clip(blade); ctx.fillStyle = 'rgba(245,242,232,0.95)';
      ctx.beginPath(); ctx.arc(a.x, a.y, o.s * 0.2, 0, TAU); ctx.fill(); ctx.restore();
      const hnd = M.transformPoint(new DOMPoint(-0.3, -0.6));
      const handle = new Path2D(); capsuleScreen(handle, b, hnd, o.s * 0.08);
      paperShape(ctx, handle, '#5a3b26', [(b.x + hnd.x) / 2, (b.y + hnd.y) / 2], L, depth * 0.6, o.s, { shadowA: 0.3 });
      const fp = new Path2D(); fp.addPath(hl.parts[0].q, M); fp.addPath(hl.parts[4].q, M);
      ctx.fillStyle = C.skin; ctx.fill(fp);
      ctx.save(); ctx.clip(fp); ctx.strokeStyle = 'rgba(50,25,15,0.35)'; ctx.lineWidth = 1.5; ctx.stroke(fp); ctx.restore();
    }
  }
  function capsuleScreen(p, a, b, r) { capsule(p, a.x, a.y, b.x, b.y, r); }

  // ---------- busts (KF1 only) ----------
  function drawBust(ctx, x, y, s, who, eyes, tilt, L) {
    const C = CAST[who];
    const torso = new Path2D();
    torso.moveTo(x - 2.8 * s, y + 4.6 * s); torso.lineTo(x - 2.55 * s, y + 2.4 * s);
    torso.quadraticCurveTo(x - 2.25 * s, y + 1.55 * s, x - 0.9 * s, y + 1.45 * s);
    torso.lineTo(x + 0.9 * s, y + 1.45 * s);
    torso.quadraticCurveTo(x + 2.25 * s, y + 1.55 * s, x + 2.55 * s, y + 2.4 * s);
    torso.lineTo(x + 2.8 * s, y + 4.6 * s); torso.closePath();
    paperShape(ctx, torso, C.sleeve, [x, y + 3 * s], L, s * 0.12, 2.5 * s);
    // collar
    ctx.fillStyle = C.cuff; ctx.beginPath(); ctx.ellipse(x, y + 1.5 * s, 0.75 * s, 0.25 * s, 0, 0, TAU); ctx.fill();
    const neck = new Path2D(); neck.rect(x - 0.36 * s, y + 0.55 * s, 0.72 * s, 0.95 * s);
    paperShape(ctx, neck, C.skin, [x, y + s], L, s * 0.05, s, { noShadow: true });
    ctx.save(); ctx.translate(x, y); ctx.rotate(tilt * Math.PI / 180); ctx.translate(-x, -y);
    const M = ctx.getTransform();
    const head = new Path2D(); head.ellipse(x, y, 0.8 * s, s, 0, 0, TAU);
    // ears

  function drawWood(ctx, x0, y0, w, h, seed, persp) {
    const r = rng(seed);
    ctx.fillStyle = '#7d5638'; ctx.fillRect(x0, y0, w, h);
    // planks
    const n = 7;
    for (let k = 0; k <= n; k++) {
      const t = k / n, yy = persp ? y0 + h * Math.pow(t, 1.6) : y0 + h * t;
      ctx.fillStyle = 'rgba(40,22,12,0.55)'; ctx.fillRect(x0, yy - 1.5, w, 3);
      ctx.fillStyle = 'rgba(255,220,180,0.10)'; ctx.fillRect(x0, yy + 1.5, w, 1.5);
    }
    for (let i = 0; i < 140; i++) {
      const t = r(), yy = persp ? y0 + h * Math.pow(t, 1.6) : y0 + h * t;
      const amp = 2 + r() * 6, fr = 0.004 + r() * 0.01, ph = r() * TAU;
      ctx.strokeStyle = r() < 0.6 ? 'rgba(55,32,18,' + (0.12 + r() * 0.2) + ')' : 'rgba(200,150,105,' + (0.08 + r() * 0.12) + ')';
      ctx.lineWidth = 0.6 + r() * 1.8;
      ctx.beginPath();
      for (let xx = x0; xx <= x0 + w; xx += 16) { const yv = yy + Math.sin(xx * fr + ph) * amp * (persp ? (0.4 + t) : 1); xx === x0 ? ctx.moveTo(xx, yv) : ctx.lineTo(xx, yv); }
      ctx.stroke();
    }
  }
  function lightPool(ctx, cx, cy, r, warm) {
    ctx.save(); ctx.globalCompositeOperation = 'multiply';
    const g = ctx.createRadialGradient(cx, cy, 0, cx, cy, r);
    if (warm) { g.addColorStop(0, 'rgb(255,244,222)'); g.addColorStop(0.35, 'rgb(225,196,160)'); g.addColorStop(0.7, 'rgb(105,82,66)'); g.addColorStop(1, 'rgb(34,27,25)'); }
    ctx.fillStyle = g; ctx.fillRect(0, 0, VW, VH); ctx.restore();
  }

  function drawLamp(ctx, base, joint, head, aim, size, on) {
    const metal = on ? '#3b3734' : '#22252c';
    ctx.lineCap = 'round';
    if (base) {
      ctx.fillStyle = metal; ctx.beginPath(); ctx.ellipse(base[0], base[1], size * 0.75, size * 0.28, 0, 0, TAU); ctx.fill();

      ctx.strokeStyle = on ? 'rgba(255,225,180,0.35)' : 'rgba(150,170,200,0.18)'; ctx.lineWidth = size * 0.03;
      ctx.beginPath(); ctx.moveTo(base[0] - size * 0.03, base[1]); ctx.quadraticCurveTo(joint[0] - size * 0.04, joint[1], head[0] - size * 0.03, head[1]); ctx.stroke();

    const g = ctx.createLinearGradient(0, -0.55 * size, 0, 0.55 * size);
    g.addColorStop(0, on ? 'rgba(255,220,170,0.35)' : 'rgba(150,170,205,0.22)'); g.addColorStop(0.5, 'rgba(0,0,0,0)'); g.addColorStop(1, 'rgba(0,0,0,0.4)');

    ctx.save(); ctx.globalCompositeOperation = 'screen';

    ctx.restore();
  }

  }

  }
