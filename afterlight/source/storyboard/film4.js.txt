// AFTERLIGHT · film 4 · The Same Water
// Keyframe module. Plain browser JS, Canvas 2D only, deterministic (seeded).
(function () {
  'use strict';

  // ---------- seeded helpers ----------
  function rng(seed) {
    let a = seed >>> 0;
    return function () {
      a |= 0; a = (a + 0x6D2B79F5) | 0;
      let t = Math.imul(a ^ (a >>> 15), 1 | a);
      t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
      return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
    };
  }

  // ---------- palette ----------
  const P = {
    floor: '#2b3236', floorGap: '#1d2326',
    rim: '#b9b1a3', rimDark: '#8f887c', rimLight: '#d6cfc2',
    water: [38, 74, 82], waterDeep: [24, 52, 60],
    light: [226, 240, 255],               // pale, short-wavelength-leaning white (the paper: shorter wavelengths felt warmer)
    wall: '#cfc6b6', wallDark: '#b8ae9c',
    metal: '#3a3e42', metalLight: '#5d6368',
    A: { skin: '#875736', skinDark: '#5e3a22', sleeve: '#c48a3a', sleeveDark: '#94661f', hair: '#1c1714', trousers: '#4a3f33' },
    B: { skin: '#d9b391', skinDark: '#a9825f', sleeve: '#8ea385', sleeveDark: '#64785c', hair: '#5a4532', trousers: '#3c4440' },
    wood: '#9a6b42', woodDark: '#6e4a2c'
  };

  const cache = {};

  // paper grain overlay, cached
  function grain(W, H) {
    const key = 'grain' + W + 'x' + H;
    if (cache[key]) return cache[key];
    const c = document.createElement('canvas'); c.width = 320; c.height = 180;
    const g = c.getContext('2d');
    const id = g.createImageData(320, 180); const r = rng(77);
    for (let i = 0; i < id.data.length; i += 4) {
      const v = 128 + (r() - 0.5) * 70;
      id.data[i] = id.data[i + 1] = id.data[i + 2] = v; id.data[i + 3] = 26;
    }
    g.putImageData(id, 0, 0);
    // fibres
    for (let i = 0; i < 260; i++) {
      g.strokeStyle = r() < 0.5 ? 'rgba(255,255,255,0.10)' : 'rgba(0,0,0,0.10)';
      g.lineWidth = 0.5;
      const x = r() * 320, y = r() * 180, a = r() * Math.PI;
      g.beginPath(); g.moveTo(x, y); g.lineTo(x + Math.cos(a) * 6, y + Math.sin(a) * 6); g.stroke();
    }
    cache[key] = c; return c;
  }

  // caustic field: spots = [{x,y,r,k}] in 0..1 coords of the image
  function caustic(seed, spots, cw, ch) {
    cw = cw || 256; ch = ch || 144;
    const key = 'c' + seed + JSON.stringify(spots) + cw + 'x' + ch;
    if (cache[key]) return cache[key];
    const c = document.createElement('canvas'); c.width = cw; c.height = ch;
    const g = c.getContext('2d'); const id = g.createImageData(cw, ch);
    const r = rng(seed); const s1 = r() * 6, s2 = r() * 6, s3 = r() * 6;
    for (let y = 0; y < ch; y++) {
      for (let x = 0; x < cw; x++) {
        const u = x * 0.11, v = y * 0.11 * (cw / ch) * (144 / 256) * 1.0;
        let lit = 0;
        for (const s of spots) {
          const dx = x / cw - s.x, dy = (y / ch - s.y) * (ch / cw) * (s.sq || 1);
          const d = Math.sqrt(dx * dx + dy * dy) / s.r;
          lit += s.k * Math.max(0, 1 - d * d);
        }
        const f = 1 + lit * 0.9;
        const a = Math.sin(u * 1.3 * f + Math.sin(v * 0.9 + s1) * 1.7) +
                  Math.sin(v * 1.1 * f + Math.sin(u * 0.7 + s2) * 1.9) +
                  Math.sin((u + v) * 0.6 + s3);
        let ridge = 1 - Math.min(1, Math.abs(a) * 1.5); ridge = ridge * ridge * ridge;
        const amb = 0.18 + 0.55 * lit;
        const L = Math.min(1, ridge * amb * 1.6 + lit * 0.32);
        const deep = 0.5 + 0.5 * Math.sin(u * 0.2 + v * 0.15 + s1);
        const i = (y * cw + x) * 4;
        const br = [P.water[0] * (1 - deep * 0.3), P.water[1] * (1 - deep * 0.25), P.water[2] * (1 - deep * 0.2)];
        id.data[i] = br[0] + (P.light[0] - br[0]) * L;
        id.data[i + 1] = br[1] + (P.light[1] - br[1]) * L;
        id.data[i + 2] = br[2] + (P.light[2] - br[2]) * L;
        id.data[i + 3] = 255;
      }
    }
    g.putImageData(id, 0, 0); cache[key] = c; return c;
  }

  // jittered ("cut") path primitives
  function cutRRect(ctx, x, y, w, h, rad, seed, j) {
    const r = rng(seed); j = j == null ? 1.6 : j;
    const pts = [];
    const corner = (cx, cy, a0) => { for (let i = 0; i <= 6; i++) { const a = a0 + i / 6 * Math.PI / 2; pts.push([cx + Math.cos(a) * rad, cy + Math.sin(a) * rad]); } };
    const edge = (x0, y0, x1, y1) => { const n = Math.max(2, Math.floor(Math.hypot(x1 - x0, y1 - y0) / 40)); for (let i = 1; i < n; i++) pts.push([x0 + (x1 - x0) * i / n, y0 + (y1 - y0) * i / n]); };
    corner(x + w - rad, y + rad, -Math.PI / 2); edge(x + w, y + rad, x + w, y + h - rad);
    corner(x + w - rad, y + h - rad, 0); edge(x + w - rad, y + h, x + rad, y + h);
    corner(x + rad, y + h - rad, Math.PI / 2); edge(x, y + h - rad, x, y + rad);
    corner(x + rad, y + rad, Math.PI); edge(x + rad, y, x + w - rad, y);
    ctx.beginPath();
    pts.forEach((p, i) => { const q = [p[0] + (r() - 0.5) * j * 2, p[1] + (r() - 0.5) * j * 2]; i ? ctx.lineTo(q[0], q[1]) : ctx.moveTo(q[0], q[1]); });
    ctx.closePath();
  }
  function cutEllipse(ctx, cx, cy, rx, ry, rot, seed, j) {
    const r = rng(seed); j = j == null ? 1.4 : j; const n = 44;
    ctx.beginPath();
    for (let i = 0; i < n; i++) {
      const a = i / n * Math.PI * 2; const jj = (r() - 0.5) * j * 2;
      const ex = Math.cos(a) * (rx + jj), ey = Math.sin(a) * (ry + jj);
      const x = cx + ex * Math.cos(rot) - ey * Math.sin(rot), y = cy + ex * Math.sin(rot) + ey * Math.cos(rot);
      i ? ctx.lineTo(x, y) : ctx.moveTo(x, y);
    }
    ctx.closePath();
  }
  function paper(ctx, fill, sh) {
    ctx.save();
    if (sh !== false) { ctx.shadowColor = 'rgba(0,0,0,0.38)'; ctx.shadowBlur = sh || 7; ctx.shadowOffsetX = 3; ctx.shadowOffsetY = 5; }
    ctx.fillStyle = fill; ctx.fill(); ctx.restore();
    ctx.save(); ctx.strokeStyle = 'rgba(255,255,255,0.18)'; ctx.lineWidth = 1; ctx.stroke(); ctx.restore();
  }
  function poly(ctx, pts) { ctx.beginPath(); pts.forEach((p, i) => i ? ctx.lineTo(p[0], p[1]) : ctx.moveTo(p[0], p[1])); ctx.closePath(); }

  // ---------- body parts ----------
  // Hand in local frame: +x from wrist toward fingertips; handed = +1/-1 puts thumb on -y or +y side.
  function hand(ctx, wx, wy, ang, s, who, pose, handed, opt) {
    opt = opt || {};
    const c = P[who];
    ctx.save(); ctx.translate(wx, wy); ctx.rotate(ang); ctx.scale(1, handed);
    const L = { flat: [0.44, 0.50, 0.47, 0.36], curl: [0.17, 0.19, 0.18, 0.15], point: [0.50, 0.17, 0.16, 0.14], open: [0.46, 0.52, 0.49, 0.38] }[pose];
    const spread = { flat: [-0.10, -0.03, 0.04, 0.12], curl: [0.0, 0.0, 0.0, 0.05], point: [-0.06, 0.05, 0.06, 0.1], open: [-0.24, -0.08, 0.08, 0.26] }[pose];
    const ys = [-0.22, -0.075, 0.075, 0.21];
    const fw = 0.125 * s;
    const fingers = ys.map((y, i) => {
      const bx = 0.5 * s, by = y * s * (pose === 'curl' ? 0.95 : 1);
      return [bx, by, bx + Math.cos(spread[i]) * L[i] * s, by + Math.sin(spread[i]) * L[i] * s];
    });
    const thumbA = pose === 'curl' ? -0.35 : pose === 'open' ? -1.05 : -0.75;
    const thumbL = pose === 'curl' ? 0.26 : 0.4;
    const th = [0.13 * s, -0.24 * s, 0.13 * s + Math.cos(thumbA) * thumbL * s, -0.24 * s + Math.sin(thumbA) * thumbL * s];
    const palm = [[-0.02 * s, -0.19 * s], [0.18 * s, -0.27 * s], [0.52 * s, -0.27 * s], [0.56 * s, 0], [0.52 * s, 0.26 * s], [0.2 * s, 0.25 * s], [-0.02 * s, 0.18 * s]];
    // shadow + outline pass
    ctx.save();
    if (opt.shadow !== false) { ctx.shadowColor = 'rgba(0,0,0,' + (opt.shadowA || 0.35) + ')'; ctx.shadowBlur = opt.shadowBlur || 6; ctx.shadowOffsetX = opt.sx || 3; ctx.shadowOffsetY = (opt.sy || 5) * handed; }
    ctx.strokeStyle = c.skinDark; ctx.fillStyle = c.skinDark; ctx.lineCap = 'round';
    ctx.lineWidth = fw + 4; fingers.forEach(f => { ctx.beginPath(); ctx.moveTo(f[0], f[1]); ctx.lineTo(f[2], f[3]); ctx.stroke(); });
    ctx.lineWidth = fw * 1.15 + 4; ctx.beginPath(); ctx.moveTo(th[0], th[1]); ctx.lineTo(th[2], th[3]); ctx.stroke();
    poly(ctx, palm); ctx.lineWidth = 4; ctx.lineJoin = 'round'; ctx.stroke(); ctx.fill();
    ctx.restore();
    // fill pass
    ctx.strokeStyle = c.skin; ctx.fillStyle = c.skin; ctx.lineCap = 'round';
    ctx.lineWidth = fw; fingers.forEach(f => { ctx.beginPath(); ctx.moveTo(f[0], f[1]); ctx.lineTo(f[2], f[3]); ctx.stroke(); });
    ctx.lineWidth = fw * 1.15; ctx.beginPath(); ctx.moveTo(th[0], th[1]); ctx.lineTo(th[2], th[3]); ctx.stroke();
    poly(ctx, palm); ctx.fill();
    // knuckle / curl creases
    ctx.strokeStyle = 'rgba(0,0,0,0.22)'; ctx.lineWidth = 1.4;
    if (pose === 'curl') { ctx.beginPath(); ctx.moveTo(0.58 * s, -0.25 * s); ctx.quadraticCurveTo(0.66 * s, 0, 0.58 * s, 0.24 * s); ctx.stroke(); }
    fingers.forEach(f => { const mx = f[0] + (f[2] - f[0]) * 0.55, my = f[1] + (f[3] - f[1]) * 0.55; ctx.beginPath(); ctx.moveTo(mx, my - fw * 0.3); ctx.lineTo(mx, my + fw * 0.3); ctx.stroke(); });
    // nails as tiny light marks
    ctx.fillStyle = 'rgba(255,240,230,0.35)';
    fingers.forEach(f => { if (Math.hypot(f[2] - f[0], f[3] - f[1]) > 0.3 * s) { ctx.beginPath(); ctx.arc(f[2] - Math.cos(0) * fw * 0.15, f[3], fw * 0.22, 0, 7); ctx.fill(); } });
    ctx.restore();
  }

  // Arm from shoulder S to wrist Wr, sleeve covering the upper part.
  function arm(ctx, S, Wr, who, w0, w1, sleeveFrac, shadow) {
    const c = P[who]; const dx = Wr[0] - S[0], dy = Wr[1] - S[1]; const len = Math.hypot(dx, dy);
    const nx = -dy / len, ny = dx / len;
    const at = (t, w) => [S[0] + dx * t + nx * w, S[1] + dy * t + ny * w];
    const wAt = t => w0 + (w1 - w0) * t;
    const sf = sleeveFrac == null ? 0.58 : sleeveFrac;
    // skin forearm
    poly(ctx, [at(sf - 0.05, wAt(sf) / 2), at(1, w1 / 2), at(1.02, 0), at(1, -w1 / 2), at(sf - 0.05, -wAt(sf) / 2)]);
    ctx.save(); if (shadow !== false) { ctx.shadowColor = 'rgba(0,0,0,0.35)'; ctx.shadowBlur = 6; ctx.shadowOffsetX = 3; ctx.shadowOffsetY = 5; }
    ctx.fillStyle = c.skin; ctx.fill(); ctx.restore();
    ctx.strokeStyle = c.skinDark; ctx.lineWidth = 2; ctx.stroke();
    // sleeve
    poly(ctx, [at(-0.02, w0 / 2), at(sf, wAt(sf) / 2 + 3), at(sf + 0.015, 0), at(sf, -wAt(sf) / 2 - 3), at(-0.02, -w0 / 2)]);
    ctx.save(); if (shadow !== false) { ctx.shadowColor = 'rgba(0,0,0,0.35)'; ctx.shadowBlur = 7; ctx.shadowOffsetX = 3; ctx.shadowOffsetY = 5; }
    ctx.fillStyle = c.sleeve; ctx.fill(); ctx.restore();
    ctx.strokeStyle = c.sleeveDark; ctx.lineWidth = 2; ctx.stroke();
    // fold lines
    ctx.strokeStyle = 'rgba(0,0,0,0.18)'; ctx.lineWidth = 1.5;
    [0.25, 0.42].forEach(t => { const a = at(t, wAt(t) * 0.35), b = at(t + 0.04, -wAt(t) * 0.2); ctx.beginPath(); ctx.moveTo(a[0], a[1]); ctx.lineTo(b[0], b[1]); ctx.stroke(); });
    return Math.atan2(dy, dx);
  }

  // overhead person: head + shoulders, facing direction 'face' (radians, toward basin)
  function topPerson(ctx, x, y, face, who, tilt, seed) {
    const c = P[who];
    ctx.save(); ctx.translate(x, y); ctx.rotate(face - Math.PI / 2);
    // shoulders (back toward camera edge)
    cutEllipse(ctx, 0, -18, 128, 50, 0, seed, 2); paper(ctx, c.sleeve, 10);
    ctx.strokeStyle = c.sleeveDark; ctx.lineWidth = 2; ctx.stroke();
    // collar
    cutEllipse(ctx, 0, 2, 38, 20, 0, seed + 1, 1); ctx.fillStyle = c.skinDark; ctx.fill();
    // head (seen from above: hair dominates), tilted
    ctx.rotate(tilt || 0);
    cutEllipse(ctx, 0, 26, 40, 46, 0, seed + 2, 1.2); paper(ctx, c.hair, 8);
    // ears
    ctx.fillStyle = c.skin;
    cutEllipse(ctx, -40, 30, 7, 11, 0, seed + 3, 0.6); ctx.fill();
    cutEllipse(ctx, 40, 30, 7, 11, 0, seed + 4, 0.6); ctx.fill();
    // brow/nose tip visible at the forward edge (looking down at the water)
    cutEllipse(ctx, 0, 66, 24, 10, 0, seed + 5, 0.6); ctx.fillStyle = c.skin; ctx.fill();
    // part line in hair
    ctx.strokeStyle = 'rgba(255,255,255,0.10)'; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(-6, -12); ctx.quadraticCurveTo(-2, 20, 4, 54); ctx.stroke();
    ctx.restore();
  }

  function ripples(ctx, x, y, n, r0, alpha, sq) {
    ctx.save(); ctx.strokeStyle = 'rgba(226,240,255,' + alpha + ')';
    for (let i = 0; i < n; i++) { ctx.lineWidth = 1.6 - i * 0.3; ctx.beginPath(); ctx.ellipse(x, y, r0 + i * 16, (r0 + i * 16) * (sq || 0.8), 0, 0, 7); ctx.globalAlpha = 1 - i / (n + 1); ctx.stroke(); }
    ctx.restore();
  }

  function finish(ctx, W, H) {
    ctx.drawImage(grain(W, H), 0, 0, 1280, 720);
    const g = ctx.createRadialGradient(640, 360, 260, 640, 360, 820);
    g.addColorStop(0, 'rgba(0,0,0,0)'); g.addColorStop(1, 'rgba(0,0,0,0.55)');
    ctx.fillStyle = g; ctx.fillRect(0, 0, 1280, 720);
  }

  // ---------- OVERHEAD SET (K1, K2, K5) ----------
  const BASIN = { x: 210, y: 150, w: 860, h: 420, rim: 38 };

  function overhead(ctx, W, H, o) {
    ctx.save(); ctx.scale(W / 1280, H / 720);
    if (o.zoom) { ctx.translate(640, 360); ctx.scale(o.zoom, o.zoom); ctx.translate(-o.cx, -o.cy); }
    const B = BASIN;
    // floor tiles
    ctx.fillStyle = P.floorGap; ctx.fillRect(-400, -400, 2100, 1500);
    const r = rng(11);
    for (let ty = -360; ty < 1100; ty += 96) for (let tx = -380; tx < 1700; tx += 96) {
      cutRRect(ctx, tx + 3, ty + 3, 90, 90, 4, (tx * 7 + ty * 13) & 0xffff, 1);
      const v = 40 + Math.floor(r() * 10); ctx.fillStyle = 'rgb(' + v + ',' + (v + 7) + ',' + (v + 11) + ')'; ctx.fill();
    }
    // basin rim (stone, cut paper)
    cutRRect(ctx, B.x - B.rim, B.y - B.rim, B.w + B.rim * 2, B.h + B.rim * 2, 46, 21, 2); paper(ctx, P.rim, 14);
    cutRRect(ctx, B.x - B.rim + 8, B.y - B.rim + 8, B.w + B.rim * 2 - 16, B.h + B.rim * 2 - 16, 40, 22, 1.5);
    ctx.strokeStyle = P.rimLight; ctx.lineWidth = 2; ctx.stroke();
    // people bodies
    (o.people || []).forEach((p, i) => topPerson(ctx, p.x, p.y, p.face, p.who, p.tilt, 300 + i * 10));
    // water
    const inner = () => cutRRect(ctx, B.x, B.y, B.w, B.h, 22, 23, 1.2);
    const spotsN = (o.spots || []).map(s => ({ x: (s.x - B.x) / B.w, y: (s.y - B.y) / B.h, r: s.r / B.w, k: s.k, sq: B.w / B.h * (B.h / B.w) }));
    const cimg = caustic(o.seed || 5, spotsN, 300, 147);
    ctx.save(); inner(); ctx.clip();
    ctx.imageSmoothingEnabled = true; ctx.drawImage(cimg, B.x, B.y, B.w, B.h);
    // inner wall shadow (rim casts onto water)
    const gs = ctx.createLinearGradient(0, B.y, 0, B.y + 50); gs.addColorStop(0, 'rgba(0,0,0,0.45)'); gs.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = gs; ctx.fillRect(B.x, B.y, B.w, 50);
    const gl = ctx.createLinearGradient(B.x, 0, B.x + 40, 0); gl.addColorStop(0, 'rgba(0,0,0,0.35)'); gl.addColorStop(1, 'rgba(0,0,0,0)');
    ctx.fillStyle = gl; ctx.fillRect(B.x, B.y, 40, B.h);
    // submerged limbs
    (o.limbs || []).filter(l => l.state === 'flat').forEach(l => {
      const a = arm(ctx, l.S, l.W, l.who, 64, 40, 0.5, false);
      hand(ctx, l.W[0], l.W[1], a + (l.twist || 0), l.s || 92, l.who, l.pose || 'flat', l.handed, { shadowA: 0.25, sx: 2, sy: 3 });
    });
    // water veil over submerged skin
    ctx.fillStyle = 'rgba(' + P.waterDeep.join(',') + ',0.40)'; ctx.fillRect(B.x, B.y, B.w, B.h);
    ctx.globalCompositeOperation = 'screen'; ctx.globalAlpha = 0.55; ctx.drawImage(cimg, B.x, B.y, B.w, B.h);
    ctx.globalAlpha = 0.5;
    (o.limbs || []).filter(l => l.state === 'flat').forEach(l => {
      const a = arm(ctx, l.S, l.W, l.who, 64, 40, 0.5, false);
      hand(ctx, l.W[0], l.W[1], a + (l.twist || 0), l.s || 92, l.who, l.pose || 'flat', l.handed, { shadow: false });
    });
    ctx.globalAlpha = 1;
    // beam pools: pale, soft-edged, on the water
    (o.spots || []).forEach(s => {
      const g = ctx.createRadialGradient(s.x, s.y, 0, s.x, s.y, s.r * 1.3);
      g.addColorStop(0, 'rgba(226,240,255,' + (0.30 * s.k) + ')'); g.addColorStop(0.6, 'rgba(226,240,255,' + (0.14 * s.k) + ')'); g.addColorStop(1, 'rgba(226,240,255,0)');
      ctx.fillStyle = g; ctx.beginPath(); ctx.ellipse(s.x, s.y, s.r * 1.3, s.r * 1.15, 0, 0, 7); ctx.fill();
    });
    ctx.globalCompositeOperation = 'source-over';
    // ripples where arms enter / near resting hands
    (o.limbs || []).filter(l => l.state === 'flat').forEach(l => ripples(ctx, l.entry[0], l.entry[1], 3, 26, 0.32, 0.6));
    (o.rings || []).forEach(q => ripples(ctx, q[0], q[1], q[2], q[3], q[4] || 0.3, 0.8));
    ctx.restore();
    // arm portions outside the water, re-drawn over rim (clip to outside of inner)
    ctx.save(); ctx.beginPath(); ctx.rect(-400, -400, 2100, 1500); cutRRect(ctx, B.x, B.y, B.w, B.h, 22, 23, 1.2);
    ctx.clip('evenodd');
    (o.limbs || []).filter(l => l.state === 'flat').forEach(l => arm(ctx, l.S, l.W, l.who, 64, 40, 0.5));
    ctx.restore();
    // lifted / gesturing limbs (above water: larger, with cast shadow on the water)
    (o.limbs || []).filter(l => l.state !== 'flat').forEach(l => {
      // cast shadow on water
      ctx.save(); inner(); ctx.clip();
      ctx.fillStyle = 'rgba(8,20,26,0.40)'; ctx.filter = 'blur(10px)';
      ctx.beginPath(); ctx.ellipse(l.W[0] + 34, l.W[1] + 40, (l.s || 92) * 0.6, (l.s || 92) * 0.38, Math.atan2(l.W[1] - l.S[1], l.W[0] - l.S[0]), 0, 7); ctx.fill();
      ctx.filter = 'none'; ctx.restore();
      const a = arm(ctx, l.S, l.W, l.who, 70, 46, 0.5);
      hand(ctx, l.W[0], l.W[1], a + (l.twist || 0), (l.s || 92) * 1.08, l.who, l.pose || 'curl', l.handed, { shadowA: 0.45, sx: 10, sy: 14, shadowBlur: 14 });
      // drips
      if (l.drips) { const rr = rng(l.drips); ctx.fillStyle = 'rgba(226,240,255,0.75)';
        for (let i = 0; i < 6; i++) { const t = rr(); ctx.beginPath(); ctx.ellipse(l.W[0] + (rr() - 0.5) * 60 + Math.cos(a) * 40, l.W[1] + (rr() - 0.5) * 50 + Math.sin(a) * 40, 2.2 + t * 2, 3 + t * 3, a, 0, 7); ctx.fill(); }
        ctx.save(); inner(); ctx.clip(); ripples(ctx, l.W[0] + 30, l.W[1] + 44, 3, 12, 0.4, 0.75); ctx.restore(); }
    });
    // props
    if (o.cup) { const [cx, cy] = o.cup;
      cutEllipse(ctx, cx, cy, 30, 30, 0, 61, 1); paper(ctx, P.wood, 8);
      cutEllipse(ctx, cx, cy, 22, 22, 0, 62, 0.8); ctx.fillStyle = P.woodDark; ctx.fill();
      cutEllipse(ctx, cx - 3, cy - 3, 14, 13, 0, 63, 0.6); ctx.fillStyle = 'rgb(' + P.water.join(',') + ')'; ctx.fill(); }
    if (o.towel) { const [tx, ty] = o.towel; cutRRect(ctx, tx, ty, 120, 70, 8, 64, 2); paper(ctx, '#e6e0d3', 8);

  function kneel(ctx, x, baseY, facing, who, look, armTo, seed) {
    const c = P[who];
    ctx.save(); ctx.translate(x, baseY); ctx.scale(facing, 1);
    // folded legs
    cutRRect(ctx, -70, -70, 150, 70, 30, seed, 1.5); paper(ctx, c.trousers, 8);
    cutEllipse(ctx, -78, -12, 22, 14, 0, seed + 1, 1); ctx.fillStyle = c.skinDark; ctx.fill(); // heel
    // torso leaning toward basin
    poly(ctx, [[-50, -60], [-30, -230], [40, -238], [62, -64]]);
    paper(ctx, c.sleeve, 10); ctx.strokeStyle = c.sleeveDark; ctx.lineWidth = 2; ctx.stroke();
    // neck + head, tilted up by 'look'
    ctx.save(); ctx.translate(14, -240); ctx.rotate(-look);
    cutRRect(ctx, -12, -36, 26, 40, 8, seed + 2, 0.8); ctx.fillStyle = c.skin; ctx.fill();
    cutEllipse(ctx, 6, -74, 40, 46, 0, seed + 3, 1.2); paper(ctx, c.skin, 8);
    // hair cap on the back of the head
    ctx.save(); cutEllipse(ctx, 6, -74, 41, 47, 0, seed + 3, 1.2); ctx.clip();
    ctx.fillStyle = c.hair; ctx.beginPath(); ctx.ellipse(-16, -92, 44, 42, -0.4, 0, 7); ctx.fill(); ctx.restore();
    // face features in profile: brow, nose, eye
    ctx.fillStyle = c.skin; poly(ctx, [[44, -82], [56, -66], [44, -62]]); ctx.fill();
    ctx.fillStyle = 'rgba(20,14,10,0.85)'; ctx.beginPath(); ctx.ellipse(30, -80, 3.2, 4, 0, 0, 7); ctx.fill();
    ctx.strokeStyle = 'rgba(0,0,0,0.35)'; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(22, -88); ctx.lineTo(38, -90); ctx.stroke();
    ctx.restore();
    ctx.restore();
    // arm in world coords
    if (armTo) {

  }

  function lampHead(ctx, x, y, aim, on, seed, scale) {
    scale = scale || 1;
    ctx.save(); ctx.translate(x, y); ctx.rotate(aim); ctx.scale(scale, scale);
    // yoke
    ctx.strokeStyle = P.metal; ctx.lineWidth = 7; ctx.beginPath(); ctx.moveTo(-34, -6); ctx.lineTo(-34, -40); ctx.lineTo(34, -40); ctx.lineTo(34, -6); ctx.stroke();
    // shade: truncated cone pointing +y
    poly(ctx, [[-22, -24], [22, -24], [42, 40], [-42, 40]]); paper(ctx, P.metal, 9);
    ctx.fillStyle = P.metalLight; poly(ctx, [[-22, -24], [-8, -24], [-24, 40], [-42, 40]]); ctx.fill();
    // lens
    cutEllipse(ctx, 0, 40, 42, 9, 0, seed, 0.6);
    ctx.fillStyle = on ? 'rgb(236,245,255)' : '#25292c'; ctx.fill();
    if (on) { ctx.save(); ctx.globalCompositeOperation = 'screen'; const g = ctx.createRadialGradient(0, 44, 0, 0, 44, 70); g.addColorStop(0, 'rgba(226,240,255,0.7)'); g.addColorStop(1, 'rgba(226,240,255,0)'); ctx.fillStyle = g; ctx.beginPath(); ctx.arc(0, 44, 70, 0, 7); ctx.fill(); ctx.restore(); }
    ctx.restore();
  }

  function beam(ctx, x0, y0, w0, x1, y1, rx, ry, alpha) {
    ctx.save(); ctx.globalCompositeOperation = 'screen';
    const g = ctx.createLinearGradient(x0, y0, x1, y1);
    g.addColorStop(0, 'rgba(226,240,255,' + alpha + ')'); g.addColorStop(1, 'rgba(226,240,255,' + alpha * 0.35 + ')');
    ctx.fillStyle = g; ctx.filter = 'blur(3px)';
    poly(ctx, [[x0 - w0, y0], [x0 + w0, y0], [x1 + rx, y1], [x1 - rx, y1]]); ctx.fill();
    ctx.filter = 'none'; ctx.restore();
  }

  }
