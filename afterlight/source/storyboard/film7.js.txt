// AFTERLIGHT film 7: The Man Who Stole the Month
// Attic red-figure idiom: terracotta figures reserved on black gloss, relief lines, meander bands.
(function () {
  'use strict';
  var CL = '#c96d3e', CLD = '#b45f34', LN = '#140d0a', DIL = '#7d4224', WH = '#efe5cf',
      BK = '#18110d', RED = '#7e2416', WAX = '#6a3c1d', WAXL = '#8a5428';

  function rng(seed) { var s = seed >>> 0; return function () { s = (s * 1664525 + 1013904223) >>> 0; return s / 4294967296; }; }
  function rad(d) { return d * Math.PI / 180; }
  function dirv(a, L) { return [Math.sin(rad(a)) * L, Math.cos(rad(a)) * L]; }
  function add(p, v) { return [p[0] + v[0], p[1] + v[1]]; }

  // ---------- ground: black gloss pot surface, meander bands ----------
  function meander(ctx, y0, h) {
    ctx.fillStyle = CL; ctx.fillRect(0, y0, 1280, h);
    ctx.strokeStyle = LN; ctx.lineWidth = 3;
    ctx.beginPath(); ctx.moveTo(0, y0 + 1.5); ctx.lineTo(1280, y0 + 1.5); ctx.moveTo(0, y0 + h - 1.5); ctx.lineTo(1280, y0 + h - 1.5); ctx.stroke();
    var u = h * 1.05, t = y0 + 6, b = y0 + h - 6, hh = b - t;
    ctx.lineWidth = Math.max(3, h * 0.12); ctx.lineCap = 'square'; ctx.lineJoin = 'miter';
    for (var x = -u * 0.3; x < 1280 + u; x += u) {
      var P = [[0, 0.88], [0, 0.12], [0.78, 0.12], [0.78, 0.66], [0.36, 0.66], [0.36, 0.42]];
      ctx.beginPath();
      P.forEach(function (p, i) { var X = x + p[0] * u, Y = t + p[1] * hh; if (i) ctx.lineTo(X, Y); else ctx.moveTo(X, Y); });
      ctx.moveTo(x, t + 0.88 * hh); ctx.lineTo(x + u, t + 0.88 * hh);
      ctx.stroke();
    }
    ctx.lineCap = 'butt';
  }
  function ground(ctx) {
    ctx.fillStyle = BK; ctx.fillRect(0, 0, 1280, 720);
    var g = ctx.createRadialGradient(640, 360, 80, 640, 360, 760);
    g.addColorStop(0, '#271b15'); g.addColorStop(1, '#0b0806');
    ctx.fillStyle = g; ctx.fillRect(0, 0, 1280, 720);
    var s = ctx.createLinearGradient(260, 0, 460, 0);
    s.addColorStop(0, 'rgba(255,240,220,0)'); s.addColorStop(0.5, 'rgba(255,240,220,0.045)'); s.addColorStop(1, 'rgba(255,240,220,0)');
    ctx.fillStyle = s; ctx.fillRect(260, 0, 200, 720);
    meander(ctx, 18, 40);
    meander(ctx, 652, 46);
  }

  // ---------- drawing utilities ----------
  function poly(ctx, pts, fill, lw, stroke) {
    ctx.beginPath(); pts.forEach(function (p, i) { if (i) ctx.lineTo(p[0], p[1]); else ctx.moveTo(p[0], p[1]); }); ctx.closePath();
    if (fill) { ctx.fillStyle = fill; ctx.fill(); }
    if (lw) { ctx.strokeStyle = stroke || LN; ctx.lineWidth = lw; ctx.lineJoin = 'round'; ctx.stroke(); }
  }
  function line(ctx, pts, lw, col) {
    ctx.beginPath(); pts.forEach(function (p, i) { if (i) ctx.lineTo(p[0], p[1]); else ctx.moveTo(p[0], p[1]); });
    ctx.strokeStyle = col || LN; ctx.lineWidth = lw; ctx.lineCap = 'round'; ctx.lineJoin = 'round'; ctx.stroke();
  }
  function curve(ctx, a, c, b, lw, col) {
    ctx.beginPath(); ctx.moveTo(a[0], a[1]); ctx.quadraticCurveTo(c[0], c[1], b[0], b[1]);
    ctx.strokeStyle = col || LN; ctx.lineWidth = lw; ctx.lineCap = 'round'; ctx.stroke();
  }
  function disc(ctx, x, y, r, fill, lw, stroke) {
    ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2);
    if (fill) { ctx.fillStyle = fill; ctx.fill(); }
    if (lw) { ctx.strokeStyle = stroke || LN; ctx.lineWidth = lw; ctx.stroke(); }
  }
  // tube segments: outline pass then fill pass so joints are seamless
  function tubes(ctx, segs, lw) {
    ctx.lineCap = 'round'; ctx.lineJoin = 'round';
    segs.forEach(function (s) { line(ctx, s.p, s.w + lw * 2, LN); });
    segs.forEach(function (s) { line(ctx, s.p, s.w, s.c || CL); });
  }

  // hand at wrist w, pointing along absolute angle a (deg), in local figure coordinates
  function hand(ctx, w, a, mode, lw) {
    var palmC = add(w, dirv(a, 7));
    var segs = [{ p: [w, palmC], w: 11 }];
    function fing(off, L, wd) { var base = add(palmC, dirv(a, 3)); segs.push({ p: [add(palmC, dirv(a + off * 0.4, 2)), add(base, dirv(a + off, L))], w: wd || 3.4 }); }
    if (mode === 'open') { fing(-28, 11); fing(-10, 13); fing(8, 13); fing(24, 11); segs.push({ p: [add(palmC, dirv(a + 70, 3)), add(palmC, dirv(a + 55, 11))], w: 3.6 }); }
    else if (mode === 'point') { fing(0, 17, 3.6); fing(-10, 4); fing(-20, 3); segs.push({ p: [add(palmC, dirv(a + 70, 3)), add(palmC, dirv(a + 35, 9))], w: 3.6 }); }
    else if (mode === 'count') { fing(-14, 15); fing(8, 15); fing(28, 4); segs.push({ p: [add(palmC, dirv(a + 70, 3)), add(palmC, dirv(a + 50, 8))], w: 3.6 }); }
    else { // grip / fist
      segs.push({ p: [palmC, add(palmC, dirv(a, 5))], w: 10 });
      segs.push({ p: [add(palmC, dirv(a + 70, 3)), add(palmC, dirv(a + 30, 8))], w: 3.6 });
    }
    tubes(ctx, segs, lw);
    if (mode === 'grip' || mode === 'fist') {
      for (var i = -1; i <= 1; i++) { var q = add(palmC, dirv(a, 4)); var n = dirv(a + 90, i * 2.6); line(ctx, [add(q, n), add(add(q, n), dirv(a - 90, 3))], lw * 0.5, DIL); }
    }
  }

  // ---------- head ----------
  function head(ctx, o, lw) {
    var nose = o.nose || 0, fem = o.fem;
    var P = [[-9, -246], [-17, -262], [-19, -280], [-12, -294], [0, -298], [11, -292], [16, -282], [17, -277],
             [24 + nose, -268 + nose * 0.2], [19, -265], [20, -261], [17, -259], [19, -256], [15, -251], [7, -248], [7, -238], [-8, -238]];
    if (fem) { P[8] = [21, -269]; P[13] = [13, -252]; }
    poly(ctx, P, CL, lw);
    // eye
    ctx.beginPath(); ctx.moveTo(6, -278); ctx.quadraticCurveTo(11, -282, 15, -278); ctx.quadraticCurveTo(11, -275.5, 6, -278);
    ctx.strokeStyle = LN; ctx.lineWidth = lw * 0.8; ctx.stroke();
    disc(ctx, 11.5, -278.3, 1.6, LN);
    if (o.brow) curve(ctx, [5, -285], [11, -289 - o.brow], [16, -284], lw * 0.8);
    // ear
    curve(ctx, [-4, -277], [-10, -271], [-4, -265], lw * 0.7, DIL);
    if (o.mouth === 'open') { ctx.beginPath(); ctx.ellipse(17, -259.5, 3, 2.4, 0, 0, Math.PI * 2); ctx.fillStyle = LN; ctx.fill(); }
    else if (o.mouth === 'grin') curve(ctx, [12, -259], [16, -256], [20, -260], lw * 0.8);
    else if (o.mouth === 'sleep') { curve(ctx, [6, -278], [11, -276], [15, -279], lw); }
    // hair
    if (o.hair === 'bald') {
      poly(ctx, [[-10, -292], [-19, -284], [-20, -268], [-13, -258], [-8, -264], [-11, -276], [-6, -288]], WH, lw * 0.6, LN);
      for (var i = 0; i < 4; i++) curve(ctx, [-16 + i * 2, -284 + i * 5], [-18 + i, -276 + i * 4], [-12 + i, -266 + i * 2], lw * 0.4, DIL);
    } else if (o.hair === 'sakkos') {
      poly(ctx, [[12, -293], [-2, -305], [-26, -300], [-38, -282], [-33, -262], [-16, -256], [-10, -270], [-4, -290]], CL, lw);
      var r = rng(7); for (var k = 0; k < 9; k++) disc(ctx, -28 + r() * 22, -295 + r() * 32, 1.6, LN);
      curve(ctx, [-30, -268], [-20, -260], [-10, -268], lw * 0.7);
      poly(ctx, [[14, -289], [6, -296], [-2, -293], [4, -287], [10, -285]], LN, lw * 0.5, CL);
    } else if (o.hair) {
      poly(ctx, [[15, -287], [6, -299], [-10, -298], [-20, -286], [-19, -266], [-12, -258], [-7, -265], [-7, -280], [4, -289]], LN, lw * 0.6, CL);
      ctx.save(); ctx.globalAlpha = 0.6; for (var j = 0; j < 4; j++) disc(ctx, 8 - j * 7, -294 + j, 1.4, null, lw * 0.4, CL); ctx.restore();
    }
    if (o.beard) {
      var bc = o.beard === 'white' ? WH : LN;
      poly(ctx, [[-6, -262], [-2, -255], [6, -252], [10, -253], [15, -251], [17, -244], [12, -232], [4, -230], [-6, -238], [-10, -250]], bc, lw * 0.6, o.beard === 'white' ? LN : CL);
      if (o.beard === 'white') for (var m = 0; m < 4; m++) curve(ctx, [-2 + m * 4, -252], [0 + m * 4, -242], [2 + m * 3, -233], lw * 0.4, DIL);
      if (o.mouth === 'open') { ctx.beginPath(); ctx.ellipse(16, -258.5, 3, 2.4, 0, 0, Math.PI * 2); ctx.fillStyle = LN; ctx.fill(); }
    }
  }

  // ---------- figure ----------
  // o: x,y (feet/ground origin), s scale, dir 1|-1, lean (rad), tilt (rad), garment 'chiton'|'peplos'|'himation'|'none',
  // armF/armB [upperAngle, foreAngle] absolute degrees (0 = down, 90 = forward, 180 = up), handF/handB modes,
  // legF/legB [thigh, shin], footF/footB angle, sit bool
  function fig(ctx, o) {
    var s = o.s, lw = 2.3 / s, lean = o.lean || 0, tilt = o.tilt || 0;
    var res = {};
    function rotUp(p) { var c = Math.cos(lean), sn = Math.sin(lean), dx = p[0], dy = p[1] + 150; return [dx * c - dy * sn, dy * c + dx * sn - 150]; }
    function W(p, up) { var q = up ? rotUp(p) : p; return [o.x + o.dir * s * q[0], o.y + s * q[1]]; }
    function arm(sh, a) { var e = add(sh, dirv(a[0], 50)); var w = add(e, dirv(a[1], 45)); return [sh, e, w]; }
    function leg(hp, a) { var k = add(hp, dirv(a[0], 74)); var an = add(k, dirv(a[1], 72)); return [hp, k, an]; }
    var aF = arm([7, -237], o.armF || [10, 20]), aB = arm([-7, -237], o.armB || [-8, 0]);
    var lF = leg([6, -150], o.legF || [4, 0]), lB = leg([-6, -150], o.legB || [-6, -2]);
    res.handF = W(add(aF[2], dirv(o.armF ? o.armF[1] : 20, 9)), true);
    res.handB = W(add(aB[2], dirv(o.armB ? o.armB[1] : 0, 9)), true);
    res.kneeF = W(lF[1], false); res.headTop = W([0, -298], true); res.face = W([20, -262], true);
    ctx.save(); ctx.translate(o.x, o.y); ctx.scale(o.dir * s, s);
    function legDraw(L, fa) {
      var ft = add(L[2], dirv(fa == null ? 90 : fa, 17));
      tubes(ctx, [{ p: [L[0], L[1]], w: 19 }, { p: [L[1], L[2]], w: 13 }, { p: [L[2], ft], w: 8 }], lw);
      if (!o.sit) line(ctx, [add(L[1], [-3, -4]), add(L[1], [3, -2])], lw * 0.6, DIL);
    }
    function armDraw(A, mode, up) {
      ctx.save(); if (up) { ctx.translate(0, -150); ctx.rotate(lean); ctx.translate(0, 150); }
      tubes(ctx, [{ p: [A[0], A[1]], w: 13 }, { p: [A[1], A[2]], w: 10 }], lw);
      var a = Math.atan2(A[2][0] - A[1][0], A[2][1] - A[1][1]) * 180 / Math.PI;
      hand(ctx, A[2], a, mode || 'grip', lw);
      ctx.restore();
    }
    armDraw(aB, o.handB, true);
    legDraw(lB, o.footB); legDraw(lF, o.footF);
    // garment
    ctx.save(); ctx.translate(0, -150); ctx.rotate(lean); ctx.translate(0, 150);
    var g = o.garment || 'chiton';
    if (g !== 'none') {
      var hem = o.sit ? -138 : (g === 'chiton' ? -76 : -10);
      var flare = o.sit ? 2 : (g === 'chiton' ? 8 : 16);
      var tp = [[-18, -242], [19, -242], [21, -200], [19, -165], [22 + flare, hem], [-22 - flare, hem], [-22, -170], [-21, -205]];
      poly(ctx, tp, CL, lw);
      var r = rng(o.seed || 3);
      for (var i = 0; i < 6; i++) {
        var x0 = -14 + i * 6 + r() * 2;
        curve(ctx, [x0, -226], [x0 + 2 + r() * 2, (-226 + hem) / 2], [x0 * (1 + flare / 30) + r() * 3, hem - 2], lw * (i % 2 ? 0.45 : 0.6), i % 2 ? DIL : LN);
      }
      if (g === 'peplos') { line(ctx, [[-23, -186], [22, -186]], lw * 0.8); for (var d = -36; d < 38; d += 8) disc(ctx, d, hem - 6, 1.6, LN); line(ctx, [[-36, hem - 11], [38, hem - 11]], lw * 0.5); }
      if (g === 'himation') {
        poly(ctx, [[-19, -240], [-6, -244], [24, -165], [30, -60], [12, -40], [-4, -150]], CL, lw);
        for (var h2 = 0; h2 < 4; h2++) curve(ctx, [-12 + h2 * 4, -232], [8 + h2 * 4, -170], [12 + h2 * 5, -60], lw * 0.5, h2 % 2 ? DIL : LN);
      }
      if (g === 'chiton') line(ctx, [[-21, -172], [20, -172]], lw * 0.7);
    } else {
      poly(ctx, [[-18, -242], [19, -242], [20, -195], [16, -150], [-18, -150], [-21, -200]], CL, lw);
      curve(ctx, [2, -228], [12, -224], [16, -214], lw * 0.5, DIL);
    }
    if (o.sit && g !== 'none') {
      var k = lF[1], long = g !== 'chiton';
      poly(ctx, [[-22, -160], [k[0] - 6, k[1] - 12], [k[0] + 9, k[1] - 3], [k[0] + 8, k[1] + (long ? 58 : 24)], [k[0] - 14, k[1] + (long ? 62 : 28)], [k[0] - 12, k[1] + 6], [-22, -134]], CL, lw);
      for (var f = 0; f < 3; f++) curve(ctx, [-12, -150 + f * 6], [k[0] * 0.5, k[1] - 4 + f * 5], [k[0] - 2 - f * 4, k[1] + 10 + f * 8], lw * 0.5, f % 2 ? DIL : LN);
    }
    // head
    ctx.save(); ctx.translate(0, -242); ctx.rotate(tilt); ctx.translate(0, 242);
    head(ctx, o, lw);
    ctx.restore();
    ctx.restore();
    armDraw(aF, o.handF, true);
    ctx.restore();
    return res;
  }

  // ---------- props ----------
  function moon(ctx, x, y, r) {
    disc(ctx, x, y, r + 3, CL);
    disc(ctx, x, y, r, WH, 2.4, LN);
    ctx.save(); ctx.globalAlpha = 0.28;
    disc(ctx, x - r * 0.3, y - r * 0.2, r * 0.22, DIL); disc(ctx, x + r * 0.25, y + r * 0.3, r * 0.16, DIL); disc(ctx, x + r * 0.35, y - r * 0.35, r * 0.1, DIL);
    ctx.restore();
  }
  function star(ctx, x, y, r) {
    ctx.fillStyle = WH; ctx.beginPath();
    for (var i = 0; i < 8; i++) { var a = i * Math.PI / 4, rr = i % 2 ? r * 0.35 : r; ctx.lineTo(x + Math.cos(a) * rr, y + Math.sin(a) * rr); }
    ctx.closePath(); ctx.fill();
  }
  function stars(ctx, seed, n, box) {
    var r = rng(seed);
    for (var i = 0; i < n; i++) star(ctx, box[0] + r() * (box[2] - box[0]), box[1] + r() * (box[3] - box[1]), 3 + r() * 3);
  }
  function sun(ctx, x, y, r) {
    for (var i = 0; i < 16; i++) { var a = i * Math.PI / 8; line(ctx, [[x + Math.cos(a) * (r + 6), y + Math.sin(a) * (r + 6)], [x + Math.cos(a) * (r + (i % 2 ? 18 : 30)), y + Math.sin(a) * (r + (i % 2 ? 18 : 30))]], i % 2 ? 4 : 6, CL); }
    disc(ctx, x, y, r, CL, 3, LN); disc(ctx, x, y, r * 0.72, null, 1.4, DIL);
  }
  // round crest-case (lophion): cylinder box
  function box(ctx, x, y, w, h, open, lid) {
    var rx = w / 2, ry = w * 0.16;
    poly(ctx, [[x - rx, y], [x - rx, y + h], [x + rx, y + h], [x + rx, y]], CL, 2.4);
    ctx.beginPath(); ctx.ellipse(x, y + h, rx, ry, 0, 0, Math.PI); ctx.fillStyle = CL; ctx.fill(); ctx.strokeStyle = LN; ctx.lineWidth = 2.4; ctx.stroke();
    ctx.fillStyle = CL; ctx.fillRect(x - rx + 1.2, y + h - 2, w - 2.4, 4);
    line(ctx, [[x - rx, y + h * 0.3], [x + rx, y + h * 0.3]], 1.6); line(ctx, [[x - rx, y + h * 0.7], [x + rx, y + h * 0.7]], 1.6);
    for (var i = -rx + 10; i < rx - 4; i += 14) { line(ctx, [[x + i, y + h * 0.35], [x + i + 7, y + h * 0.5], [x + i, y + h * 0.65]], 1.4, DIL); }
    ctx.beginPath(); ctx.ellipse(x, y, rx, ry, 0, 0, Math.PI * 2); ctx.fillStyle = open ? BK : CL; ctx.fill(); ctx.strokeStyle = LN; ctx.lineWidth = 2.4; ctx.stroke();
    if (open) { ctx.beginPath(); ctx.ellipse(x, y, rx - 5, ry - 3, 0, 0, Math.PI * 2); ctx.strokeStyle = CL; ctx.lineWidth = 1.2; ctx.stroke(); }
    if (lid) { // lid lying elsewhere: lid = [lx, ly, rotation]
      ctx.save(); ctx.translate(lid[0], lid[1]); ctx.rotate(lid[2]);
      ctx.beginPath(); ctx.ellipse(0, 0, rx + 3, ry + 2, 0, 0, Math.PI * 2); ctx.fillStyle = CL; ctx.fill(); ctx.strokeStyle = LN; ctx.lineWidth = 2.4; ctx.stroke();
      ctx.beginPath(); ctx.ellipse(0, 0, rx * 0.6, ry * 0.6, 0, 0, Math.PI * 2); ctx.strokeStyle = DIL; ctx.lineWidth = 1.4; ctx.stroke();
      disc(ctx, 0, -3, 5, CL, 2, LN);
      ctx.restore();
    }
  }
  // wax tablet: frame + wax inset; melt in 0..1
  function tablet(ctx, x, y, w, h, rot, melt, seed) {
    ctx.save(); ctx.translate(x, y); ctx.rotate(rot || 0);
    poly(ctx, [[-w / 2, -h / 2], [w / 2, -h / 2], [w / 2, h / 2], [-w / 2, h / 2]], CL, 2.4);
    var m = w * 0.1, r = rng(seed || 11);
    if (!melt) {
      poly(ctx, [[-w / 2 + m, -h / 2 + m], [w / 2 - m, -h / 2 + m], [w / 2 - m, h / 2 - m], [-w / 2 + m, h / 2 - m]], WAX, 1.4);
      for (var row = 0; row < 5; row++) {
        var yy = -h / 2 + m + (row + 0.8) * (h - 2 * m) / 6, xx = -w / 2 + m + 6;
        ctx.beginPath(); ctx.moveTo(xx, yy);
        while (xx < w / 2 - m - 8) { xx += 3 + r() * 4; ctx.lineTo(xx, yy + (r() - 0.5) * 6); }
        ctx.strokeStyle = WAXL; ctx.lineWidth = 1.6; ctx.stroke();
      }
    } else {
      // wax slumped to the lower edge, running over the frame
      var top = -h / 2 + m + (h - 2 * m) * 0.45 * melt;
      poly(ctx, [[-w / 2 + m, -h / 2 + m], [w / 2 - m, -h / 2 + m], [w / 2 - m, h / 2 - m], [-w / 2 + m, h / 2 - m]], '#8b4a28', 1.2);
      ctx.beginPath(); ctx.moveTo(-w / 2 + m, h / 2 - m);
      ctx.lineTo(-w / 2 + m, top + 10);
      for (var i = 0; i <= 8; i++) { var X = -w / 2 + m + i * (w - 2 * m) / 8; ctx.lineTo(X, top + Math.sin(i * 1.7) * 6 + 4); }
      ctx.lineTo(w / 2 - m, h / 2 - m); ctx.closePath(); ctx.fillStyle = WAX; ctx.fill();
      var drips = [[-0.3, 0.9], [-0.05, 1.4], [0.22, 0.7], [0.36, 1.2]];
      drips.forEach(function (d) {
        var X = d[0] * w, L = d[1] * h * 0.42 * melt;
        ctx.beginPath(); ctx.moveTo(X - 6, h / 2 - m); ctx.lineTo(X - 4, h / 2 + L - 6);
        ctx.quadraticCurveTo(X, h / 2 + L + 6, X + 4, h / 2 + L - 6); ctx.lineTo(X + 6, h / 2 - m); ctx.closePath();
        ctx.fillStyle = WAX; ctx.fill(); ctx.strokeStyle = LN; ctx.lineWidth = 1.2; ctx.stroke();
      });
      for (var k = 0; k < 3; k++) curve(ctx, [-w / 2 + m + 10, top + 16 + k * 12], [0, top + 22 + k * 12 + r() * 6], [w / 2 - m - 10, top + 14 + k * 12], 1.3, WAXL);
    }
    line(ctx, [[-w / 2 - 3, -h * 0.18], [-w / 2 - 3, h * 0.18]], 3);
    ctx.restore();
  }
  function stone(ctx, x, y, r) {
    ctx.beginPath(); ctx.ellipse(x, y, r, r * 0.8, 0.3, 0, Math.PI * 2);
    ctx.fillStyle = 'rgba(239,229,207,0.25)'; ctx.fill(); ctx.strokeStyle = WH; ctx.lineWidth = 2.2; ctx.stroke();
    ctx.beginPath(); ctx.ellipse(x - r * 0.3, y - r * 0.25, r * 0.28, r * 0.14, 0.3, 0, Math.PI * 2); ctx.fillStyle = WH; ctx.fill();
  }
  function staff(ctx, top, bot) { line(ctx, [top, bot], 9, LN); line(ctx, [top, bot], 4.5, CL); }
  function column(ctx, x, y0, y1, w) {
    poly(ctx, [[x - w / 2, y0 + 30], [x + w / 2, y0 + 30], [x + w / 2 + 4, y1], [x - w / 2 - 4, y1]], CL, 2.4);
