def splat4(v, cam, acc, t, a, b, present, tau, heat=0.6, double=0, voice=0.0, dz=9.0, S=900.0, expo=1.0, gam=1.0):
    age = present - v.s; born = age >= -0.5
    vis = np.where(born, np.exp(-np.maximum(age, 0) / min(tau, 2.2)), 0) * np.clip(1 + age * 2, 0, 1)
    vis = vis * (1 + 0.9 * voice * np.exp(-np.maximum(age, 0) / 2.0))
    lum_c = v.lum * 0.85 + 0.15
    hw = np.where(born, heat * 0.55 * (1 - np.exp(-np.maximum(age, 0) / 3)) * np.exp(-np.maximum(age, 0) / (tau * 2.5 + 20)), 0)
    X = (v.u - 0.5) * v.aspect * S; Y = (v.v - 0.5) * S; Z = (age + v.rnd - 0.5) * dz
    # arrival: the points gather out of a soft Gaussian cloud, large and dim, then settle small and sharp
    ain = np.clip((t - a) / 2.6 * 1.6 - v.rnd * 0.6, 0, 1); k = (1 - ain) ** 2
    wx, wy = wind(t)
    X = X + v.jit[:, 0] * 150 * k - wx * 60 * k; Y = Y + v.jit[:, 1] * 95 * k + 40 * k; Z = Z + v.jit[:, 2] * 140 * k
    radm = 1 + 6.0 * k
    vis = vis * (0.12 + 0.88 * ain ** 1.5)
    # departure: each point lets go at its own moment; four in ten rise as embers in the wind, cooling; the rest go dark
    td = b - 1.4 + v.r2 * 2.6; tau_e = np.maximum(t - td, 0); gone = t > td
    ember = v.rnd < 0.4
    def disp(te):

def picture(v, cam, t, a, b, present, voice, expo=1.0, gam=1.0, dz=9.0, S=900.0):
    """the source frame itself, through the same camera: it condenses out of static and burns away from a noise front"""
    i0 = int(present); f = present - i0; i1 = min(i0 + 1, v.F - 1)
    fr = v.frames[i0] * (1 - f) + v.frames[i1] * f; h, w = fr.shape[:2]
    cs = np.array([[0, 0], [1, 0], [1, 1], [0, 1]], np.float32)
    P = np.stack([(cs[:, 0] - 0.5) * v.aspect * S - cam.get('cx', 0), (cs[:, 1] - 0.5) * S - cam.get('cy', 0), np.zeros(4) - cam.get('pz', 0) * dz], 1)
    Q = P @ rotmat(cam['yaw'], cam['pitch'], cam.get('roll', 0)).T
    z = Q[:, 2] + cam['dist']; fl = (H / 2) / np.tan(np.radians(cam.get('fov', 50)) / 2)
    dst = np.stack([Q[:, 0] * fl / z + W / 2, Q[:, 1] * fl / z + H / 2], 1).astype(np.float32)
    M = cv2.getPerspectiveTransform((cs * [w, h]).astype(np.float32), dst)
    yy, xx = np.mgrid[0:h, 0:w]; fe = np.clip(np.minimum(np.minimum(xx, w - 1 - xx) / (0.10 * w), np.minimum(yy, h - 1 - yy) / (0.12 * h)), 0, 1) ** 1.5
    pic4 = cv2.warpPerspective(np.ascontiguousarray(np.dstack([fr, fe]).astype(np.float32)), M, (W, H), flags=cv2.INTER_LINEAR, borderValue=0)
    pic, fea = pic4[..., :3], pic4[..., 3]
    ua = min(1.0, max(0.0, (t - a) / 2.8))
    if ua < 1: pic = pic * ua + gaussian_filter(pic, (3 + 16 * (1 - ua), 3 + 16 * (1 - ua), 0)) * (1 - ua)
    pic = np.clip(pic, 0, 1) ** gam * expo
    lum = pic @ np.array([0.3, 0.59, 0.11], np.float32)
    warm = pic * np.array([1.0, 0.84, 0.64], np.float32) * 0.72 + thermal(lum * 0.85 + 0.08) * lum[..., None] ** 0.6 * 0.38
    warm = warm * (0.9 + 0.35 * voice)
    nzo = noisef(int(a * 10)); nzi = noisef(int(a * 10) + 1)
    ain = ease_np(np.clip((t - a) / 2.8 * 1.5 - nzi * 0.5, 0, 1))
    td = b - 1.4 + nzo * 1.7; dt = t - td
    al = ain * ease_np(np.clip(1 - dt / 1.5, 0, 1)) * np.clip(lum * 7, 0, 1) * fea
    edge = np.exp(-((dt - 0.35) / 0.45) ** 2) * 0.38 * (dt > -0.2) * ain * np.clip(lum * 3, 0, 1) * fea   # the burning front
    return warm, al, thermal(0.55 + 0.2 * nzo) * edge[..., None] * 0.75

def resolve4(acc, gain=1.7):
    sig = [0.6, 1.6, 4.2]; C = np.zeros((H, W, 3), np.float32); A = np.zeros((H, W), np.float32)
    for bb in range(3):
        a = acc[bb]
        if not a[..., 3].any(): continue
        bl = np.stack([gaussian_filter(a[..., ch], sig[bb]) for ch in range(4)], -1); C += bl[..., :3]; A += bl[..., 3]
    cov = 1 - np.exp(-A * 1.6 * gain); col = C / np.maximum(A, 1e-6)[..., None]
    return col * cov[..., None], cov

def swarms(t, energy):
    out = np.zeros((H, W, 3), np.float32)
    for i, (a, b) in enumerate(T.AV_T):
        if not (a <= t < b + EMBER_LIFE): continue
        u = (t - a) / (b - a)
        if t < b:
            img = SW.render(T.SWARMS[i], min(u, 1), t, energy); L_ = img @ np.array([0.3, 0.59, 0.11], np.float32)
            heat = thermal(np.clip(L_ * 0.75 + 0.03, 0, 0.8)) * (L_[..., None] ** 0.7) * 0.8 + img * 0.06
            a_ = sm(0, 0.3, u) * (1 - sm(0.6, 1.0, u))
            out = 1 - (1 - out) * (1 - np.clip(heat * a_, 0, 1))
        e = swarm_embers(i); te = t - (a + 0.75 * (b - a))
        if te > 0:
            tau = te * (0.7 + 0.6 * e['r']); cool = np.clip(1 - tau / EMBER_LIFE, 0, 1)
            x = e['x'] + np.sin(tau * 1.5 + e['ph']) * 30 * tau + e['j'][:, 0] * 20 * tau + wind(t)[0] * (tau * 50 + tau ** 2 * 70) * (0.5 + e['r']); y = e['y'] - (tau * 60 + tau ** 2 * 80) * (0.6 + 0.8 * e['r'])
            br = cool ** 1.3 * (0.6 + 0.4 * np.sin(tau * 21 + e['ph'] * 5)) * (0.4 + 0.6 * e['l'])
            ok = (x >= 0) & (x < W - 1) & (y >= 0) & (y < H - 1) & (br > 0.01); ix = x[ok].astype(int) + W * y[ok].astype(int)
            col = thermal(0.25 + 0.75 * cool[ok])
            acc = np.zeros((H, W, 3), np.float32)
            for ch in range(3): acc[..., ch] = np.bincount(ix, br[ok] * col[:, ch], W * H).reshape(H, W)
            acc = gaussian_filter(acc, (1.0, 1.0, 0)) * 3.0; out = 1 - (1 - out) * (1 - np.clip(acc, 0, 1))
    return out

yy, xx = np.mgrid[0:H, 0:W].astype(np.float32); nx = (xx - W / 2) / (W / 2); ny = (yy - H / 2) / (W / 2); r2 = nx * nx + ny * ny

def static(t, amt):
    if amt < 0.01: return 0

# ---------------- the light the song makes: caustics, dome, record, glints, thermal horizon, the last thread ----------------
_c = np.random.default_rng(31)
def _nf(sig):
    f = gaussian_filter(_c.standard_normal((180, 320)), sig); return (f / f.std()).astype(np.float32)
CN = [_nf(7), _nf(7), _nf(9), _nf(9)]
def caustic(t, sharp=0.07):
    f1 = CN[0] * math.cos(0.31 * t) + CN[1] * math.sin(0.31 * t)
    f2 = CN[2] * math.cos(0.23 * t + 1.3) + CN[3] * math.sin(0.23 * t + 1.3)
    c = np.exp(-(f1 / sharp) ** 2) + np.exp(-(f2 / sharp) ** 2) * 0.8
    c = cv2.resize(c.astype(np.float32), (W, H), interpolation=cv2.INTER_CUBIC)
    return np.clip(c, 0, 2)
GOLD = np.array([1.0, 0.76, 0.42], np.float32); COPPER = np.array([0.85, 0.36, 0.12], np.float32)
YY, XX = np.mgrid[0:H, 0:W].astype(np.float32)
def dome(t, amt):
    amt = amt * 1.4
    """close source, distant return: a caustic hemisphere whose rim carries the light"""
    if amt < 0.01: return 0
    cx, cy, Rr = W * 0.5, H * 1.08, W * 0.60
    r = np.sqrt((XX - cx) ** 2 + (YY - cy) ** 2)
    rim = np.exp(-((r - Rr) / 5.0) ** 2) * (0.55 + 0.45 * np.cos((XX - cx) / Rr * 2.2 + t * 0.4))
    shell = np.exp(-np.maximum(Rr - r, 0) / 110) * (r < Rr) * caustic(t * 0.8, 0.06) * 0.55
    return (rim + shell)[..., None] * GOLD * amt
def record(t, amt, thread):
    """the golden record: grooves catch the light in a slow sector; a horizontal line of sound runs into it"""
    if amt < 0.01 and thread < 0.01: return 0
    cx, cy = W * 0.58, H * 0.50; dx, dy = XX - cx, (YY - cy) * 1.35
    r = np.sqrt(dx * dx + dy * dy); th = np.arctan2(dy, dx)
    groove = (0.5 + 0.5 * np.cos(r * 0.85 + 0.6 * np.sin(r * 0.05))) ** 24 * ((r > 0.16 * H) & (r < 0.52 * H))
    lit = np.maximum(np.cos(th - t * 0.45), 0) ** 4 + 0.35 * np.maximum(np.cos(th - t * 0.45 + 3.0), 0) ** 6
    g = groove * lit * np.clip((r - 0.16 * H) / 20, 0, 1)
    line = np.exp(-((YY - cy - 0.0) / 1.6) ** 2) * np.clip(1 - XX / (cx + 0.1 * W), 0, 1) ** 0.6 * (0.6 + 0.4 * np.sin(XX * 0.05 - t * 9) ** 2)
    line = line + np.exp(-((YY - cy) / 9.0) ** 2) * np.clip(1 - XX / (cx), 0, 1) * 0.15
    return (g * 0.7 * amt)[..., None] * GOLD + (line * thread)[..., None] * np.array([1.0, 0.62, 0.25], np.float32)
GL = np.random.default_rng(91)
GLINTS = [dict(x=W * (0.12 + 0.07 * i + 0.03 * GL.random()), y=H * (0.14 + 0.62 * GL.random()), t0=T.SEC['D'][0] + 0.6 * i, rel=2.0 + 1.3 * ((i * 5) % 7)) for i in range(9)]
def glints(t):
    """independent entrances, unequal releases: each glint keeps a horizontal tail of pixels that ends at its own length"""
    out = np.zeros((H, W, 3), np.float32)
    for g in GLINTS:
        u = t - g['t0']
        if u < 0: continue
        life = sm(0, 0.25, u) * (1 - sm(g['rel'], g['rel'] + 1.2, u))
        if life <= 0.005: continue
        x0, y0 = g['x'], g['y']; x1, x2 = int(max(0, x0 - 90)), int(min(W, x0 + 520)); y1, y2 = int(max(0, y0 - 90)), int(min(H, y0 + 90))
        xx = XX[y1:y2, x1:x2] - x0; yy = YY[y1:y2, x1:x2] - y0
        tw = 0.8 + 0.2 * math.sin(t * 7 + x0)
        star = np.exp(-(xx ** 2 + yy ** 2) / 6) * 1.6 + np.exp(-(yy / 1.0) ** 2 - (xx / 70) ** 2) * 0.7 + np.exp(-(xx / 1.0) ** 2 - (yy / 55) ** 2) * 0.7
        halo = np.exp(-(xx ** 2 + yy ** 2) / 260) * 0.35
        L_ = min(1.0, u / 2.5) * 460 * (0.5 + 0.5 * ((g['x'] * 13) % 1))
        pix = (xx > 4) & (xx < L_) & (np.abs(yy) < 1.6) & (((xx.astype(np.int32) // 4) % 3) == 0)
        tail = pix * np.clip(1 - xx / max(L_, 1), 0, 1) ** 1.3 * 0.9
        blk = star[..., None] * np.array([1.0, 0.95, 0.85]) * tw + halo[..., None] * np.array([0.8, 0.35, 0.7]) + tail[..., None] * np.array([1.0, 0.66, 0.3])
        out[y1:y2, x1:x2] += blk.astype(np.float32) * life
    return out
def horizon(amt):
    if amt < 0.01: return 0
    return (np.exp(-(H - YY) / 70.0) * (0.75 + 0.25 * np.sin(XX * 0.004)))[..., None] * COPPER * amt
def last_thread(t):
    """after everything: a hairline that breaks into four pixels refusing to go dark"""
    a0 = T.END - 4.2
    if t < a0: return 0
    u = t - a0; y = H * 0.30; x0, x1 = W * 0.30, W * 0.30 + W * 0.42 * (1 - sm(1.2, 2.6, u) * 0.85)
    th = np.exp(-((YY - y) / (0.9 - 0.4 * sm(0, 2.0, u))) ** 2) * ((XX > x0) & (XX < x1)) * sm(0, 0.8, u) * (1 - sm(1.4, 2.6, u))
    out = th[..., None] * np.array([1.0, 0.6, 0.22], np.float32) * 0.8
    for k in range(4):
        px = W * (0.44 + 0.052 * k); pa = sm(1.6 + 0.1 * k, 2.2, u)
        out = out + (np.exp(-((XX - px) ** 2 + (YY - y) ** 2) / 2.2) * pa)[..., None] * np.array([1.0, 0.8, 0.45], np.float32)
    return out

def frame(t):
    voice, energy, low, hk = env_full(t)
    img = np.zeros((H, W, 3), np.float32) + BG
    img = img + sky(t, energy)
    A0, A1 = T.SEC['A']; B0, B1 = T.SEC['B']; C0, C1 = T.SEC['C']; D0, D1 = T.SEC['D']
    # under the pictures: thermal mass, the dome in the groove, the record in the break
    img = img + horizon((sm(B0 - 0.5, B0 + 1.5, t) * (1 - sm(D1 - 0.5, D1 + 0.8, t))) * (0.12 + 0.4 * low))
    img = img + dome(t, sm(B0 - 0.4, B0 + 2.0, t) * (1 - sm(B1 - 0.6, B1 + 1.2, t)) * (0.25 + 0.55 * low) + sm(D0 - 0.3, D0 + 1.5, t) * (1 - sm(D1 - 0.3, D1 + 0.6, t)) * (0.2 + 0.4 * low))
    rec = sm(C0 - 1.0, C0 + 1.5, t) * (1 - sm(C1 - 0.4, C1 + 0.8, t))
    img = img + record(t, rec * (0.7 + 0.5 * hk), rec * (0.25 + 0.75 * hk))
    if t < T.SILENCE:
        acc = new_acc(); any_ = False
        live = [s_ for s_ in T.SHOTS if s_[1] - 0.1 <= t < s_[2] + 1.4 + EMBER_LIFE]
        for s_ in sorted(live, key=lambda q: -q[1]):          # the incoming picture lies under the one that is burning away
            k, a, b, c0, c1, o = s_
            v = vol(k); u = ease((t - a) / (b + 1.4 - a)); cam = {kk: lerp(c0[kk], c1[kk], u) for kk in c0}
            present = min(v.F - 1, max(0.0, (t - a + 0.4) * T.FPS * o.get('rate', 0.5)))
            vv = voice if any(a < vt < b for (_, _, _, vt, _) in T.VOICES) else 0.0
            pw, pa, pe = picture(v, cam, t, a, b, present, vv, o.get('exp', 1.0), o.get('gam', 1.0))
            pw = pw * (0.92 + 0.16 * low)
            img = img * (1 - pa[..., None] * 0.92) + pw * pa[..., None] + pe
            splat4(v, cam, acc, t, a, b, present, o.get('tau', 10), o.get('heat', 0.6), o.get('double', 0), vv, expo=o.get('exp', 1.0), gam=o.get('gam', 1.0)); any_ = True
        if any_:
            c, cov = resolve4(acc); img = img * (1 - cov[..., None]) + c
            img = img + gaussian_filter(np.clip(img - 0.75, 0, None), 10) * 0.35
        img = img + record(t, rec * 0.22, 0) if rec > 0.01 else img
        img = img + static(t, 0.03)
        if t >= T.AV0 - 0.2: img = 1 - (1 - np.clip(img, 0, 1)) * (1 - swarms(t, energy))
        img = img + glints(t)
    else:

    img = img + last_thread(t)

    img = np.clip(img, 0, 1.4)[DOME] * VIG * DOMEM
    gx, gy = int(t * 97) % 64, int(t * 53) % 64
    img = img + GRAIN[gy:gy + H, gx:gx + W, None] * 0.016
    img = img * (1 - sm(T.END - 0.8, T.END, t)) * sm(0.0, 1.0, t + 0.2)
    return np.clip(img, 0, 1)
